\begin{problem}
Prove that all following languages are undecidable.

\begin{enumerate}
    \item Prove that there is no Turing machine that decides the language
    \[\text{EQUAL} := \{ \left( \langle M_1 \rangle, \langle M_2 \rangle \right) \mid  L(M_1) = L(M_2) \text{ for Turing machine $M_1$ and $M_2$}\}\]
    \item Prove that there is no Turing machine that decides the language
    \[\text{REG} := \{  \langle M \rangle \mid  L(M) \text{ is regular}\}\]
\end{enumerate}
\end{problem}
