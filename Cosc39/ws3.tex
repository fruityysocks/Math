\begin{problem}{1}
Construct an $\varepsilon$DFA to recognize the language
\[
L_1 := \{ w \in \{0,1\}^* : w \text{ starts with a } 1 \text{ and ends with a different } 1 \}.
\]
\end{problem}

\begin{solution}
    \bbni 
    The DFA rejects all strings that start with a zero, and then uses $\varepsilon$-transitions to check if the last symbol is a $1$.  
    \begin{center}
        \begin{tikzcd}
        {} \arrow[r] &  \circled{\phantom{1}} \arrow[r, "1"] & \circled{\phantom{1}} \arrow[out=75, in=100, loop, swap, "\Sigma"] \arrow[r, "\varepsilon"] & \circled{\phantom{1}} \arrow[r, "1"] & \dcircled{\phantom{1}}  
    \end{tikzcd}
    \end{center}
\end{solution}

\begin{problem}{2}
Construct an $\varepsilon$DFA to recognize the language
\[
L_2 := \{ w \in \{0,1\}^* : w \text{ starts with } 1 \text{ or ends with } 1 \}.
\]
\textit{Hint: Sure you can do product construction. But can you make a smaller $\varepsilon$DFA?}
\end{problem}


\begin{solution}
    \bbni
    The DFA accepts all strings that start with $1$. If a string does not start with $1$ it uses $\varepsilon$-transitions to check if the last symbol is a $1$. 
    \begin{center}
        \begin{tikzcd}
        {} & {} & \dcircled{\phantom{1}} \arrow[out=35, in=75, loop, swap, "\Sigma"] \\ 
        {} \arrow[r] &  \circled{\phantom{1}} \arrow [ur, "1"] \arrow[r, "0"] & \circled{\phantom{1}} \arrow[out=75, in=100, loop, swap, "\Sigma"] \arrow[r, "\varepsilon"] & \circled{\phantom{1}} \arrow[r, "1"] & \dcircled{\phantom{1}}  
    \end{tikzcd}
    \end{center}
\end{solution}

\begin{problem}{3}
    Construct an $\varepsilon$DFA to recognize the language that contains all binary strings that can be decomposed into substrings, where each substring starts with a $1$ and ends with a different $1$. In notation, $L_3$ is just the Kleene star of $L_1$, that is, $L_3 := L_1^*$.
\end{problem}

\begin{solution}
    \bbni 
    The DFA accepts the empty string, rejects all strings that start with a zero, and then uses $\varepsilon$-transitions to check if the last symbol is a $1$.  

     \begin{center}

    \end{tikzcd}
    \end{center}
\end{solution}

\begin{problem}{4}
Construct an $\varepsilon$DFA to recognize the language of the expression $((00)^*(0+1)^*1)^*$.
\end{problem}


\begin{solution}
    \bbni 
    The DFA accepts the empty string, and rejects all strings that don't end with a $1$.

\begin{problem}{6}\textit{[To think about later]}
For a string $w \in \{0,1\}^*$, $\mathrm{rev}(w)$ is $w$ written backwards (or $\varepsilon$ if $w = \varepsilon$). For a language $L$
recognized by some DFA $D$, define $\mathrm{rev}(L) := \{\mathrm{rev}(x) : x \in L\}$. Design an $\varepsilon$DFA to recognize
$\mathrm{rev}(L)$.
\end{problem}
