\begin{problem}{1}
Recall that the spectrum and maximal spectrum of a ring $R$ are the sets:
\[
\Spec(R) \coloneqq\left\{ P \subset R \;\; \big|\;\; \text{$P$ is a prime ideal}\right\}\]
and 
\[\mSpec(R) \coloneqq\left\{ P \subset R \;\; \big|\;\; \text{$P$ is a maximal ideal}\right\} 
\]
Given any subset $\cS \subset R$ define 
\[
\V(\cS) \coloneqq \left\{P \in \Spec(R) \;\; \big| \;\; \cS \subset P\right \}.
\]
\begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
	\item Describe $\Spec(\K)$ and $\mSpec(\K)$ when $\K$ is a field. 
	\item Describe both $\Spec(\K[x]/\langle x^{3}\rangle)$ and $\mSpec(\K[x]/\langle x^{3}\rangle)$ when $\K$ is a field. 
	\item For an arbitrary ring $R$, compute $\V(\{0\})$ and $\V(\{1\})$. 
	\item Explain the natural poset structure on $\Spec(R)$. Describe the poset for $\Spec(\Z)$. 
\end{enumerate}
\end{problem}

\begin{solution}
    \bbni 
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Let $\K$ be a field. Then $\K$ only has $1$ proper ideal: the ideal generated by $0$. Note that this ideal is prime because $\K$ is an integral domain. Moreover it is also maximal because $\K/\langle 0 \rangle = \K$ is a field. So, 
        \[\Spec(\K) = \mSpec(\K) = \{\langle 0 \rangle\}\]

        \item Note that the ideal generated by $0$ is not prime in this ring because $x \cdot x^2 \in \langle 0 \rangle $ even though neither $x$ nor $x^2$ are in $\langle x\rangle$. Then we get, \[\Spec(\K[x]/\langle x^3 \rangle) =  \mSpec(\K[x]/\langle x^3 \rangle) = \{\langle x \rangle\}\] 
        
        \item Let $R$ be an arbitrary ring, then, 
        \[\V(\{0\}) = \Spec(R)\]
        since every ideal contains $0$ and, 
        \[\V(\{1\}) = \emptyset\]
        since $1$ is not contained in any proper ideal of $R$.
        
        \item 
        The natural poset structure on $\Spec(R)$ is given by inclusion of ideals, that is, for $P_1, P_2 \in \Spec(R)$, we write $P_1 \le P_2$ iff $P_1 \subseteq P_2$.

        The poset for $\Spec(\Z) = $ contains prime ideals in $\Z$. The zero ideal is contained in every other ideal and every other prime ideal is maximal. 
        
    \end{enumerate}
\end{solution}

A \emph{topological space} is a set $X$ together with a collection of subsets $\tau$, which satisfies the following conditions: i) $\varnothing \in \tau$ and $X \in \tau$, ii) $\tau$ is closed under arbitrary unions, and iii) $\tau$ is closed under finite intersections. We call $\tau$ a \emph{topology} on $X$, the elements of $\tau$ the open subsets of $X$, and a subset $Z \subset X$ is closed if and only if $X\setminus Z$ is open (i.e., $X \setminus Z \in \tau$). If $S \subset X$ is a subset, the \emph{subspace topology} on $S$ is the topology where the open sets of $S$ are all sets of the form $S \cap U \subset S$ where $U \subset X$ is open in $X$. A map $f: X \to Y$ of topological spaces is \emph{continuous} if and only if $f^{-1}(U) \subset X$ is open for all open subsets $U \subset Y$, i.e., the preimage of an open set is open.

\begin{problem}{2}
\textbf{The Zariski Topology.} We will now place a topology on the set $\Spec(R)$,  called the Zariski topology after Oscar Zariski. 
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Prove that if  $\cS, \cT \subset R$ are arbitrary sets then $\V(\cS) \cap \V(\cT) = \V(\cS \cup \cT)$. Explain why your argument extends to an \emph{arbitrary} intersection. 
        \item Prove that if  $\cS, \cT \subset R$ are arbitrary sets then $\V(\cS) \cup \V(\cT) = \V(\cS \cdot \cT)$ where 
    \[
    \cS \cdot \cT \coloneqq \{ st \;\; |\;\; s \in \cS,\; t \in \cT\}.
    \]
        \item Prove that the sets of the form $\V(\cS)$ are the closed sets of a topology on $\Spec(R)$. We call this topology the Zariski topology on $\Spec(R)$. (Hint: Rephrase the definition of a topology in terms of closed sets instead of open sets.) 
        \item Take a break and Google Oscar Zariski and read about his life. 
    \end{enumerate}
\end{problem}
    
\begin{solution}
    \bbni 
    \begin{enumerate} [label=(\alph*),ref=\theenumi\alph*]
        \item We prove this for arbitrary intersections. Let $\cS_i$, $i \in \Lambda$ where $\Lambda$ is some indexing set, be arbitrary subsets of $R$. Then by definition,
        \[\underset{i\in \Lambda}{\bigcap}\V(\cS_i) = \underset{i \in \Lambda}{\bigcap}\{[P] \in \Spec(R), \cS_i \subseteq P\} = \left \{[P] \in \Spec(R), \underset{i\in \Lambda}{\bigcup}\cS_i \subseteq P \right \} \]
        And 
        \[\V \left (\underset{i \in \Lambda}{\bigcup} \cS_i  \right ) = \left \{[P] \in \Spec(R), \underset{i \in \Lambda}{\bigcup} \cS_i \subseteq P \right \} \]
        That is, $ \underset{i\in \Lambda}{\bigcap}\V(\cS_i) = \V \left (\underset{i \in \Lambda}{\bigcup} \cS_i  \right )$. Hence proved.

        \item Let $\cS, \cT$ be arbitrary subsets of $R$. Then by definition, 
        \[\V(\cS \cdot \cT) = \{[P] \in \Spec(R), \, \cS\cdot \cT \subseteq P\}\]
        And 
        \[\V(\cS) \cup \V(\cT) = \{[P] \in \Spec(R), \cS \subseteq P\} \cup \{[P] \in \Spec(R), \cT \subseteq P\}\] 
        Note that $\cS \cdot \cT \subset \cS, \cT$. Then the containment $\V(\cS \cdot \cT) \subseteq \V(\cS) \cup \V(\cT)$ is obvious. 
        
        To prove that $\V(\cS \cdot \cT) \supseteq \V(\cS) \cup \V(\cT)$, it suffices to show that if $\cS \cdot \cT \subseteq P$ then either $\cS \subseteq P$ or $\cT \subseteq P$. 
        
        Assume $\cS \cdot \cT \subseteq P$. Let $s \in \cS$ and $t \in \cT$. Then, since $P$ is a prime ideal and $st \in P$ then either $s \in P$ or $t \in P$. 
        If both $s, t \in P$ we are done. Else wlog, assume that $s \notin P$.
        But note that $\forall t' \in \cT$, $s\cdot t' \in P$. Hence, $t' \in P, \, \forall t' \in \cT$, i.e. $\cT \subseteq P$. 

        Hence, $\V(\cS) \cup \V(\cT) = \V(\cS \cdot \cT)$. 

        \item To prove that the sets of the form $\V(\cS)$ are the closed sets of the Zariski topology on $\Spec(R)$ we must show that (i) $\emptyset, \, \Spec(R)$ are closed sets, (ii) the finite union of closed sets is closed, and (iii) arbitrary intersection of closed sets is closed. 
        
        \begin{enumerate}
            \item [(i)] Note that $\emptyset = \{\V(\{1\})\}$ since $1$ is not contained in any proper ideal of $R$, and $\Spec(R) = \{\V(\{0\})\}$ since $0$ is contained in every ideal and thus in every prime ideal. Both $\V(\{1\})$ and $\V(\{0\})$ are closed sets, hence $\emptyset$ and $\Spec(R)$ are closed. 
            \item [(ii)] From part (b) we know that the finite union of two closed sets is closed, and by induction we can extend this to a finite number of closed sets. Hence, the finite union of closed sets is closed. 
            \item [(iii)] From part (a) we know that any arbitrary intersection of closed sets is closed. 
        \end{enumerate}
        Hence, sets of the form $\V(\cS)$ are the closed sets of the Zariski topology on $\Spec(R)$.
    \end{enumerate}    
\end{solution}

A function $f: X \to Y$ of topological spaces is \emph{continuous} if and only if $f^{-1}(U) \subset X$ is open for all open subsets $U \subset Y$, i.e., the preimage of an open is open. A \emph{homeomorphism} $f: X \to Y$ is a continuous map of topological spaces with a continuous inverse $f^{-1}:Y \to X$. A useful fact is that a continuous map $f:X \to Y$ is a homeomorphism if and only if it is bijective and $f(U) \subset Y$ is open for every open subsets $U \subset X$.

A \emph{category} is a collection of objects together with, for each pair of objects, a set of morphisms between them, equipped with an associative composition law and an identity morphism for each object. A \emph{functor} between categories $\textbf{C}$ and $\textbf{D}$ is a mapping that assigns to each object of $\textbf{C}$ an object of $\textbf{D}$ and to each morphism in $\textbf{C}$ a morphism in $\textbf{D}$, preserving identity morphisms and composition. A functor is \emph{contravariant} if it reverses the order of composition and \emph{covariant} if it preserves the order of composition.
