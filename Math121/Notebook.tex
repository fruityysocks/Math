\begin{problem}{1}
	 Let $R$ be a ring and consider $R[[t]]$ the ring of formal power series with coefficients in $R$. The goal of this exercise is prove that:
	 \[
	 \left(R[[t]]\right)^{\times} = \left\{ u + tf(t) \;\; \big| \;\; u \in R^{\times}, \;\; f(t) \in R[[t]] \right\}.
	 \]
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
	 	\item Elements of $R[[t]]$ can be written as $f=\sum_{k=0}^{\infty} f_k t^k$ for $f_{k} \in R$. We call $f_0$ the constant (or lead) term of $f$. If $f=\sum_{k=0}^{\infty} f_k t^k$ and $g=\sum_{k=0}^{\infty} g_k t^k$ then find a formula for the lead term of $fg$. 
		\item Using the formula from (a) prove that if $f$ is a unit then $f_0 \in R^{\times}$. 
		\item Find a formula for the coefficient of $t^k$ appearing in $fg$. 
		\item Assume that $f_0 \in R^{\times}$. Prove that $f$ is a unit by explicitly constructing an element $g \in R[[t]]$ such that $fg = 1$. (Hint: the condition that $fg = 1$ places constraints on each coefficient of $fg$. Use the constraints together with part (c) to recursively solve for the coefficients of $g$.)	
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni 
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Formula for the constant term of $fg$ is $f_0g_0$. 
        
        \item Assume $f$ is a unit. Then there exists some $f' \in R[[t]]$ such that $ff' = 1$.  By part (1), the constant term of $ff'$ is $f_0f'_0$. The constant term of $1$ is $1$, so $f_0f'_0 =1$. Hence $f_0$ is a unit, and $f_0 \in R^\times$.  
        
        \item The formula for the coefficient of $t^k$ is $\sum_{i=0}^kf_ig_{k-i}$.
        
        \item Assume $f_0 \in R^\times$. Let its inverse be $g_0$. For $f$ to be a unit, every non-leading coefficient of $fg$ must be $0$. That is, 
        \[fg = f_0g_0 + \sum_{k=1}^\infty\left(\sum_{i=0}^kf_ig_{k-i}t^k\right)\]
        where $\sum_{i=0}^kf_ig_{k-i} = 0$ for all $k$. Then solving for $g_k$ gives us, 
        \[g_k = -\frac{1}{f_0}\sum_{i=0}^{k-1}f_ig_{k-i} = -g_0\sum_{i=0}^{k-1}f_ig_{k-i}\]
    \end{enumerate}
\end{solution}

\begin{problem}{2}
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
		\item Prove that an element $u \in R$ is a unit if and only if the principal ideal $\langle u \rangle = R$.
		\item Prove that if $ab$ and $b$ are units, then $a$ is a unit.
		\item Prove that if $u$ is a unit and $a \in R$, then $a$ and $ua$ generate the same principal ideal.
		\item In the ring $\Z/ \langle 12 \rangle$, determine which elements are units.
	\end{enumerate}
\end{problem}

\begin{solution}
    \bbni 
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item $(\implies)$ Assume $u \in R$ is a unit. 
        
        Then there exists some $u' \in R$ such that $uu' = 1 \in \langle u\rangle$. If $1$ is in the ideal generated by $u$ then $\forall r \in R, \, r \in \langle u \rangle$. Hence $\langle u \rangle = R$.  
        
        $(\impliedby)$ Assume $\langle u \rangle = R$ for some $u \in R$. 
        
        Then $1 \in \langle u \rangle$. That is, by definition of an ideal, $1$ is a multiple of $u$. Hence, there exists some $u' \in R$ such that $uu' =1$. Hence, $u$ is a unit.  

        \item  Let $a, b \in R$, and $ab, \, b$ be units. Then there exist $x, b^{-1} \in R$ such that 
        \[abx = 1, \quad bb^{-1} = 1\]
        Then we have, 
        \begin{align*}
            abx = bb^{-1} \implies 
            \frac{abx}{bb^{-1}} = 1  \implies 
            a \cdot \frac{x}{b^{-1}} = 1
        \end{align*}

         Hence, $a$ is a unit and $\frac{x}{b^{-1}}$ is its multiplicative inverse.

        \item  Let $u \in R^\times, a \in R$.  
        
      

        Note that the containment $\langle ua \rangle \subseteq \langle a \rangle$ holds by definition. So we only need to show that 
        $\langle a \rangle \subseteq \langle ua \rangle$. Since $u$ is a unit, there exists some $u^{-1} \in R$ such that $uu^{-1} =1$. Then, note that $uu^{-1}a = a \in \langle ua \rangle$, i.e. $\langle a \rangle \subseteq \langle ua \rangle$. 

        Hence $a$ and $ua$ generate the same principal ideal. 

        \item  $(\Z/\langle 12 \rangle)^\times  = \{1, \, 5, \, 7, \, 11\}$
    \end{enumerate}
\end{solution}

\begin{problem}{3}
    Given two rings $R$ and $S$ we let $\Hom_{\text{Ring}}(R,S)$ be the set of ring homomorphism from $R$ to $S$. When there is little room for confusion we will often drop the subscript $\text{Ring}$ and simply write $\Hom(R,S)$. 
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
		\item Prove or disprove: If $R$ and $S$ are rings, then $\Hom(R,S)$ is a ring under point-wise multiplication and point-wise addition. 
		\item Give an explicit description of the hom-sets below:
		\begin{enumerate}
			\item $\Hom(\Z, \Z)$
			\item $\Hom(\Z/\langle 2 \rangle, \Z)$
			\item $\Hom(\Z, \Z/\langle 2 \rangle)$
			\item $\Hom(\Z, \C[x])$
		\end{enumerate}
		\item Based on part (b) state and prove a characterization of $\Hom(\Z,R)$ for any ring $R$. 
	\end{enumerate}
\end{problem}

\begin{solution}
    \bbni 
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Let $f, g \in \Hom(R,S)$. Note that since ring homomorphisms are unital $f(1_R) = 1_S$ and $g(1_R) = 1_S$. Assume for the sake of contradiction that $\Hom(R,S)$ is a ring under pointwise addition. Then, \[1 = (f+g)(1) = f(1_R) + g(1_R) = 1 + 1 = 2\]
        
        A contradiction! Hence $\Hom(R,S)$ is not a ring under pointwise addition and pointwise multiplication. 

        \item  
        \begin{enumerate}[label=(\roman*),ref=\theenumi\roman*]
           \item $\Hom(\Z, \Z) = \{\id\}$ 
			\item $\Hom(\Z/\langle 2 \rangle, \Z) = \emptyset$
			\item $\Hom(\Z, \Z/\langle 2 \rangle) = \{\Z \twoheadrightarrow \Z/\langle 2 \rangle \}$
			\item $\Hom(\Z, \C[x]) = \{\Z \hookrightarrow \C[x]\}$
        \end{enumerate}

        \item Note that since both $\Z$ and $R$ are rings, any homomorphism $\phi: \Z  \to R$ is unital, i.e. $\tau(1) = 1_R$. To be a homomorphism we must have $\forall a \in \Z$, 
        \[\tau(a) = \tau\left(\sum_{i=1}^a 1\right) = \sum_{i=1}^a\tau (1)\]
        This completely determines the homomorphism, and thus there is a unique (unital) homomorphism from $\Z$ to any ring $R$. 
    \end{enumerate}
\end{solution}

\begin{problem}{4}
    Let $\phi:R \to S$ be a ring homomorphism.
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
		\item Prove that if $J \subset S$ is an ideal then $\phi^{-1}(J) \subset R$ is an ideal.
		\item Assuming $\phi$ is injective, give a nice description of $\phi^{-1}(J)$ for an ideal $J \subset S$. 
		\item Find a counterexample showing that if $I \subset R$ is an ideal $\phi(I)$ need not be an ideal.
		\item Prove that if $\phi$ is surjective then $\phi(I)$ is an ideal of $S$. 
		\item Let $I \subset R$ be an ideal. Prove that if $\ker(\phi) \subset I$ then $\phi^{-1}(\phi(I))=I$.
		\item Is the assumption that $\ker(\phi) \subset I$ needed in part (5)?
	\end{enumerate}
\end{problem}

\begin{solution}
    \bbni

    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Let $a, b \in \phi^{-1}(J)$. Then $\phi(a), \phi(b) \in J$. Since ideals are closed under addition, $\phi(a) + \phi(b) = \phi(a+b) \in J$. Thus $\phi^{-1}(a+b) \in \phi^{-1}(J)$. 
        
        Similarly let $r \in R$, and $a \in \phi^{-1}(J)$. Then $\phi(r) \cdot \phi(a)  = \phi(r \cdot a)\in J$. Thus, $\phi^{-1}(r \cdot a) \in \phi^{-1}(J)$. 

        \item If $\phi$ is injective then, 
        \[\phi^{-1}(J) = \{r \in R \, | \, \phi(r) \in J \}\]

        \[\phi^{-1}(J) = J \cap \phi(R) = J \cap R\]

        \item Let $R = \Z$, $S = \Z[x]$ and define the homomorphism to be $\phi: R \hookrightarrow S$. Then consider the image of the ideal generated by $3$. It does not contain $3x$ and hence is not closed under multiplication.  
        
        \item Let $I \subset R$ be an ideal and assume that $\phi$ is surjective. Let $a, b \in I$. We want to show that $\phi(I)$ is a subgroup of the additive group of $S$ and is closed under multiplication by elements of $S$. 
        
        Consider $\phi(a) - \phi(b)$. Note that since $\phi$ is a homomorphism the following holds,
        \[\phi(a) - \phi(b) = \phi(a-b)\]
        and since $a-b \in I$, $\phi(a-b) \in \phi(I)$. Moreover since $\phi(0) = 0 \in \phi(I)$, $\phi(I)$ is a subgroup of $S$ under addition. 

        Now note that since $\phi$ is surjective, $\forall s \in S \, \exists r \in R$ such that $s = \phi(r)$. Then since $a \in I$, $r \in R$ and $I$ is an ideal of $R$ we have $ra \in I$. So, 
        \[s \cdot \phi(a) = \phi(r) \phi(a) = \phi (ra) \in \phi(I) \]
        Hence, $\phi(I)$ is closed under multiplication by elements of $S$ and thus is an ideal of $S$.
        
        \item Assume $\ker(\phi) \subset I$. 
        
        $I \subset \phi^{-1}(\phi(I))$

        Since $\phi^{-1}(\phi(I))$ is the pre-image of $\phi(I)$, this containment is straightforward.

        
        $\phi^{-1}(\phi(I)) \subseteq I$

        Let $x \in \phi^{-1}(\phi(I))$. That is $\phi(x) = \phi(a)$ for some $a \in I$. 

        \[\phi(x-a) = 0 \implies x-a \in \ker(\phi) \subset I\]
        
        Since $I$ is closed under addition $x \in I$. Hence, 
        $\phi^{-1}(\phi(I)) \subseteq I$

        \item Yes. As a counterexample, consider the quotient map $\phi: \Z \to \Z/\langle 3 \rangle$ and the ideal generated by $ 2$, $I = \langle {2} \rangle$. Note that $ 0 \in \phi(I) \implies \ker(\phi) = 3\Z \subset \phi^{-1}(\phi(I))$. 
        That is $3 \in \phi^{-1}(\phi(I))$ but $3 \notin I$. 
    \end{enumerate}
\end{solution}

\begin{problem}{1} 
    \textbf{Operations on Ideals.} 
    
    Throughout this problem let $I$ and $J$ be ideals in a ring $R$.
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
		\item Prove or disprove that $I \cap J$ is an ideal. 
		\item Prove or disprove that $I \cup J$ is an ideal.
		\item Prove or disprove that the product of ideals, defined below, is an ideal:
		\[
		IJ \coloneqq \left\{ a_1b_1+\cdots+a_tb_t\;\; \big| \;\; a_i\in I\;\; b_i \in J, \; \; \text{for all $t$}\right\}.
		\]
		\item Prove or disprove that the sum of ideals, defined below, is an ideal:
		\[
		I + J \coloneqq \left\{ a + b \;\; \big| \;\; a \in I\;\; b \in J\right\}.
		\]
		\item Prove or disprove that the radical of an ideal, defined below, is an ideal:
		\[
		\sqrt{I} \coloneqq \left\{ a \in R \;\; \big| \;\; a^{k} \in I \text{ for some $k\in \Z_{\geq1}$} \right\}.
		\]
	\end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item $I \cap J$ is an ideal. 
        
        Let $a, b \in I \cap J$, $r \in R$.  

        \begin{enumerate}[label=(\roman*),ref=\theenumi\roman*]
            \item $I \cap J$ is a subgroup because intersection of subgroups is a subgroup. 
            \item Note that $a, b \in I, J$. Since $I$ and $J$ are closed under addition and inverses, $a - b \in I, J$. Hence $a - b \in I, J$
            \item $r \cdot a \in I \cap J$ since $ra \in I$ and $ra \in J$. 
        \end{enumerate}

        Hence $I \cap J$ is an additive subgroup of $R$ and is closed under multiplication from elements of $R$. 

        \item $I \cup J$ is not an ideal. 
        
        Consider the ideals $I = \langle 2 \rangle, \, J = \langle 3 \rangle$ in the ring $R = \Z$. Note that $5 \notin I \cup J$ even though $2,3 \in I \cup J$. Hence, $I \cup J$ is not closed under addition and is not an ideal.

        \item $IJ$ is an ideal. 
        
        Let $s\in IJ$, $r \in R$.  

        \begin{enumerate}[label=(\roman*),ref=\theenumi\roman*]
            \item By definition, $IJ$ includes all finite sums of elements in $I, J$. So it contains $0$ and is closed under addition and inverses. Hence it is an additive subgroup. 
            \item Consider $r \cdot s \in I J$. 
            \[r \cdot s = r (a_1b_1+\cdots+a_tb_t) = r \cdot a_1b_1+\cdots+ r \cdot a_tb_t\]

            Note that $r \cdot a_i \in I$ for all $i$ and thus $rs \in IJ$ for all $r$.  
        \end{enumerate}
        Hence $I J$ is an additive subgroup of $R$ and is closed under multiplication from elements of $R$. 

        \item $I + J$ is an ideal. 
        
        Let $p = a_1+b_1, \, q = a_2 + b_2 \in I + J$, $r \in R$.  

        \begin{enumerate}[label=(\roman*),ref=\theenumi\roman*]
            \item Note that $0 \in I + J$ since $0 \in I, J$. 
            \item Consider $p - q$. 
            \[p + q  = a_1+b_1 - (a_2 + b_2 ) = (a_1 - a_2) + (b_1 - b_2)\]

            Note that $(a_1 - a_2) = a$ for some $a\in I$  and $(b_1 - b_2) = b$ for some $b \in I$. Hence, $I + J$ contains zero and is closed under addition and inverses and thus is an additive subgroup. 

            \item Consider $r \cdot p$. 
            \[r \cdot p = r \cdot (a_1 + b_1) = r a_1 + r b_1\]

            Note that $ra_1 \in I$ and $rb_1 \in J$ by definition of ideals. Hence $rp \in I + J$. 
        \end{enumerate}
        Thus $I+J$ is an additive subgroup of $R$ and is closed under multiplication from elements of $R$. 

        \item $\sqrt{I}$ is an ideal. 
        
        Let $a^n, b^m \in I$, and $a, b, r \in R$.  

        \begin{enumerate}[label=(\roman*),ref=\theenumi\roman*]
            \item Note that $0 \in \sqrt{I}$ since $0^1 \in I$. 
            \item Consider $(a-b)^{nm}$. 
            \[(a-b)^{nm} = \sum_{i=0}^{nm} \binom{nm}{i}(a)^i(-b)^{nm-i}\]

            \begin{itemize}
                \item If $i \geq n$, then $a^n \in I \implies  \binom{nm}{i}(a)^i(-b)^{nm-i} \in I$. 
                \item If $i < n$, then $nm - i > nm -n = n(m-1) = nm - i \geq m$. 
                And $ b^{m} \in I \implies  \binom{nm}{i}(a)^i(-b)^{nm-i} \in I$. 
            \end{itemize}
            Since every term in the sum lies in $I$, the whole sum lies in $I$. So $a-b \in \sqrt{I}$.
            
            \item Note that if $a^n \in I$ then $(ra)^n \in I$. Hence $ra \in \sqrt{I}$. 
        \end{enumerate}
        Thus $\sqrt{I}$ is an additive subgroup of $R$ and is closed under multiplication from elements of $R$. 
    \end{enumerate}
\end{solution}

\begin{problem}{2}
    Let $\K$ be a field. The rings $\Z$, $\K[x]$, and $\K[[t]]$ are all principal ideal domains (PIDs) meaning domains where every ideal is generated by a single element. If $I= \langle a \rangle$ and $J = \langle b \rangle$ give, with proof, a careful concrete description of generators for $IJ$, $I\cap J$, $I+J$, and $\sqrt{I}$ in each of these rings.
\end{problem}


\begin{solution}
    \bbni

    Let $I =  \langle a \rangle, \, J = \langle b \rangle$. Then, we have 
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item $\Z$ 
        \begin{enumerate}[label=(\roman*),ref=\theenumi\roman*]
            \item $IJ = \langle ab \rangle$ 
            \item $I \cap J = \langle \text{lcm} (a,b) \rangle$ 
            \item $I + J = \langle \gcd (a, b) \rangle$ 
            \item Let $a = p_1^{e_1}\cdots p_k^{e_k}$, then $\sqrt{I} = \langle p_1\cdots p_k \rangle$, where $p_i$ is a prime element. 
        \end{enumerate}
        \item $\K[x]$
        \begin{enumerate}[label=(\roman*),ref=\theenumi\roman*]
            \item $IJ = \langle ab \rangle$ 
            \item $I \cap J = \langle \text{lcm} (a,b) \rangle$ 
            \item $I + J = \langle \gcd (a, b)\rangle$ 
            \item Let $a = p_1^{e_1}\cdots p_k^{e_k}$, then $\sqrt{I} = \langle p_1\cdots p_k \rangle$, where $p_i$ is an irreducible element. 
        \end{enumerate}
        \item $\K[[t]]$
        Note that elements of $\K[[t]]$ are of two forms: $a \in \K$ or $a \cdot t^n, \, a\in \K$. So all proper ideals are of the form $\langle 0 \rangle$ or $\langle t^n \rangle$. Let $I = \langle t^n \rangle, \, J = \langle t^m \rangle$. 
        If $I$ is trivial, then $IJ = I \cap J = \sqrt{I} = \langle 0 \rangle$, and $I + J = J$. Assuming neither $I$ nor $J$ are trivial we have,
        \begin{enumerate}[label=(\roman*),ref=\theenumi\roman*]
            \item $IJ = \langle t^{n+m}\rangle$ 
            \item $I \cap J = \langle t^{\max(n,m)} \rangle$ 
            \item $I + J = \langle t^{\min(n,m)}\rangle$ 
            \item $\sqrt{I} = \langle t \rangle$. 
        \end{enumerate}
    \end{enumerate}
\end{solution}

\textbf{Universal Properties.} Throughout this course we will see numerous constructions and operations (e.g., quotients, tensor products, localizations, completions, etc.) each of these will have an explicit construction where we describe the objects precisely as sets. However, there is an alternative perspective, which is just as useful, describing these objects via their \emph{universal properties}, which roughly asks what makes each object special amongst all other such objects. The standard framework for a universal property is, ``Object X is unique up to isomorphism among all objects having property P.''

\hrulefill


\begin{problem}{3}
    \textbf{Universal Property of Quotients.} 
    
    Let $R$ be a ring and $I \subseteq R$ an ideal. Let $\pi\colon R \to R/I$ be the quotient map. Suppose $\varphi \colon R \to S$ is a ring homomorphism such that $I \subseteq \ker(\varphi)$. Define a function
    \[
    \widetilde{\varphi} \colon R/I \to S
    \quad \text{by} \quad
    \widetilde{\varphi}(r+I)=\varphi(r).
    \]
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
    	\item Prove that $\widetilde{\varphi}$ is well-defined.

    	\item Prove that $\widetilde{\varphi}$ is a ring homomorphism.

    	\item Prove that $\widetilde{\varphi} \circ \pi = \varphi$.

    	\item Prove that $\widetilde{\varphi}$ is the unique ring homomorphism $R/I \to S$ with the property that
    	\[
  	  \widetilde{\varphi}\circ \pi = \varphi.
    	\]
    	\item Explain in words what this says about the quotient ring $R/I$.
    \end{enumerate}
\end{problem}


\begin{solution}
    \bbni 
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Let $r + I = r' +I \in R/I$. Then, $r - r' \in I$. 
        
        \[\widetilde{\varphi}(r+I) = \varphi(r) = \varphi(r + (r - r')) = \varphi(r') = \widetilde{\varphi}(r'+I)\]
        
        Hence, $\widetilde{\varphi}$ is well-defined.

        \item Let $\overline{a}, \overline{b} \in R/I$, we check if the three conditions hold. 
        
        \textbf{Unital:} Since $\varphi$ is a ring homomorphism, we have \[\widetilde{\varphi}(1_{R/I}) = \varphi(1_R) = 1_S\]  

        \textbf{Addition:} Again, since $\varphi$ is a ring homomorphism, we have
        \[\widetilde{\varphi}(\overline{a}) + \widetilde{\varphi}(\overline{b}) = \varphi(a) + \varphi(b) = \varphi(a+b) = \widetilde{\varphi}(\overline{a+b})\]


        \textbf{Multiplication:} Again, $\varphi$ is a ring homomorphism, we have
        \[\widetilde{\varphi}(\overline{a}) \cdot \widetilde{\varphi}(\overline{b}) = \varphi(a) \cdot \varphi(b) = \varphi(a \cdot b) = \widetilde{\varphi}(\overline{a\cdot b})\]

        Hence, $\widetilde{\varphi}$ is a ring homomorphism. 

        \item Let $a \in R$. Then, 
        \[\widetilde{\varphi} \circ \pi (a) = \widetilde{\varphi} (\overline{a}) = \varphi(a)\]
        Since $a$ was an arbitrary element in $R$, this holds for all elements, and hence $\widetilde{\varphi} \circ \pi = \varphi$. 

        \item Assume there exists some other homomorphism $\sigma: R/I \to S$ such that $\sigma \circ \pi = \varphi$. Then for any $\overline{a} \in R/I$,  
        \[ \sigma (\overline a) = \sigma \circ \pi (a) = \varphi(a) = \widetilde{\varphi}(a)\]

        \item Every homomorphism $\varphi$ such that $I \subseteq \ker(\varphi)$ can be written as $\widetilde{\varphi} \circ \pi$ for some $\widetilde{\varphi}$.   
    \end{enumerate}    
\end{solution}

\begin{problem}{4}
In this problem, we wish to consider the evaluation map: 
    \[
    \ev_i \colon \Z[x] \to \C,
    \quad
    f(x)\mapsto f(i)
    \]
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Prove that $\ev_{i}$ is a ring homomorphism.

        \item Show that $\langle x^2+1 \rangle \subset \ker(\ev_i)$.

        \item Use Problem 3 to produce a ring homomorphism
        \[
        \widetilde{\ev_i}\colon \Z[x]/\langle x^2+ 1\rangle \to \C.
        \]

        \item What does $\widetilde{\ev_i}$ do to the class of $x$ in $\Z[x]/\langle x^2 +1\rangle$?
        
        \item Prove that $\Z[x]/\langle x^2+1\rangle$ is isomorphic to $\Z[i]$. Is $\langle x^2 +1\rangle$ prime and/or maximal?
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Let $f, g \in \Z[x]$, we check if the three conditions hold. 
        
        \textbf{Unital:} Let $f_1$ be the constant function $1$. Then, 

        \[\ev_i (f_1 (x)) = f_1(i) = 1\]

        \textbf{Addition:} 
        Since $\Z[x]$ is a ring under pointwise addition and multiplication, we can write, 
        \[\ev_i(f(x)) + \ev_i(g(x)) = f(i) + g(i) = (f+g)(i) = \ev_i((f+g)(x))\]

        \textbf{Multiplication:} 
        Since $\Z[x]$ is a ring under pointwise addition and multiplication, we can write, 
        \[\ev_i(f(x)) \cdot \ev_i(g(x)) = f(i) \cdot g(i) = (f\cdot g)(i) = \ev_i((f\cdot g)(x))\]

        Hence, $\ev_i$ is a ring homomorphism. 

        \item Note that $\ev_i(x^2 + 1) = i^2 + 1 = - 1 + 1 = 0$. For any $f(x) (x^2+1) \in \langle x^2+ 1\rangle$ we have $\ev_i(f(x) (x^2+1)) = f(i) \cdot 0 = 0 \in \ker(\ev_i)$, Thus the ideal generated by $\langle x^2 + 1 \rangle$ under $\ev_i$ maps to $0$ and hence is a subset of the kernel. 
        
        \item From  Q3 we know that if $\langle x^2 + 1 \rangle \subseteq \ker(\ev_i)$ there exists a unique homomorphism,
         \[\widetilde{\ev_i}: \Z[x]/\langle x^2 +1 \rangle \to \C, \qquad \widetilde {\ev_i}(\overline{f(x)}) = f(i)\] 
        
        \item  $\widetilde{\ev_i} (\overline{x}) = x(i) = i$.
        
        \item Define a map $\varphi: \Z[x] \to \Z[i], \, \varphi(f(x)) \to f(i)$. We will show that $\ker(\varphi) = \langle x^2 + 1 \rangle$. 
        
        $\langle x^2 + 1 \rangle \subseteq \ker(\varphi)$
        
        Note,
        \[\varphi(x^2 + 1) = i^2+ 1 = 0 \in \ker(\varphi)\]

        $\ker(\varphi) \subseteq \langle x^2 + 1 \rangle$

        Let $\varphi(f(x)) \in \ker(\varphi)$. If $f(i)=0$, then $x^2+1 \mid f(x)$ in $\mathbb{Z}[x]$ (since $x^2+1$ is the minimal polynomial of $i$ over $\Q$ and hence is primitive over $\Z$), and we have, 

        \[f(x) \in \langle x^2 + 1 \rangle  \implies \ker(\varphi) \subseteq \langle x^2 + 1 \rangle\]

        Then by the first isomorphism theorem, 
        \[\Z[x]/\langle x^2 + 1 \rangle \iso \im(\varphi) = \Z[i]\]
        
        Since $\Z[x]$ is an integral domain but not a field, $\langle x^2+ 1 \rangle$ is prime but not maximal.
    \end{enumerate}
\end{solution}

\begin{problem}{5}
    \textbf{Universal Property of Polynomial Rings.} 
    
    The goal of this exercise is to prove the following characterization of polynomial rings. Let $\varphi\colon R \to S$ be a ring homomorphism. For any element $s \in S$ there is a unique ring homomorphism $\psi\colon R[x] \to S$ such that $\psi(x) = s$ and the following diagram commutes:
    \[
    \begin{tikzcd}[row sep = 3.5em, column sep = 3.5em]
        R \arrow[r, "\varphi"] \arrow[d, hook] & S  \\
        R[x] \arrow[ur, dashed, swap,"\psi"]
    \end{tikzcd}.
    \]
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]

        \item Prove there is at most one ring homomorphism $\psi:R[x] \to S$ with the above properties. 
        \item Construct such a homomorphism explicitly by setting
        \[
        \psi\!\left(\sum_{i=0}^n a_i x^i\right)=\sum_{i=0}^n \varphi(a_i)s^i.
        \]
        \item Conclude that to give a ring homomorphism $R[x]\to S$ is equivalent to giving:
        \begin{itemize}
            \item a ring homomorphism $R\to S$, and
            \item an element of $S$.
        \end{itemize}
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni

    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Suppose there exists another homomorphism $\sigma: R[x] \to S$ such that $\sigma(x) = \psi(x) = s$. Also, note that sinct the diagram commutes $\psi|_R = \sigma|_R = \varphi$. 
        
        Then for any $\sum_{i=0}^n a_ix^i \in R[x]$ we have,

        \begin{align*}
            \sigma \left(\sum_{i=0}^n a_ix^i\right) &= \sigma(a_1)\cdot\sigma(x) + \cdots + \sigma(a_n)\cdot\sigma(x^n) \\
            &= \varphi(a_1)\cdot\psi(x) + \cdots + \varphi(a_n)\cdot\psi(x^n) \\
            &= \psi(a_1)\cdot\psi(x) + \cdots + \psi(a_n)\cdot\psi(x^n) \\
            &= \psi \left(\sum_{i=0}^n a_ix^i\right)
        \end{align*}
        
        Hence, $\sigma = \psi$. 

        \item Let $\psi: R[x] \to S$ such that 
        \[ \psi(x) = s, \quad \psi(a) = \varphi(a) \quad \forall a \in R\]

        \item Hence for any ring homomorphism $\sigma: R[x] \to S$ we have 
        \begin{center}
            \begin{tikzcd}[row sep = 3.5em, column sep = 3.5em]
             R \arrow[r, "\varphi"] \arrow[d, hook] & S  \\
             R[x] \arrow[ur, swap,"\psi"]
            \end{tikzcd}
        \end{center}
    \end{enumerate}
\end{solution}

\begin{problem}{6} 
    \textbf{Evaluation maps.} 
    
    Let $R$ be a ring and let $a\in R$. Define the evaluation map
    \[
    \ev_a \colon R[x]\to R,
    \quad
    f(x)\mapsto f(a).
    \]
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
    \item Prove that $\ev_a$ is a surjective ring homomorphism.

    \item Prove that $\ker(\ev_a) = \langle x-a\rangle$.

    \item Use Problem 3 to show that $\ev_a$ factors through a ring homomorphism
    \[
    R[x]/(x-a)\to R.
    \]
    
    \item Explain what this means if $R$ is a field or $R$ is an integral domain.
    
    \item Give an example of a ring $R$ and prime/maximal ideal $I \subset R[x]$ such that $I \neq \ker(\ev_a)$ for all $a \in R$. 
\end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item For all $r \in R$ consider the constant functions $r$ represented by $f_r \in R[x]$. Then, 
        \[\ev_a(f_r(x)) = f_r(a) = r\]

        \item $\ker(\ev_a) \subseteq \langle x-a \rangle$. 
        
        Note that for any $p(x) \in \ker(\ev_a)$ we have 
        \[p(x) = (x-a)q(x) + r(x)\]
        for some $q(x), r(x) \in R[x]$. Evaluating at $\ev_a$ we have, 
        \[\ev_a(p(x)) = (a-a)q(a) + r(a) = 0 \implies r(a) = 0 \] 
        That is 
        \[ p(x) = (x-a)q(x) \implies p(x) \in \langle x -a \rangle\]
        
        
        $\langle x -a \rangle \subseteq  \ker(\ev_a)$. 
        
        
        Let $p \in \langle x - a \rangle$. Since $\langle x -a \rangle$ irreducible we have $p = f(x) (x-a)$, for some $f(x) \in R[x]$,
        \begin{align*}
            \ev_a(p) &= \ev_a(f(x)(x-a)) = f(a)(a-a) = 0  \in \ker(\ev_a)
        \end{align*}
        Hence, $\ker(\ev_a) = \langle x-a \rangle$. 

        \item Since $\ker(\ev_a) \supseteq \langle x-a \rangle$ from problem 3 we know we have an induced ring homomorphism $\widetilde{\ev_a}$ such that
        \begin{center}
        \begin{tikzcd}[row sep=3em, column sep=3em]
            R [x] \arrow[]{r}{\pi} \arrow[swap]{d}{\ev_a} & R[x]/\langle x-a \rangle  \arrow[] {dl} {\widetilde{\ev_a}} \\
            R & {}
        \end{tikzcd}
      \end{center}

      \item If $R$ is a domain this means $\langle x -a \rangle$ is prime and also that $\widetilde{\ev_a} = \id$. 
      
      If $R$ is a field $\langle x- a\rangle$ is maximal and $R \iso R[x] /\langle x-a \rangle$. Also since all domains are fields we have $\widetilde{\ev_a} = \id$
      
      \item $R = \C, I = \langle 0 \rangle$. 
    \end{enumerate}
\end{solution}

 \textbf{Polynomial Division.} Recall the following classical theorem: 


\begin{theorem}\label{thm:division-alg}
Let $g \in \K[x]$ be a non-zero polynomial. Every polynomial $f\in \K[x]$ can be written uniquely as $f=qg+r$ where $q,r \in \K[x]$, and either $r=0$ or $\deg(r) < \deg(g)$. 
\end{theorem}

While this theorem might seem innocuous when first learned as a teenager, it turns out to be very powerful when studying ideals and polynomials in $\K[x]$. The proof of this theorem also provides an algorithm for computing $q$ and $r$, which when presented in pseudocode looks something like

\begin{algorithm}
\caption{Division Algorithm}
\begin{algorithmic}
\Require $g, f$
\Ensure $q, r$
\State $q \gets 0$; $r \gets f$
\While{$r \neq 0$ \textbf{and} $\LT(g)$ divides $\LT(r)$}
    \State $q \gets q + \LT(r)/\LT(g)$
    \State $r \gets r - (\LT(r)/\LT(g))g$
\EndWhile
\end{algorithmic}
\end{algorithm}

The goal of this worksheet is to begin exploring how we might extend this theorem to polynomial rings in more variables. Along the way we will see how these explorations will lead to both computational tools for studying the commutative algebra of polynomial rings over fields, but also gives us useful tools for proving theorems in this setting as well. 

\begin{problem}{1}
Use Theorem 1 to prove the following:
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item If $f\in \K[x]$ is a non-zero polynomial then $f$ has at most $\deg(f)$ roots in $\K$.
        \item If $g\in \K[x]$ is a non-zero polynomial then $f \in \langle g \rangle$ if and only if $r=0$. 
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni 

    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Assume for the sake of contradiction that $f$ has more than $n = \deg (f)$ roots, say $m$ roots: $a_1, \cdots a_m$. Then by Theorem 1, dividing by $(x-a_i)$ for all $ 1 \leq i \leq m$ we get $f = c(x- a_1) \cdots (x-a_m)$, for some constant $c$ and $r =0$. But then we have $m$ $x$ terms, i.e., $x^m$ is in $f$, but since $\deg(f) < m$ this is not possible. Hence a contradiction. 
        
        \item $(\implies)$ 
        
        Assume $f \in \langle g \rangle$. Then there exists some $q \in \K[x]$ such that $f = qg$. Then, 
        \[f = qg + r \implies f - qg = r = 0\]

        $(\impliedby)$ 

        Assume $r = 0$. Then by Theorem 1 we have, 
        \[f = qg + 0 = qg\]
        That is there exists some $q \in \K[x]$ such that $f = qg$ and by definition of an ideal, $f \in \langle g \rangle$. 
    \end{enumerate}
\end{solution}

\begin{definition}
    A monomial ordering on $\K[x_1,\ldots,x_n]$ is a relation $<$ on $\Z^{n}_{\geq0}$ satisfying the following:
    \begin{enumerate}
        \item[(i)] $<$ is a total ordering on $\Z^{n}_{\geq0}$,
        \item[(ii)] if $\alpha < \beta$ and $\gamma \in \Z^{n}_{\geq0}$ then $\alpha+\gamma < \beta + \gamma$, and
        \item[(iii)] $<$ is a well-ordering on $\Z^{n}_{\geq0}$.
    \end{enumerate}
\end{definition}

Recall that a \emph{total ordering} means that of any two elements $\alpha,\beta \in \Z^{n}_{\geq0}$ exactly one of the following statements is true: $\alpha < \beta$, $\alpha = \beta$, or $\beta < \alpha$. That $<$ is a well-ordering on $\Z^{n}_{\geq0}$  means that every non-empty subset of $\Z^{n}_{\geq0}$ has a smallest element with respect to $<$.

Once a monomial ordering has been fixed, every non-zero polynomial acquires a distinguished piece of data: its leading monomial, leading coefficient, and leading term. These are the ingredients that drive the multivariable division process just as degree and leading coefficient drive ordinary polynomial division.

\end{problem}

\begin{solution}
    \bbni

    There exists a bijection 
    \[\Z^n_{\geq 0} \ni \{\alpha = (a_1, \cdots a_n)\} \leftrightarrow \{x^\alpha = x_1^{a_1} \cdots x_n^{a_n}\} \in K[x_1, \cdots, x_n]\]
    that satisfies all the conditions because addition of vectors corresponds to addition of exponents during multiplication. 
\end{solution}

\begin{problem}{4}
    Prove or disprove that the following are monomial orderings:
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Lexicographic (Lex) Order: Given $\alpha = (\alpha_1, \dots, \alpha_n)$ and $\beta = (\beta_1, \dots, \beta_n)$ in $\mathbb{Z}_{\geq 0}^n$, we say $\beta <_{\text{lex}} \alpha$ if and only if the left-most non-zero entry of $\alpha - \beta$ is positive.
        
        \item Graded Lex Order: $\beta <_{\text{grlex}} \alpha$ if and only if
        \[
        |\beta| = \sum_{i=1}^n \beta_i < |\alpha| = \sum_{i=1}^n \alpha_i
        \]
        or $|\beta| = |\alpha|$ and $\beta <_{\text{lex}} \alpha$.

        \item Reverse Lex (RevLex) Order: $\beta <_{\text{revlex}} \alpha$ if and only if the right-most non-zero entry of $\alpha - \beta$ is negative.

        \item Graded RevLex Order: $\beta <_{\text{grevlex}} \alpha$ if and only if
        \[
        |\beta| = \sum_{i=1}^n \beta_i < |\alpha| = \sum_{i=1}^n \alpha_i
        \]
        or $|\beta| = |\alpha|$ and $\beta <_{\text{revlex}} \alpha$.

        \item Weight Order: Fix $w = (w_1, \dots, w_n) \in \mathbb{Z}^n$. Define $\beta <_w \alpha$ if and only if
        \[
        \beta \cdot w = \sum_{i=1}^n \beta_i w_i < \alpha \cdot w = \sum_{i=1}^n \alpha_i w_i.
        \]
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item It is a monomial ordering. 
        Let $\alpha = (\alpha_1, \cdots, \alpha_n), \, \beta = (\beta_1, \cdots, \beta_n) \in \Z_{\geq 0}^n$. 
        \begin{enumerate}
            \item [(i)] Total Ordering: 
            
            If $\alpha = \beta$ we're done. Otherwise, $\alpha - \beta$ has a leftmost non-zero entry. If this entry is positive, $\beta <_{\text{lex}} \alpha$, else $\alpha <_{\text{lex}} \beta$. 

            \item [(ii)] Let $\alpha <_\text{lex} \beta$, that is the leftmost non-zero entry of $\alpha - \beta$ is negative. Then for all $\gamma \in \Z_{\geq 0}^n$ we have, 
            \[(\alpha + \gamma) - (\beta + \gamma) = \alpha + \beta\]
            That is the leftmost non-zero entry remains unchanged and hence $(\alpha + \gamma) <_\text{lex} (\beta + \gamma)$. 

            \item [(iii)] Let $S \subseteq \Z_{\geq 0}^n$ be non-empty. Since $\Z_{\geq 0}$ is well-ordered we let $m_1 = \min \{\alpha_1 \mid \alpha \in S\}$. Then, among the elements with first coordinate $m_1$ we let $m_2 = \min \{\alpha_2 \mid \alpha \in S\}$ and so on for all $n$ coordinates. This gives us, 
            \[ m = (m_1, \cdots, m_n) \in S\]
            which is the lex-minimal element. 
        \end{enumerate}

        \item It is a monomial ordering. 
        Let $\alpha = (\alpha_1, \cdots, \alpha_n), \, \beta = (\beta_1, \cdots, \beta_n) \in \Z_{\geq 0}^n$. 
        \begin{enumerate}
            \item [(i)] Total Ordering: 
            
            If $\alpha = \beta$ we're done. 
            Else if $|\alpha| = |\beta|$, by part (a) we have an ordering. 
            Otherwise, $|\alpha| \neq |\beta|$ has a leftmost non-zero entry. If $|\alpha| > |\beta|$, $\beta <_{\text{grlex}} \alpha$, else $\alpha <_{\text{grlex}} \beta$. 

            \item [(ii)] Let $\alpha <_\text{grlex} \beta$, and $\gamma \in \Z_{\geq 0}^n$. 
            
            \begin{itemize}
                \item If $|\alpha |  < |\beta|$, we have
                \begin{align*}
                  |\alpha| + |\gamma|  < |\beta| + |\gamma| \implies   |\alpha +\gamma |  < |\beta +\gamma| 
                \end{align*}
                
                \item If $|\alpha | = |\beta|$ and $\alpha <_\text{lex} \beta$ we have an ordering by part (a). 
            \end{itemize}
            
            \item [(iii)] Let $S \subseteq \Z_{\geq 0}^n$ be non-empty. Since all entries are non-negative there exists $m = \min\{|\alpha| \mid \alpha \in S\}$. Then, we take the lex-minimal element in the subset $H \subseteq S$, $H = \{|\alpha| = m \mid \alpha \in S\}$. This is the grlex-minimal element.  
        \end{enumerate}

        \item It is a monomial ordering. 
        Let $\alpha = (\alpha_1, \cdots, \alpha_n), \, \beta = (\beta_1, \cdots, \beta_n) \in \Z_{\geq 0}^n$. 
        \begin{enumerate}
            \item [(i)] Total Ordering: 
            
            If $\alpha = \beta$ we're done. Otherwise, $\alpha - \beta$ has a rightmost non-zero entry. If this entry is negative, $\beta <_{\text{revlex}} \alpha$, else $\alpha <_{\text{revlex}} \beta$. 

            \item [(ii)] Let $\alpha <_\text{revlex} \beta$, that is the leftmost non-zero entry of $\alpha - \beta$ is positive. Then for all $\gamma \in \Z_{\geq 0}^n$ we have, 
            \[(\alpha + \gamma) - (\beta + \gamma) = \alpha + \beta\]
            That is the rightmost non-zero entry remains unchanged and hence $(\alpha + \gamma) <_\text{revlex} (\beta + \gamma)$. 

            \item [(iii)] Let $S \subseteq \Z_{\geq 0}^n$ be non-empty. Since $\Z_{\geq 0}$ is well-ordered we let $m_n = \min \{\alpha_n \mid \alpha \in S\}$. Then, among the elements with first coordinate $m_n$ we let $m_{n-1} = \min \{\alpha_{n-1} \mid \alpha \in S\}$ and so on for all $n$ coordinates. This gives us, 
            \[ m = (m_1, \cdots, m_n) \in S\]
            which is the revlex-minimal element. 
        \end{enumerate}

        \item It is a monomial ordering by (b) and (c). 
        
        \item Not a monomial ordering. For instance, if $w = (1,1), \, \alpha = (0,2), \, \beta = (2,0)$ we have $\beta \cdot w = \alpha \cdot w$ but $\alpha \neq \beta$. 
    \end{enumerate}
\end{solution}

\begin{problem}{5} \textbf{Leading Terms.} 
    
    Fix a monomial ordering $<$ on $\mathbb{Z}_{\geq 0}^n$. Let $f = \sum_\alpha c_\alpha x^\alpha \in K[x_1, \dots, x_n]$ be a non-zero polynomial. Define the leading monomial $\mathrm{LM}(f)$, leading coefficient $\mathrm{LC}(f)$, and leading term $\mathrm{LT}(f)$.

    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Compute $\mathrm{LM}(f)$, $\mathrm{LC}(f)$, and $\mathrm{LT}(f)$ for
        \[
        f = x^3 + 3x^2 y + 5xy^3 - 7z^2 \in K[x,y,z]
        \]
        with respect to Lex, Graded Lex, and Graded RevLex (assume $x > y > z$).
        
        \item Prove that if $f, g \in K[x_1, \dots, x_n]$ are non-zero, then
        \[
        \mathrm{LM}(fg) = \mathrm{LM}(f)\mathrm{LM}(g), \quad \mathrm{LT}(fg) = \mathrm{LT}(f)\mathrm{LT}(g).
        \]
        
        \item Explain which property of a monomial ordering is used in part (b).
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item 
        \begin{itemize}
            \item Lex: \[\LM(f) = x^3, \quad \LC(f) = 1, \quad \LT(f) = x^3\]
            \item Graded Lex: \[\LM(f) = xy^3, \quad \LC(f) = 5, \quad \LT(f) = 5xy^3\]
            \item Graded RevLex: \[\LM(f) = xy^3, \quad \LC(f) = 5, \quad \LT(f) = 5xy^3\]
        \end{itemize}

        \item Let $f = \LT(f) + \sum_{\alpha < \LM(f)} c_\alpha x^\alpha, \, g = \LT(g) + \sum_{\beta < \LM(g)} d_\beta x^\beta \in K[x_1, \cdots, x_n]$ be non-zero. Then the product $fg$ is, 
        \[fg = \sum_{\alpha, \beta} c_\alpha d_\beta x^{\alpha+\beta}\]
        Note that $\LT(f)\LT(g) = c_0d_0 x^{\alpha_0 + \beta_0}$ appears in $fg$. We know for all $\alpha$s and $\beta$s, $\alpha < \alpha_0$ and $\beta < \beta_0$ so
        \[\alpha + \beta < \alpha_0 + \beta_0\]
        Hence $x^{\alpha + \beta} < x^{\alpha_0 + \beta_0}$, and $x^{\alpha+\beta} = \LM(f) \LM(g) =  \LM(fg)$. Moreover $\LT(fg) = \LC(fg) \LM(fg)$, where $\LC(fg) = \LC(f) \LC(g)$ by definition, and we showed $\LM(fg) =  \LM(f) \LM(g)$. Hence $\LT(fg) = \LT(f) \LT(g)$. 

        \item (ii) property is used. 
    \end{enumerate}
\end{solution}

\begin{problem}{6}
    Let $G = (g_1, \dots, g_s)$ be an ordered list of non-zero polynomials in $K[x_1, \dots, x_n]$.

    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Write pseudocode for a division algorithm which takes as input $f$ and $G$ and returns polynomials $q_1, \dots, q_s, r$ such that
        \[
        f = q_1 g_1 + \cdots + q_s g_s + r,
        \]
        where no monomial in $r$ is divisible by any $\mathrm{LM}(g_1), \dots, \mathrm{LM}(g_s)$.
        \item In the algorithm above, what should happen if the leading monomial of the current polynomial is not divisible by any $\mathrm{LM}(g_i)$?
        \item Prove that your algorithm terminates. What plays the role of “degree” in the proof?
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item 
        % \begin{algorithm}[H]
        % \caption{Division by an ordered list $G=(g_1,\ldots,g_s)$}
        \begin{algorithmic}
        \Require $f, g_1,\ldots,g_s$
        \Ensure $q_1,\ldots,q_s,r$
        \State $q_1,\ldots,q_s \gets 0$; $r\gets 0$; $p\gets f$
        \While{$p\neq 0$}
            \State $\mathrm{divides}\gets 0$
            \For{$i=1,\ldots,s$}
                \If{$\LT(g_i)$ divides $\LT(p)$}
                    \State $q_i\gets q_i+\LT(p)/\LT(g_i)$
                    \State $p\gets p-(\LT(p)/\LT(g_i))g_i$
                    \State $\mathrm{divides}\gets 1$
                    \State \textbf{break}
                \EndIf
            \EndFor
            \If{$\mathrm{divides}=0$}
                \State $r\gets r+\LT(p)$
                \State $p\gets p-\LT(p)$
            \EndIf
        \EndWhile
        \end{algorithmic}
        % \end{algorithm}

        \item We add it to the remainder term $r$, and continue. 
        
        \item $\LM(f)$ plays the role of degree. At every step, we update either $p \rightarrow p - \LT(p)$ (if no $g_i$ divides $\LM(p)$) or $p \rightarrow \LT(p)/\LT(g_i) \cdot g_i$ (for some $g_i$ dividing $\LM(p)$). So the leading term strictly decreases. Hence, the program must terminate.  
    \end{enumerate}
\end{solution}

\begin{problem}{1}
Recall that the spectrum and maximal spectrum of a ring $R$ are the sets:
\[
\Spec(R) \coloneqq\left\{ P \subset R \;\; \big|\;\; \text{$P$ is a prime ideal}\right\}\]
and 
\[\mSpec(R) \coloneqq\left\{ P \subset R \;\; \big|\;\; \text{$P$ is a maximal ideal}\right\} 
\]
Given any subset $\cS \subset R$ define 
\[
\V(\cS) \coloneqq \left\{P \in \Spec(R) \;\; \big| \;\; \cS \subset P\right \}.
\]
\begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
	\item Describe $\Spec(\K)$ and $\mSpec(\K)$ when $\K$ is a field. 
	\item Describe both $\Spec(\K[x]/\langle x^{3}\rangle)$ and $\mSpec(\K[x]/\langle x^{3}\rangle)$ when $\K$ is a field. 
	\item For an arbitrary ring $R$, compute $\V(\{0\})$ and $\V(\{1\})$. 
	\item Explain the natural poset structure on $\Spec(R)$. Describe the poset for $\Spec(\Z)$. 
\end{enumerate}
\end{problem}

\begin{solution}
    \bbni 
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Let $\K$ be a field. Then $\K$ only has $1$ proper ideal: the ideal generated by $0$. Note that this ideal is prime because $\K$ is an integral domain. Moreover it is also maximal because $\K/\langle 0 \rangle = \K$ is a field. So, 
        \[\Spec(\K) = \mSpec(\K) = \{\langle 0 \rangle\}\]

        \item Note that the ideal generated by $0$ is not prime in this ring because $x \cdot x^2 \in \langle 0 \rangle $ even though neither $x$ nor $x^2$ are in $\langle x\rangle$. Then we get, \[\Spec(\K[x]/\langle x^3 \rangle) =  \mSpec(\K[x]/\langle x^3 \rangle) = \{\langle x \rangle\}\] 
        
        \item Let $R$ be an arbitrary ring, then, 
        \[\V(\{0\}) = \Spec(R)\]
        since every ideal contains $0$ and, 
        \[\V(\{1\}) = \emptyset\]
        since $1$ is not contained in any proper ideal of $R$.
        
        \item 
        The natural poset structure on $\Spec(R)$ is given by inclusion of ideals, that is, for $P_1, P_2 \in \Spec(R)$, we write $P_1 \le P_2$ iff $P_1 \subseteq P_2$.

        The poset for $\Spec(\Z) = $ contains prime ideals in $\Z$. The zero ideal is contained in every other ideal and every other prime ideal is maximal. 
        
    \end{enumerate}
\end{solution}

A \emph{topological space} is a set $X$ together with a collection of subsets $\tau$, which satisfies the following conditions: i) $\varnothing \in \tau$ and $X \in \tau$, ii) $\tau$ is closed under arbitrary unions, and iii) $\tau$ is closed under finite intersections. We call $\tau$ a \emph{topology} on $X$, the elements of $\tau$ the open subsets of $X$, and a subset $Z \subset X$ is closed if and only if $X\setminus Z$ is open (i.e., $X \setminus Z \in \tau$). If $S \subset X$ is a subset, the \emph{subspace topology} on $S$ is the topology where the open sets of $S$ are all sets of the form $S \cap U \subset S$ where $U \subset X$ is open in $X$. A map $f: X \to Y$ of topological spaces is \emph{continuous} if and only if $f^{-1}(U) \subset X$ is open for all open subsets $U \subset Y$, i.e., the preimage of an open set is open.

\begin{problem}{2}
\textbf{The Zariski Topology.} We will now place a topology on the set $\Spec(R)$,  called the Zariski topology after Oscar Zariski. 
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Prove that if  $\cS, \cT \subset R$ are arbitrary sets then $\V(\cS) \cap \V(\cT) = \V(\cS \cup \cT)$. Explain why your argument extends to an \emph{arbitrary} intersection. 
        \item Prove that if  $\cS, \cT \subset R$ are arbitrary sets then $\V(\cS) \cup \V(\cT) = \V(\cS \cdot \cT)$ where 
    \[
    \cS \cdot \cT \coloneqq \{ st \;\; |\;\; s \in \cS,\; t \in \cT\}.
    \]
        \item Prove that the sets of the form $\V(\cS)$ are the closed sets of a topology on $\Spec(R)$. We call this topology the Zariski topology on $\Spec(R)$. (Hint: Rephrase the definition of a topology in terms of closed sets instead of open sets.) 
        \item Take a break and Google Oscar Zariski and read about his life. 
    \end{enumerate}
\end{problem}
    
\begin{solution}
    \bbni 
    \begin{enumerate} [label=(\alph*),ref=\theenumi\alph*]
        \item We prove this for arbitrary intersections. Let $\cS_i$, $i \in \Lambda$ where $\Lambda$ is some indexing set, be arbitrary subsets of $R$. Then by definition,
        \[\underset{i\in \Lambda}{\bigcap}\V(\cS_i) = \underset{i \in \Lambda}{\bigcap}\{[P] \in \Spec(R), \cS_i \subseteq P\} = \left \{[P] \in \Spec(R), \underset{i\in \Lambda}{\bigcup}\cS_i \subseteq P \right \} \]
        And 
        \[\V \left (\underset{i \in \Lambda}{\bigcup} \cS_i  \right ) = \left \{[P] \in \Spec(R), \underset{i \in \Lambda}{\bigcup} \cS_i \subseteq P \right \} \]
        That is, $ \underset{i\in \Lambda}{\bigcap}\V(\cS_i) = \V \left (\underset{i \in \Lambda}{\bigcup} \cS_i  \right )$. Hence proved.

        \item Let $\cS, \cT$ be arbitrary subsets of $R$. Then by definition, 
        \[\V(\cS \cdot \cT) = \{[P] \in \Spec(R), \, \cS\cdot \cT \subseteq P\}\]
        And 
        \[\V(\cS) \cup \V(\cT) = \{[P] \in \Spec(R), \cS \subseteq P\} \cup \{[P] \in \Spec(R), \cT \subseteq P\}\] 
        Note that $\cS \cdot \cT \subset \cS, \cT$. Then the containment $\V(\cS \cdot \cT) \subseteq \V(\cS) \cup \V(\cT)$ is obvious. 
        
        To prove that $\V(\cS \cdot \cT) \supseteq \V(\cS) \cup \V(\cT)$, it suffices to show that if $\cS \cdot \cT \subseteq P$ then either $\cS \subseteq P$ or $\cT \subseteq P$. 
        
        Assume $\cS \cdot \cT \subseteq P$. Let $s \in \cS$ and $t \in \cT$. Then, since $P$ is a prime ideal and $st \in P$ then either $s \in P$ or $t \in P$. 
        If both $s, t \in P$ we are done. Else wlog, assume that $s \notin P$.
        But note that $\forall t' \in \cT$, $s\cdot t' \in P$. Hence, $t' \in P, \, \forall t' \in \cT$, i.e. $\cT \subseteq P$. 

        Hence, $\V(\cS) \cup \V(\cT) = \V(\cS \cdot \cT)$. 

        \item To prove that the sets of the form $\V(\cS)$ are the closed sets of the Zariski topology on $\Spec(R)$ we must show that (i) $\emptyset, \, \Spec(R)$ are closed sets, (ii) the finite union of closed sets is closed, and (iii) arbitrary intersection of closed sets is closed. 
        
        \begin{enumerate}
            \item [(i)] Note that $\emptyset = \{\V(\{1\})\}$ since $1$ is not contained in any proper ideal of $R$, and $\Spec(R) = \{\V(\{0\})\}$ since $0$ is contained in every ideal and thus in every prime ideal. Both $\V(\{1\})$ and $\V(\{0\})$ are closed sets, hence $\emptyset$ and $\Spec(R)$ are closed. 
            \item [(ii)] From part (b) we know that the finite union of two closed sets is closed, and by induction we can extend this to a finite number of closed sets. Hence, the finite union of closed sets is closed. 
            \item [(iii)] From part (a) we know that any arbitrary intersection of closed sets is closed. 
        \end{enumerate}
        Hence, sets of the form $\V(\cS)$ are the closed sets of the Zariski topology on $\Spec(R)$.
    \end{enumerate}    
\end{solution}

A function $f: X \to Y$ of topological spaces is \emph{continuous} if and only if $f^{-1}(U) \subset X$ is open for all open subsets $U \subset Y$, i.e., the preimage of an open is open. A \emph{homeomorphism} $f: X \to Y$ is a continuous map of topological spaces with a continuous inverse $f^{-1}:Y \to X$. A useful fact is that a continuous map $f:X \to Y$ is a homeomorphism if and only if it is bijective and $f(U) \subset Y$ is open for every open subsets $U \subset X$.

A \emph{category} is a collection of objects together with, for each pair of objects, a set of morphisms between them, equipped with an associative composition law and an identity morphism for each object. \begin{theorem}[Hilbert's Nullstellensatz I] If $\K$ is algebraically closed, then every maximal ideal of $\K[x_1,\ldots,x_n]$ is of the form $\langle x_1-a_1,\ldots,x_n-a_n\rangle$ for some point $(a_1,\ldots,a_n) \in \K^n$.
\end{theorem}  
 
 Note this is just the first of a series of theorems that will go by Hilbert's Nullstellensatz. Recall that a $\K$-algebra is a ring $R$ together with a specified ring homomorphism $\K \to R$. A morphism of $\K$-algebras $\varphi:R\to S$ is a ring homomorphism that preserves scalars, i.e.\ $\varphi(c\cdot 1_R)=c\cdot 1_S$ for every $c\in \K$. In particular, $\K[x_1,\ldots,x_n]$ is a $\K$-algebra, every evaluation map $\ev_p:\K[x_1,\ldots,x_n]\to \K$ is a $\K$-algebra homomorphism, and if $\fm$ is a maximal ideal then $R/\fm$ is naturally a field equipped with a $\K$-algebra structure. When we say below that a residue field is ``isomorphic to $\K$,'' we mean \emph{as a $\K$-algebra}.

\begin{solution}
    \bbni
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item To show that $\ev_p$ is a ring homomorphism, we need to show that for all $f_1, f_2 \in \K[x_1, \cdots, x_n]$ the three conditions hold, 

        \textbf{Additivity:} 
        \[
            \ev_p(f_1 + f_2) = (f_1+f_2)(p) 
            = f_1(p) + f_2(p) 
            = \ev_p(f_1) + \ev_p(f_2)
        \]

\begin{problem}{3}
    \textbf{Defining Ideal of an Algebraic Set.} Let $V \subset \K^n$ be an algebraic set. The \emph{defining ideal} of $V$ is the ideal in $\K[x_1,\ldots,x_n]$ given by 
 \[
 \I(V) \coloneqq \left\{ f \in \K[x_1,\ldots,x_n] \;\; \big| \;\; f(p) = 0 \text{ for all $p \in V$} \right\}.
 \]
 \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
 	\item Prove that $\I(V)$ is in fact an ideal. 
	\item Prove that $\I(V)$ is radical. 
	\item If $V = \V(\cS)$ for some subset $\cS \subset \K[x_1,\ldots,x_n]$, show that $\langle \cS \rangle \subset \sqrt{\langle \cS \rangle} \subset \I(V)$. 
	\item\label{ex-pt:big-obstruction} Let $\K=\R$. Give an example to show that the inclusion $\sqrt{\langle \cS \rangle } \subset \I(V)$ can be proper. 
	\item If $V = \V(\cS)$ for some subset $\cS \subset \K[x_1,\ldots,x_n]$, show that $\V(\cS) = \V(\I(V))$. 
 \end{enumerate}
\end{problem}


\begin{solution}
    \bbni 
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Let $g \in \K[x_1, \cdots, x_n]$ and $f_1, f_2 \in \I(V)$. That is, $f_1(p) = f_2(p) = 0$. 
        
        \textbf{Subgroup Under Addition:} Note that, \[(f_1 + f_2)(p) = f_1(p) + f_2(p) = 0\] The  zero polynomial $0(p) = 0$ acts as the identity, and \[f_1(p) + (-f_1)(p) = 0(p) = 0\] gives us the additive inverse.  

        \textbf{Closed Under Multiplication:} Notice, 
        \[(g \cdot f)(p) = g(p) \cdot f_1(p) = g(p) \cdot 0 = 0\]
        Thus $g \cdot f \in \I(V)$ and $\I(V)$ is an ideal. 

        \item We want to show that $\I(V) = \sqrt{\I(V)}$. 

        Note that the containment $\I(V) \subseteq \sqrt{\I(V)}$ is obvious. To show the opposite containment take $f \in \sqrt{\I(V)}$. Then for some $n$ we have, 
        \[f^n(p) = (f(p))^n = 0\]
        But since $\K$ is a domain, this implies that $f(p) = 0$. Hence $f \in \I(V)$, $\I(V) \supseteq \sqrt{\I(V)}$, and $\I(V) = \sqrt{\I(V)}$.

        \item Let $V = \V(\cS)$. We know that $\langle \cS \rangle \subseteq \sqrt{\langle \cS \rangle}$. So showing that $\sqrt{\langle \cS \rangle} \subset \I(V)$ suffices. Note that 
        $\I(V) = \I(\V(\cS))$ which is the set of all functions that evaluate to $0$ at $p$, for all $p \in \V(\cS)$. And $\V(\cS)$ is the set of all points $p$ such that all functions $f \in \cS$ evaluate to $0$ at $p$. Let $f \in \sqrt{\langle \cS \rangle}$. Then by definition for some $n$, $f^n in \S$. We can then write, 
        \[f^n = \sum_ih_i(p)g_i(p), \quad h_i \in \K[x_1, \cdots, x_n], g_i \in \cS\]
        But then, 
        \[f^n(p) = \sum_ih_i(p)g_i(p) = \sum_ih_i(p)\cdot 0 = 0\]
        Moreover since $\K$ is a domain, $f^n(p) = 0 \implies f(p) = 0 \implies f$ is a function that evaluates to $0$ at $p$, hence $f \in \I(V)$. 

        \item Consider the set $\cS = \{x^2+ 1\}$. We have, 
        \[\sqrt{\langle \cS \rangle} = \sqrt{\langle x^2 + 1 \rangle}\]
        And, because the polynomial has no solutions over $\R$, we have $V = \V(\cS) = \emptyset$. But $\I(\V(\cS)) = \R[x]$. Also note that since $\langle \cS \rangle$ is a prime ideal ($x^2 +1$ is irreducible), we have $ {\langle x^2 + 1 \rangle} = \sqrt{\langle x^2 + 1 \rangle} \subset \R[x]$. Hence, the containment $\sqrt{\langle \cS \rangle} \subset \I(V)$ can be proper. 

        \item Let $V =\V(\cS)$. 
        
        $\V(\cS) \subseteq \V(\I(\V(\cS)))$

        This containment is obvious by construction. $\I(\V(\cS))$ is the set of all functions that vanish at all points in $\V(\cS)$. And $\V(\I(\V(\cS)))$ is the set of all points that all functions in $\I(\V(\cS))$ vanish at. So, $\V(\cS) \subseteq \V(\I(V))$. 

        \phantom{A}

        $\V(\cS) \supseteq \V(\I(\V(\cS)))$

        Note that by definition $\cS \subseteq \I(\V(\cS))$. Then by 2(a) $\V(\I(\V(\cS))) \subseteq \V(\cS)$.
        
        Hence, $\V(\cS) = \V(\I(\V(\cS)))$.
    \end{enumerate}    
\end{solution}

So far you have built two order-reversing operations:

\begin{problem}{5}
    \textbf{Coordinate Rings.}

    \[
    \begin{tikzcd}[column sep = 3.5em, row sep = .5em]
    \K[x_1,\ldots,x_n] \arrow[r, "\rho"] & \cR\\
    f \arrow[r, mapsto] & f|_{V}

    \]
    Here we are viewing $f \in \K[x_1,\ldots,x_n]$ as defining a function $\K^n \to \K$ given by $p \mapsto f(p)$. We call the image of $\rho$ the \emph{coordinate ring} of $V$ and denote it $\K[V]$.

\begin{solution}
    \bbni 
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Let $f_1, f_2 \in \K[x_1, \cdots, x_n]$. Note that addition and multiplication is defined pointwise in $R$. To see if $\rho$ is a ring homomorphism, we check if the three properties hold: 

        \textbf{Addition: }
        \[\rho(f_1) + \rho(f_2)(p) = f_1|_V(p) + f_2|_V(p) = (f_1+f_2)|_V(p) = \rho(f_1+f_2)(p)\]


        \textbf{Multiplication: }
        \[\rho(f_1) \cdot \rho(f_2)(p) = f_1|_V(p) \cdot f_2|_V(p) = (f_1\cdot f_2)|_V(p) = \rho(f_1\cdot f_2)(p)\]

        \textbf{Unital:}
        \[\rho(1)(p) = 1|_V(p) = 1\]

	    \item 
        \begin{enumerate}[label=(\roman*), ref=\theenumi\roman*]
            \item Let $f: (x,y) \mapsto e^{x^2 + y^2}$. Note that on $V$, $x^2 + y^2 = 1$ then, $f|_V = e^{x^2+y^2} = e^1$. Which behaves like the constant polynomial function $g(x,y) = e$ and hence lies in the ring of coordinates of $V$.

            and

            Then, $V = \{(0, 0)\}$. And $I(V) = I (\{(0,0)\}) = \langle x, y \rangle$. 


            \[\K[V] \iso \K[x,y]/\I(\V(y^2, xy, x^{17})) = \K[x,y] / \langle x, y \rangle = \K\]

            \item Let $V = \V(x^6 - 3x^5y + 3x^4y^2 + x^3y^3) = \V(x^3(x-y)^3)$. 
            
            Then if $(a, b) \in V$, either $a^3 = 0 \implies a = 0$ or $(x-y)^3 = 0 \implies x = y$. 
            
            Then $V = \{(0, a), (a, a) \, | \, a \in \K\} = \V(x) \cup \V(x-y)$. 
            \[\I(V) = \I(\V(x) \cup \V(x-y)) = \I(\V(x)) \cap \I(\V(x-y)) = \langle x \rangle \cap \langle x-y \rangle = \langle x(x-y) \rangle \]
            And 
            \[\K[V] \iso \K[x,y]/\langle x(x-y) \rangle \]
            
            \item Let $A = (x_{ij}) \in SL_3(\F_5)$, and $V = \V(\det(A) - 1, \, tr(A))$. Note that the trace formula is $x_{11} + x_{22} + x_{33}$, and we use fermat's little theorem to enforce staying in $\F_5$. Then the coordinate ring is, 
            \[\K[V] = \F_5[x_{11} , \cdots, x_{33}] / \langle \det(A) - 1, x_{11} + x_{22} + x_{33}, x_{ij}^5 - x_{ij} \rangle\]
        \end{enumerate}
    \end{enumerate}
\end{solution}

\begin{problem}{7}

        \item For each point $p \in V$, let
        \[

        \]
        be the corresponding maximal ideal. Show that the map
        \[
        \begin{tikzcd}[row sep = .5em, column sep = 3.5em]
        V \arrow[r] & \Spec(\K[V]) \\

        \end{tikzcd}
        \]
        is a homeomorphism onto its image, which consists of those maximal ideals whose residue field is isomorphic to $\K$ as a $\K$-algebra.
        \item If $\K$ is algebraically closed how can the preceding part be rephrased in terms of $\mSpec(\K[V])$?
        \item Is the Zariski topology on $V$ Hausdorff in general? If not, when is it? 
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item $V = \V(I)$ for some $I \subset \K[x_1, \cdots, x_n]$. The closed sets of $V$ are of the form $\V(J)$, where $I \subset J \subset \K[x_1, \cdots, x_n]$. We need to check the following:

        \item $\emptyset$ and $V$ are closed. 
        
        Note $\V(I) = V$, and $\V(\K[x_1, \cdots, x_n]) = \emptyset$ is closed. 

        \item Finite union of closed sets is closed.

        \item Arbitrary intersection of closed sets is closed. 
    \end{enumerate}
    \end{enumerate}

    The previous exercises suggest why valuation rings are called \emph{valuation} rings: they arise from functions that measure divisibility, generalizing the $p$-adic absolute value. Such functions are called valuations. An \emph{ordered abelian group} is an abelian group $(\Gamma,+)$ together with a total order $<$ on $\Gamma$ that is compatible with addition, in the sense that if $\alpha<\beta$ then  $\alpha+\gamma<\beta+\gamma$ for all $\alpha,\beta,\gamma \in \Gamma$.   We will frequently extend the total ordering on an ordered abelian group $\Gamma$ to $\Gamma \cup \{\infty\}$ by making the convention that $\gamma < \infty$ for all $\gamma \in \Gamma$. Similarly, we will extend addition to $\Gamma\cup\{\infty\}$ by declaring $\infty + \gamma = \gamma + \infty = \infty$ for all $\gamma \in \Gamma$.

    \begin{definition}
    Let $(\Gamma,+,<)$ be an ordered abelian group. A  \emph{$\Gamma$-valued valuation}, or just a \emph{valuation}, on $\K$ is a function: $\nu:\K \to \Gamma\cup\{\infty\}$, which satisfies the following axioms for all $x,y\in \K$:
    \begin{enumerate}
    \item $\nu(xy) = \nu(x) + \nu(y)$
    \item $\nu(x + y) \geq \min\left\{ \nu(x), \nu(y)\right\}$, and
    \item $\nu(x) = \infty$ if and only if $x=0$.
    \end{enumerate}
    \end{definition}

    Note that this means $\nu$ defines a group homomorphism $\nu:\K^{\times} \to \Gamma$. Indeed, it is standard to define a valuation as a group homomorphism $\nu: \K^{\times} \to \Gamma$ satisfying axiom (ii), and then extend it to all of $\K$ by setting $\nu(0)=\infty$. This convention is implicit throughout the remainder of the worksheet. 

    The \emph{value group} of a valuation $\nu:\K^{\times} \to \Gamma$ is the image of $\nu$. A valuation is a \emph{discrete valuation} if its value group is isomorphic to $\Z$. If $\FF \subset \K$ is a subfield, we say that $\nu$ is a $\FF$-valuation if $\nu(\lambda) = 0 $ for all $\lambda \in \FF^{\times}$.

Fix a field $\K$. Unless otherwise specified the ring throughout this worksheet will be $\K[x_1,\ldots,x_n]$. In the previous worksheet we introduced monomial orderings and leading terms. In this worksheet we return to the multivariable division algorithm, and then use it to study monomial ideals, initial ideals, and Gr\"obner bases. The main goal is to push the story all the way to a major finiteness theorem: every ideal of $\K[x_1,\ldots,x_n]$ is finitely generated.

Recall a \emph{monomial ordering} on $\K[x_1,\ldots,x_n]$ is a total well-ordering $<$ on $\Z^{n}_{\geq0}$ such that if $\alpha < \beta$ then $\alpha + \gamma < \beta+\gamma$ for all $\gamma \in \Z^{n}_{\geq0}$. Using the bijection between monomials in $\K[x_1,\ldots,x_n]$  and elements of $\Z^{n}_{\geq0}$ we think of a monomial ordering as giving us a way to compare monomials by saying $x^{\alpha} < x^{\beta}$ if and only if $\alpha < \beta$. In Worksheet 1.3 we saw several important monomial orders: lexicographic (lex) order, graded lexicographic (grlex) order, and graded reverse lexicographic (grevlex) order. Please review that worksheet for their definitions. 

Returning to our goal of having a division algorithm in $\K[x_1,\ldots,x_n]$ we are faced with two challenges compared to the familiar setting of $\K[x]$: 
\begin{enumerate}[label=(\roman*),ref=\theenumi\roman*]
\item  In several variables the notion of ``leading term'' is ambiguous. In one variable, the leading term is simply the term of highest degree. In several variables this no longer suffices: for instance, what should the leading term of $x^2+xy+y^2$ be? All three terms have total degree~2.
\item  We want a division algorithm that can answer questions about ideals, i.e., membership, intersections, sums, products, radicals. In $\K[x]$ every ideal is principal, so dividing by a single generator is enough. For $n>1$, however, $\K[x_1,\ldots,x_n]$ is no longer a PID, so ideals generally require multiple generators. We therefore need a way to divide a polynomial by an entire \emph{list} of polynomials.
\end{enumerate}
As seen on Worksheet 1.3, once a monomial ordering has been fixed, we give a meaningful definition of leading term, leading monomial, and leading coefficient; solving issue (i). Note while we normally write $\LT(f)$, $\LM(f)$ and $\LC(f)$ for these quantities; they do depend on the chosen order $<$, and so would more properly be called $\LT_{<}(f)$, $\LM_{<}(f)$ and $\LC_{<}(f)$ to highlight this fact. 

To resolve issue (ii), we generalize long division as follows. In single-variable polynomial long division, we repeatedly ask: ``Does the leading term of the divisor divide the current leading term of the dividend?'' If yes, we subtract off the appropriate multiple; if not, we are done. The multivariable algorithm works the same way, except that instead of checking a single divisor we scan through an \emph{ordered} list of divisors $G=(g_1,\ldots,g_s)$ and use the first one whose leading term divides the current leading term. More precisely, the algorithm maintains a ``working polynomial'' $p$ (initialized to $f$), quotients $q_1,\ldots,q_s$ (initialized to $0$), and a remainder $r$ (initialized to $0$), then repeats:
\begin{itemize}
\item \textbf{Try to Divide.} Walk through the list $g_1,g_2,\ldots,g_s$ in order. If you find some $g_i$ whose leading term $\LT(g_i)$ divides $\LT(p)$, subtract the appropriate multiple: update $q_i\leftarrow q_i+\LT(p)/\LT(g_i)$ and $p\leftarrow p-(\LT(p)/\LT(g_i))\,g_i$. Then return to the start of the list and try again with the new~$p$.

\item \textbf{Move to Remainder.} If \emph{no} $g_i$ in the list has a leading term dividing $\LT(p)$, then the current leading term of $p$ cannot be reduced further. Move it to the remainder: $r\leftarrow r+\LT(p)$ and $p\leftarrow p-\LT(p)$.
\end{itemize}
We continue until the working polynomial $p$ is equal to $0$. Proving this algorithm terminates as expected, which you will do later in this worksheet, amounts essentially to proving the following theorem

\begin{theorem}\label{thm:div-alg}
Fix a monomial ordering $<$ on $\K[x_1,\ldots,x_n]$ and let $G=(g_1,\ldots,g_s)$ be an ordered list of nonzero polynomials. If $f\in \K[x_1,\ldots,x_n]$ then there exist $q_1,\ldots,q_s,r\in \K[x_1,\ldots,x_n]$ such that
\[
f=q_1g_1+\cdots+q_sg_s+r,
\]
where no monomial appearing in $r$ is divisible by any of $\LM(g_1),\ldots,\LM(g_s)$. \end{theorem}

Written in pseudocode the algorithm for multivariable polynomial division is shown below.

\begin{algorithm}[H]
\caption{Division by an ordered list $G=(g_1,\ldots,g_s)$}
\begin{algorithmic}
\Require $f, g_1,\ldots,g_s$
\Ensure $q_1,\ldots,q_s,r$
\State $q_1,\ldots,q_s \gets 0$; $r\gets 0$; $p\gets f$
\While{$p\neq 0$}
    \State $\mathrm{divides}\gets 0$
    \For{$i=1,\ldots,s$}
        \If{$\LT(g_i)$ divides $\LT(p)$}
            \State $q_i\gets q_i+\LT(p)/\LT(g_i)$
            \State $p\gets p-(\LT(p)/\LT(g_i))g_i$
            \State $\mathrm{divides}\gets 1$
            \State \textbf{break}
        \EndIf
    \EndFor
    \If{$\mathrm{divides}=0$}
        \State $r\gets r+\LT(p)$
        \State $p\gets p-\LT(p)$
    \EndIf
\EndWhile
\end{algorithmic}
\end{algorithm}

Question~2 reveals an important new feature of polynomial division in several variables. Unlike the one-variable case, the remainder obtained by dividing a polynomial \(f\) by polynomials \(g_1,\dots,g_t\) depends on two choices: i) the order of our list of divisors and ii) the monomial order we have fixed. Because of this, division does \emph{not} immediately solve the ideal membership problem in several variables. In other words, even if dividing $f$ by the list $(g_1,\dots,g_t)$ produces a nonzero remainder, it may still happen that $f\in \langle g_1,\dots,g_t\rangle$.  So, in several variables, a nonzero remainder does \emph{not} by itself show that $f \notin \langle g_1,\dots,g_t\rangle$. We will eventually develop a way to fix this problem. Before we do, however, it is useful to study a special class of ideals for which ideal membership \emph{is} controlled purely by divisibility of monomials.

\begin{definition}
An ideal $I\subseteq \K[x_1,\ldots,x_n]$ is a \emph{monomial ideal} if it can be generated by monomials.
\end{definition}

We shall see shortly that a monomial ideal is entirely determined by the set of monomials belonging to the ideal. Since monomials in $\K[x_1,\ldots, x_n]$ are in bijection with $\Z^{n}_{\geq0}$ this means we can -- and often do think of monomial ideals as corresponding to subsets of $\Z^{n}_{\geq0}$. With this viewpoint, the following classical result guarantees that monomial ideals are well-behaved.

\begin{theorem}[Dickson's Lemma]\label{thm:dickson}
Every subset of $\Z_{\geq0}^n$ has only finitely many minimal elements with respect to the coordinatewise order~$\leq$. 
\end{theorem}

Motivated by this theorem we say a \emph{minimal monomial generating set} for a monomial ideal $I$ is a finite set of monomials $\{m_1,\ldots,m_t\}$ such that $I=\langle m_1,\ldots,m_t\rangle$ and no $m_i$ divides $m_j$ for $i\neq j$. Equivalently, no generator is redundant: removing any $m_i$ from the set yields a strictly smaller ideal.

Monomial ideals are easier to study because membership is determined entirely by divisibility of monomials, and Dickson’s Lemma gives strong finiteness properties. For a general ideal, we would like to recover similar structures by extracting the leading term of each polynomial. This leads to the following.

\begin{definition}
Let $<$ be a monomial order on $\K[x_1,\ldots,x_n]$.  The \emph{initial ideal} of a nonzero ideal $I \subset \K[x_1,\ldots,x_n]$ with respect to $<$ is:
\[
\init_{<}(I) \coloneqq \langle \LT(f) \;\; \big| \;\; f \in I \rangle. 
\]

\end{definition}

It is standard to set $\init_{<}(\langle0\rangle)=\langle 0\rangle$. Note a priori there is no reason for Gr\"{o}bner bases to exist in general or be interesting. In the next few exercises you will show that Dickson's Lemma not only implies that Gr\"{o}bner bases exist, but also they will give an answer to the ideal membership problem.

	 I_1\subseteq I_2\subseteq I_3\subseteq \cdots
	\]
	stabilizes in the sense that there exists $N\in \Z_{\geq1}$ such that $I_{n}=I_{n+1}$ for all $n\geq N$. 
\end{definition}

It is common to say a ring is Noetherian if it has the ACC property where ACC is short for ascending chain condition. You will show below a ring $R$ being Noetherian is equivalent to every ideal of $R$ being finitely generated. Thus, you have proven that $\K[x_1,\ldots,x_n]$ is Noetherian. Since a field $\K$ is Noetherian, you have actually proven a special, but important case of the following theorem. 

\begin{theorem}[Hilbert's Basis Theorem]
	Let $R$ be a ring. If $R$ is Noetherian, then $R[x]$ is Noetherian.
\end{theorem}

The remainder of this worksheet is devoted to proving the equivalence between the ACC and finite generation of ideals, and proving Hilbert's basis theorem in general.

    I_1\subseteq I_2\subseteq I_3\subseteq \cdots.

        \item If $J$ is finitely generated, say $J=\langle a_1,\ldots,a_t \rangle$, explain why there exists an integer $N\in \Z_{\geq1}$ such that $a_1,\ldots,a_t \in I_{N}$. Conclude that the chain of ideals stabilizes.
        
        \item Explain why you just showed that if every ideal of $R$ is finitely generated then $R$ is Noetherian.

If $\langle a_1\rangle \neq I$, pick $a_2\in I\setminus \langle a_1\rangle$, obtaining $\langle a_1\rangle \subsetneq \langle a_1,a_2\rangle$. Continue in this way. Use the ascending chain condition to show this process must terminate, and conclude that $I$ is finitely generated.

    \end{enumerate}

\begin{problem}{13} \textbf{Hilbert's Basis Theorem.} 
    
    Let $R$ be a Noetherian ring, and let $I\subseteq R[x]$ be an ideal. For $k\in \Z_{\geq0}$, define
    \[
    J_k:=\left\{a\in R \;\; \bigg| \;\; \begin{matrix} \text{there exists $f\in I$} \\ \text{$\LT(f)  = ax^k$} \end{matrix} \right\} \cup\{0\}.
    \]
    Equivalently, $a\in J_k$ if and only if there exists $f\in I$ with $\LT(f)=ax^k$ (or $a=0$). Thus $J_k$ is the set of all leading coefficients of polynomials of degree $k$ in $I$, together with $0$. Note that we are viewing $R[x]$ as a polynomial ring in the single variable $x$ over $R$. We define the leading term of a polynomial $f = a_k x^k + a_{k-1}x^{k-1} + \cdots + a_1 x + a_0$ (with $a_k \neq 0$) to be $\LT(f) = a_k x^k$.
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Show that $J_k$ is an ideal of $R$ for all $k\geq0$. 
        \item\label{ex-part:hbt-chain} Show that the $J_{k}$ form an ascending chain $J_0\subseteq J_1\subseteq J_2\subseteq \cdots$. 
        \item Let $L=\bigcup_{d\geq 0} J_d$. Show that $L$ is an ideal in $R$.
        \item Prove that there exists $N \in \Z_{\geq0}$ such that $L = J_{k}$ for all $k\geq N$. (Hint: Use that $R$ is Noetherian.)
    \end{enumerate}
    For each $0\leq k\leq N$, choose generators $a_{k,1},\ldots,a_{k, r_k}$ for $J_k$ and polynomials $f_{k,i} \in I$ such that $\LT(f_{k,i}) = a_{k,i}x^k$. We will now show that $I$ is equal to
    \[
    M \coloneqq \left\langle f_{k,i} \;\; \big| \;\; 0 \leq k \leq N, \; 1 \leq i \leq r_k \right\rangle.
    \]
    The inclusion $M\subset I$ is clear. We prove the reverse inclusion $I\subseteq M$ by induction on degree. For the base case suppose $g \in I$ and $\deg(g) = 0$. Since $g \in I$ and $\LT(g) = cx^0=c$, we have $c\in J_{0}$. Using that $J_{0}=\langle a_{0,1},\ldots,a_{0,r_0}\rangle$ we may write $c$ as $c=b_{i}a_{0,}+\cdots b_{r_0}a_{0,r_0}$ with $b_i\in R$. For each $i$ we have $\LT(f_{0,i})=a_{0,i}x^{0}=a_{0,i}$ so $f_{0,i}=a_{0,i}$. Hence $a_{0,i}$ is in $M$ for all $i$ implying $g \in M$ proving the base case. 

    Now suppose every element of $I$ with degree less than $d$ belongs to $M$. Let $g\in I$ with $\deg(g)=d$ and $\LT(g)=cx^d$. Our goal is to find $h \in M$ with $\LT(h) = cx^d$, so that $g - h \in I$ has degree less than $d$ and the inductive hypothesis applies. Set $n = \min\{d, N\}$. Since the ascending chain $J_0 \subseteq J_1 \subseteq \cdots$ stabilizes at $J_N$, we have $J_d = J_n$, so $c \in J_n = \langle a_{n,1}, \dots, a_{n,r_n} \rangle$. Write
    \[
    c = b_1 a_{n,1} + \cdots + b_{r_n} a_{n,r_n}
    \]
    for some $b_i \in R$, and define
    \[
    h = \sum_{i=1}^{r_n} b_i x^{d-n} f_{n,i}.
    \]
    (Note: if $d\leq N$ then $n=\min\{d,N\}=d$ and the $x^{d-n}=1$.) Each $f_{n,i} \in M$, so $h \in M$.
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
    \setcounter{enumii}{4}
        \item Show that  $\LT(h) = cx^d$. From this deduce that $g-h \in I$ and either $\deg(g-h) < d$ or $g-h=0$.
        \item Using the inductive hypothesis conclude that $g-h \in M$, and show this implies $g \in M$. Conclude that $I \subset M$. Explain why this concludes the proof of Hilbert's Basis Theorem. 
        \item Prove via induction on $n$ that if $R$ is Noetherian then $R[x_1,\ldots,x_n]$ is Noetherian.  
        \item Compare this proof to the proof of Dickson's Lemma we gave above. 
    \end{enumerate}
\end{problem}

This worksheet has introduced two important concepts that will recur throughout the course: the usefulness of Gr\"{o}bner bases and the importance of finiteness conditions such as Noetherianity. A significant open thread raised by this worksheet is  that while we know theoretically that Gr\"{o}bner bases exist, we do not yet know how to actually compute them. We will return to this question later.

It is fitting that Gr\"{o}bner bases and Noetherianity are so linked. Hilbert's original proof of his Basis Theorem was non-constructive, even over $\K[x_1,\dots,x_n]$, which -- as the story goes -- came as a shock to many mathematicians of the time. Paul Gordan, a prominent algebraist, reportedly remarked, perhaps seriously, perhaps in jest, \textit{``This is not mathematics. This is theology.''} (Gordan later gave another proof of the theorem, one which in a sense also proved Dickson's Lemma some 10 years before Dickson.) On the other hand, Gr\"{o}bner bases have come to serve as the foundational tool of computational commutative algebra, allowing us to compute and make explicit a great many things that Hilbert's methods could not.
