    {Worksheet 7.2: Integral Extension Pt. II}
\end{center}

Throughout this course, ``ring'' means \textit{commutative} ring with unity and all ring homomorphisms preserve unity. In the previous worksheet we introduced integral elements and integral extensions. We saw that integrality is the mechanism by which monic polynomial equations turn algebraic generation into module generation. The goal of this worksheet is to see what integrality does to prime ideals and to use this to make dimension computable for finitely generated algebras over a field.

Let $\phi:R\to S$ be a ring map. Recall that $\phi$ induces a continuous map
\[
\begin{tikzcd}[row sep=.3em, column sep=4em]
\Spec(S) \arrow[r, "\phi^{\#}"] & \Spec(R),
\end{tikzcd}
\qquad
\fq\longmapsto \phi^{-1}(\fq).
\]
The central question of the first half of this worksheet is: \emph{how much of $\Spec(R)$ is seen by $\Spec(S)$, and how do chains of prime ideals compare?} For integral extensions the answer is surprisingly rigid. Prime ideals downstairs have primes upstairs, chains can be lifted upward, and no two comparable upstairs primes can lie over the same downstairs prime. These facts are called \emph{Lying Over}, \emph{Going Up}, and \emph{Incomparability}. A complementary statement, \emph{Going Down}, is more delicate: it is true for flat maps, and for integral extensions of domains when the base is integrally closed.

In the second half of the worksheet we use these theorems to prove and apply \emph{Noether Normalization}. The slogan is that every finitely generated algebra over a field is finite over a polynomial ring. Thus, after a finite map, an affine algebra behaves like affine space of some dimension. This is one of the main reasons polynomial rings control dimension theory.

\begin{problem}{1}
    \textbf{A First Consequence for Maximal Ideals.} 
    
    Let $\phi:R \to S$ be an injective integral ring map.
    \begin{enumerate}
        \item Suppose $S$ is a field and $R$ is a domain. Prove that $R$ is a field. Hint: If $r \neq 0 \in R$, then $\phi(r)^{-1}\in S$ is integral over $R$; write a monic equation for $\phi(r)^{-1}$ and multiply by a suitable power of $\phi(r)$.
        \item Suppose $S$ is a domain and $R$ is a field. Prove that $S$ is a field.
        \item Let $\phi:R\to S$ be an integral ring map and let $\fq\in\Spec(S)$ with contraction $\fp=\phi^{-1}(\fq)$. Show that $S/\fq$ is integral over $R/\fp$.
        \item Deduce that if $\fq$ is maximal, then $\fp$ is maximal.
        \item Prove the converse: If $\fp$ is maximal, then $\fq$ is maximal. Hint: $S/\fq$ is an integral domain integral over the field $R/\fp$.
        \item Conclude that under an integral ring map, a prime ideal upstairs is maximal if and only if its contraction downstairs is maximal.
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}
        \item If $r \neq 0 \in R$ then $\phi(r)^{-1} \in S$ is integral over $R$ with 
        \[(\phi(r)^{-1})^n + r_1(\phi(r)^{-1})^{n-1} + \cdots + r_n = 0\]
        And, 
        \begin{align*}
           0 &= \phi(r)^n\left((\phi(r)^{-1})^n + r_1(\phi(r)^{-1})^{n-1} + \cdots + r_n\right) \\
           0 &= 1 + r_1\phi(r) + \cdots + r_n\phi(r)^n  \\
           \implies 1 &= \phi(r)(r_1 + \cdots + r_n(\phi(r))^{n-1})
        \end{align*}
        That is $\phi(r)$ has a inverse in $\phi(R)$. Since $\phi$ is injective this implies that $r$ has an inverse in $R$. Since $\phi(r)$ was an arbitrary element this implies that every non-zero element in $R$ has a inverse and $R$ is a field. 

        \item We know that $s \neq 0 \in S$ is integral over $R$. That is, we have the minimal mnic polynomial, 
        \[s^n + r_1s^{n-1} + \cdots + r_n = 0\]
        Since $R$ is a field we have, 
        \begin{align*}
            0 &= s^n + r_1s^{n-1} + \cdots + r_n  \\ 
            r_n &= - (s^n + r_1s^{n-1} + \cdots + r_{n-1}s) \\ 
            -r_n &= s(s^{n-1} + r_1s^{n-2} + \cdots + r_{n-1}) \\ 
            1 = \frac{-r_n}{-r_n} &= s \left (\frac{1}{r_n}s^{n-1} + \frac{r_1}{r_n}s^{n-2} + \cdots + \frac{r_{n-1}}{r_n} \right )  
        \end{align*}
        That is $s$ has an integral inverse in $S$. Since $s$ was an arbitrary non-zero element this shows that $S$ is a field. 
        

        \item Since $S$ is integral over $R$ for any $s\in S$ we have, 
        \[s^n + r_1s^{n-1} + \cdots + r_n = 0\]
        Then for $\overline{s} \in S/\fq$ we have, 
        \[\overline{s}^n + \overline{r_1}\,\overline{s}^{n-1} + \cdots + \overline{r_n} = 0\]
        But $\overline{r_i} = r_i + \fq  \in S = r_i + \fq \cap R \in R = r_i + \fp$. 
        Hence the coefficients $\overline{r_i}$ lie in $R/\fp$ and $\overline{s}$ is integral over $R/\fp$. 

        \item If $\fq$ is maximal, then $S/\fq$ is a field and then by part 1 $R/fp$ must be a field and hence $\fp$ must be maximal. 
        
        \item If $\fp$ is maximal, then $R/\fp$ is a field, and by part 2 $S/\fq$ is integral over $R/\fp$. Then for any $\overline{s} \neq 0 \in S/\fq$ we have the minimal monic polynomial, 
        \[\overline{s}^n + \overline{r_1}\,\overline{s}^{n-1} + \cdots + \overline{r_n} = 0\]
        And since $R/\fp$ is a field we have,
        \begin{align*}
           \overline{r_n} &= - \overline{s}(\overline{s}^{n-1} + \overline{r_1}\,\overline{s}^{n-2} + \cdots + \overline{r_{n-1}} ) \\ 
           1 = \frac{\overline{r_n}}{\overline{r_n}} &= - \overline{s} \left (\frac{1}{\overline{r_n}}\overline{s}^{n-1} + \frac{\overline{r_1}}{{\overline{r_n}}}\,\overline{s}^{n-2} + \cdots + \frac{\overline{r_{n-1}}}{\overline{r_n}} \right)
        \end{align*}
        Thus $\overline{s}$ has an inverse in $S/\fq$, and hence $S/\fq$ is a field and $\fq$ is maximal. 

        \item Thus under an integral ring map $\fq$ is maximal if and only if $\fp$ is maximal. 
    \end{enumerate}
\end{solution}

\begin{problem}{2} \textbf{Prime Ideals Near a Point.} 
    
    Let $\phi:R\to S$ be a ring map and let $\fp\in\Spec(R)$. Write $U=R\setminus \fp$ and $S_{\fp}=U^{-1}S$.
    \begin{enumerate}
        \item Prove that if $\fq\in\Spec(S)$, then $\phi^{-1}(\fq)$ is a prime ideal of $R$.
        \item Prove that prime ideals of $S_{\fp}$ are in bijection with prime ideals $\fq\in\Spec(S)$ such that $\fq\cap \phi(U)=\varnothing$. Equivalently, these are the primes $\fq$ such that $\phi^{-1}(\fq)\subseteq \fp$.
        \item Suppose that $R\subseteq S$ is an injective ring map. Show that $S_{\fp}\neq 0$. Hint: If $1=0$ in $S_{\fp}$, then some element of $R\setminus \fp$ becomes zero in $S$.
        \item Suppose that $S$ is integral over $R$. Prove that $S_{\fp}$ is integral over $R_{\fp}$.
        \item Let $\mathfrak Q\in\Spec(S_{\fp})$ and let $\fq\subset S$ be the corresponding prime ideal. Show that if $\phi^{-1}_{\fp}(\mathfrak Q)=\fp R_{\fp}$, then $\phi^{-1}(\fq)=\fp$.
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}
        \item If $\fq \in \Spec(S)$ then $\fq$ is a prime ideal and we know that pre-images of prime ideals are prime, hence $\phi^{-1}(\fq)$ is a prime ideal of $R$. 
        \item We have already proved this. 
        \item Assume for contradiction that $S_{\fp} = 0$. Then $\frac{0}{1} \sim \frac{1}{1} \implies \exists t \in \phi(R \setminus \fp)$ such that $t(1 - 0) = 0 \implies t = 0 \implies t \in \ker(\phi)$ but $\phi$ is injective so $t$ must be $0$ and $0 \notin R \setminus \fp$. A contradiction. Hence, $S_{\fp} \neq 0$. 
        \item In Worksheet 7.1 Problem 17 that if $S$ is integral over $R$ and $U\subset R$ is multiplicatively closed, then $U^{-1}S$ is integral over $U^{-1}R$. Since $S_{\fp} = U^{-1}S$ with $U = R \setminus \fp$, we have $S_{\fp}$ is integral over $R_{\fp}$. 
        \item We have, 
         \[ \begin{tikzcd} R \arrow[r, "\phi"] \arrow[d] & S \arrow[d] \\ R_\mathfrak{p} \arrow[r, "\phi_\mathfrak{p}"] & S_\mathfrak{p} \end{tikzcd}\]
         Where
        \[\phi^{-1}_{\fp}(\mathfrak{Q}) = \fp R_{\fp} = \mathfrak{Q} \cap R_{\fp}\]
        And 
        \[\phi^{-1}({q}) = \phi^{-1}(\mathfrak{Q} \cap S) \cap R = \phi^{-1}_{\fp} (\mathfrak{Q}) \cap R = \fp R_{\fp} \cap R = \fp \]
    \end{enumerate}
\end{solution}

\begin{problem}{3}
    \textbf{Lying Over.} 
    
    Let $R\subseteq S$ be an integral extension of rings. The goal of this exercise is to prove the \emph{Lying Over Theorem}: for every $\fp\in\Spec(R)$, there exists $\fq\in\Spec(S)$ such that $\fq\cap R=\fp$.
    \begin{enumerate}
        \item Fix $\fp\in\Spec(R)$. Explain why $S_{\fp}$ is a nonzero integral extension of $R_{\fp}$.
        \item Choose a maximal ideal $\mathfrak N\subset S_{\fp}$. Use the theorem from the previous worksheet saying that maximal ideals contract to maximal ideals under integral extensions to prove
        \[
        \mathfrak N\cap R_{\fp}=\fp R_{\fp}.
        \]
        \item Pull $\mathfrak N$ back to a prime ideal $\fq\subset S$ and prove that $\fq\cap R=\fp$.
        \item State the correct version of Lying Over for a not-necessarily-injective integral map $\phi:R\to S$. Hint: Which prime ideals of $R$ can possibly be contractions of prime ideals of $S$?
        \item Translate Lying Over into a statement about the image of the map $\phi^{\#}:\Spec(S)\to\Spec(R)$.
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}
        \item In Worksheet 7.1 Problem 17 that if $S$ is integral over $R$ and $U\subset R$ is multiplicatively closed, then $U^{-1}S$ is integral over $U^{-1}R$. Since $S_{\fp} = U^{-1}S$ with $U = R \setminus \fp$, we have $S_{\fp}$ is integral over $R_{\fp}$. 
        \item Since maximal ideals contract to maximal ideals under integral extensions we know that $\phi_{\fp}^{-1}(\mathfrak{N})$ is a maximal ideal in $R_{\fp}$. Since there is only one maximal ideal in $R_{\fp}$, namely $\fp R_{\fp}$, we must have 
        \[\mathfrak{N} \cap R_{\fp} = \phi_{\fp}^{-1}(\mathfrak{N}) = \fp R_{\fp} \]

        \item From Problem 2 part 5, we have that if $\fq \subset S$ corresponds to $\mathfrak{N}$ then 
        \[\phi^{-1}(\fq) = \fq \cap R = \fp\]

        \item From the first isomorphism theorem we have $R/\ker(\phi) \iso \phi(R) \subseteq S$. Since $\phi$ is integral, prime ideals in $S$ correspond to prime ideals in $R/\ker(\phi)$ which also correspond to prime ideals in $R$ containing $\ker(\phi)$. So the Lying Over Theorem  for not injective integral mps says that for every $\fp\in\Spec(R)$ such that $\ker(\phi) \subseteq \fp$, there exists $\fq\in\Spec(S)$ such that $\fq\cap R=\fp$.
        
        \item The image of the map $\phi^{\#}$ is surjective onto $\Spec(R/\ker(\phi))$, or the $\im(\phi^{\#})$ is precisely the set of prime ideals in $R$ containing the kernel of the map $\phi$. 
    \end{enumerate}
\end{solution}

\begin{problem}{4}
    \textbf{First Examples of Lying Over.} 
    
    Let $\K$ be a field.
    \begin{enumerate}
        \item Show that $\K[t]$ is integral over $\K[t^2]$. For a prime ideal $\fq\subset \K[t]$, compute $\fq\cap \K[t^2]$ when $\fq=\langle t-a\rangle$.
        \item Assume $\K$ is algebraically closed and $\operatorname{char}(\K)\neq 2$. For $u=t^2$, describe the primes of $\K[t]$ lying over the maximal ideal $\langle u-a\rangle\subset \K[u]$ when $a=0$ and when $a\neq 0$.
        \item Show that $\K[t]$ is integral over $\K[t^2,t^3]$. Identify the prime of $\K[t]$ lying over the cusp point $\langle t^2,t^3\rangle\subset \K[t^2,t^3]$.
        \item Show that $\Z[i]$ is integral over $\Z$. Compute the primes of $\Z[i]$ lying over $\langle 2\rangle$, $\langle 3\rangle$, and $\langle 5\rangle$ by factoring $x^2+1$ modulo $2$, $3$, and $5$.
        \item Let $R\subseteq S$ be integral and let $I\subset R$ be an ideal. Prove that the induced map
        \[
        \{\fq\in\Spec(S)\mid \fq\supseteq IS\}\longrightarrow \{\fp\in\Spec(R)\mid \fp\supseteq I\}
        \]
        is surjective.
    \end{enumerate}    
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}
        \item $\K[t]$ is module-finite over $\K[t^2]$ and hence is integral over $\K[t^2]$.  
        \[\fq \cap \K[t^2] = \{p(t^2) \in \K[t^2] \mid p(a^2) = 0\} = \langle t^2 - a^2 \rangle \]

        \item Prime ideals that lie over the maximal ideal $ \fp =\langle t^2 - a \rangle \subseteq \K[t^2]$ are maximal ideals $\fq$ in $\K[t]$ such that $\fq \cap \K[t^2] =  \fp$. Maximal ideals in $\K[t]$ are of the form $\langle t - b \rangle$, and by part 1 we know that $\langle t - b \rangle$ lies over $\fp$ iff $b^2 = a$. 
        
        If $a = 0$, 
        \[\langle t - b \rangle \cap \K[t^2] = \langle t^2 \rangle \implies b = 0\] 
        So there is a single prime ideal $\langle t \rangle$ lying over $\langle t^2 \rangle$.  

        If $a \neq 0$, then $b = \pm \sqrt{a}$ since $\K$ is algebraically closed. Hence, two prime ideals lie over $\fp$, $\langle t - \sqrt{a} \rangle$ and $\langle t + \sqrt{a} \rangle$. 

        \item $\K[t]$ is module-finite over $\K[t^2, t^3]$ and hence is integral over $\K[t^2, t^3]$. Primes of $\K[t]$ are of the form $\langle t - b \rangle$ for $b \in \K$ and $\langle 0 \rangle$. If  $\langle t - b \rangle$ lies over $\langle t^2, t^3 \rangle$ then 
        \[t^2, t^3 \in \langle t - b \rangle \implies b^2 = b^3 = 0 \implies b = 0\]
        Thus, $\langle t \rangle$ lies over $\langle t^2, t^3 \rangle$. 


        \item $\Z[i]$ is module-finite over $\Z$ and hence is integral over $\Z$. We know $\Z[i] \iso \Z[x]/\langle x^2 + 1 \rangle$, and primes of $\Z[i]$ containing $\langle p \rangle$ correspond to prime ideals of $\Z[i]/\langle p \rangle$. Then, 
        \[\Z[i]/\langle p \rangle \iso \F_p [i] \iso \F_p[x]/\langle x^2 + 1 \rangle \]
        so primes correspond to irreducible factors of $\langle x^2 + 1 \rangle$ in $\F_p[x]$ since they generate the ideals containing it. Therefore, 
        \begin{itemize}
            \item $x^2 + 1  = (x+1)^2  \pmod 2$ so $\langle 1 + i \rangle$ lies over $\langle 2 \rangle$.
            \item $x^2 + 1$ is irreducible modulo $3$ since $3 \pmod 4 \equiv 3$ and thus $-1$ is not a square, so $\langle 3 \rangle$ lies over $\langle 3 \rangle$.
            \item $x^2 + 1 = (x+2)(x-2) \pmod 5$ so $\langle  i + 2\rangle$ and $\langle i - 2\rangle$ lie over $\langle 5 \rangle$.
        \end{itemize} 

        \item Let $\fp \in \Spec(R)$ such that $I \subseteq \fp$. Then by laying over we know that there exists $\fq in \Spec(S)$ such that $\fq \cap R = \fp$. Since $\fq$ is an ideal and we have $I \subseteq \fp \subseteq \fq$ as sets, $IS \subseteq \fq$. Hence, proving surjectivity. 
        
        
        % We know that prime ideals of the form $\fp \supseteq I$ in $R$ correspond to prime ideals in $R/I$. 
        
        % \[\begin{tikzcd}
        %     R \arrow[r, "\psi"] \arrow[d, "\pi_1"]&S \arrow[d, "\pi_2"] \\ 
        %     R/I \arrow[r, "\phi"] & S/IS
        % \end{tikzcd}\]
        
        % From Problem 1 we know that the image of the map $\phi^{\#}$ is surjective onto $\Spec(R/\ker(\phi))$, or the $\im(\phi^{\#})$ is precisely the set of prime ideals in $R$ containing the kernel of the map $\phi$. 
    \end{enumerate}
\end{solution}

The next theorem explains how integral extensions interact with inclusions of prime ideals. Geometrically, if $\fp_0\subseteq \fp_1$ in $\Spec(R)$, then $\fp_1$ is a specialization of $\fp_0$. The theorem says that specializations can be lifted through an integral extension once the starting point upstairs has been chosen.

\begin{theorem}[Going Up]
Let $R\subseteq S$ be an integral extension. Suppose
\[
\fp_0\subseteq \fp_1
\]
are prime ideals of $R$, and suppose $\fq_0\in\Spec(S)$ satisfies $\fq_0\cap R=\fp_0$. Then there exists $\fq_1\in\Spec(S)$ such that
\[
\fq_0\subseteq \fq_1
\qquad\text{and}\qquad
\fq_1\cap R=\fp_1.
\]
More generally, every chain of prime ideals downstairs can be lifted to a chain of prime ideals upstairs, after choosing a prime upstairs over the first prime in the chain.
\end{theorem}

\begin{problem}{5} \textbf{Proof of Going Up.} 
    
    Let $R\subseteq S$ be integral, let $\fp_0\subseteq \fp_1$ be prime ideals of $R$, and let $\fq_0\in\Spec(S)$ lie over $\fp_0$.
    \begin{enumerate}
        \item Prove that $S/\fq_0$ is integral over $R/\fp_0$.
        \item Show that $\fp_1/\fp_0$ is a prime ideal of $R/\fp_0$.
        \item Apply Lying Over to the integral extension $R/\fp_0\subseteq S/\fq_0$ to produce a prime ideal $\fq_1/\fq_0$ of $S/\fq_0$ lying over $\fp_1/\fp_0$.
        \item Check that the corresponding prime ideal $\fq_1\subset S$ satisfies $\fq_0\subseteq \fq_1$ and $\fq_1\cap R=\fp_1$.
        \item Prove the chain version of Going Up by induction: if
        \[
        \fp_0\subsetneq \fp_1\subsetneq\cdots\subsetneq \fp_n
        \]
        is a chain in $R$ and $\fq_0$ lies over $\fp_0$, then there is a chain
        \[
        \fq_0\subsetneq \fq_1\subsetneq\cdots\subsetneq \fq_n
        \]
        in $S$ with $\fq_i\cap R=\fp_i$ for all $i$. 
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni 
    \begin{enumerate}
        \item This follows from Problem 1 part 5. 
        \item We know that prime ideals of $R/\fp_0$ correspond to prime ideals of $R$ containing $\fp_0$. Since $\fp_0 \subseteq \fp_1$, $\overline{\fp_1} = \fp_1/\fp_0$ is a prime ideal of $R/\fp_0$. 
        \item We know that the map $R/\fp_0 \to S/\fq_0$ is an integral map. By laying over, we know there exists $\overline{\fq_1}\subseteq S/\fq_0$ that lies over $\overline{\fp_1}$. Since prime ideals of $S/\fq_0$ correspond to prime ideals of $S$ containing $\fq_0$, $\overline{\fq_1} = \fq_1/\fq_0$. 
        \item Thus, $\fq_0 \subseteq \fq_1$. We know $\fq_1/\fq_0 \cap R/\fp_0 =  \fp_1/\fp_0$. $\fq_1 \cap R$ is the pre-image of the quotinent map $\pi: R \to R/\fp_0$, then, 
        \[\pi^{-1} ((\fq_1/\fq_0) \cap (R/\fp_0)) =  \pi^{-1}(\fp_1/\fp_0) = \fp_1\]
        \item Assume for $m < n$ that if
        \[
        \fp_0\subsetneq \fp_1\subsetneq\cdots\subsetneq \fp_m
        \]
        is a chain in $R$ and $\fq_0$ lies over $\fp_0$, then there is a chain
        \[
        \fq_0\subsetneq \fq_1\subsetneq\cdots\subsetneq \fq_m
        \]
        in $S$ with $\fq_i\cap R=\fp_i$ for all $i$.
        
        Then if $\fp_m \subsetneq \fp_n$, then by parts 3 and 4, we know there exists $\fq_n \subseteq S$ such that $\fq_m \subset \fq_n$ and $\fq_n \cap R = \fp_n$. Note that the inclusion $\fq_m \subsetneq \fq_n$ is strict since the contractions correspond to $\fp_m$ and $\fp_n$, and the inclusion $\fp_m \subsetneq \fp_n$ is strict. 
        
        Thus for any $n$, if
        \[
        \fp_0\subsetneq \fp_1\subsetneq\cdots\subsetneq \fp_n
        \]
        is a chain in $R$ and $\fq_0$ lies over $\fp_0$, then there is a corresponding chain
        \[
        \fq_0\subsetneq \fq_1\subsetneq\cdots\subsetneq \fq_n
        \]
        in $S$ with $\fq_i\cap R=\fp_i$ for all $i$. 
    \end{enumerate}
\end{solution}

\begin{problem}{6} \textbf{Incomparability.} 
    
    Let $R\subseteq S$ be an integral extension. The goal of this exercise is to prove: if $\fq\subseteq \fq'$ are prime ideals of $S$ and $\fq\cap R=\fq'\cap R$, then $\fq=\fq'$.
    \begin{enumerate}
        \item Set $\fp=\fq\cap R=\fq'\cap R$. Explain why $R/\fp\subseteq S/\fq$ is an integral extension of domains.
        \item Let $\mathfrak P=\fq'/\fq\subset S/\fq$. Show that $\mathfrak P\cap (R/\fp)=\langle 0\rangle$.
        \item Suppose $0\neq b\in \mathfrak P$. Choose a monic equation of minimal degree
        \[
        b^n+a_1b^{n-1}+\cdots+a_{n-1}b+a_n=0
        \]
        with $a_i\in R/\fp$. Prove that $a_n=0$.
        \item Use the fact that $S/\fq$ is a domain to get a monic equation for $b$ of smaller degree, a contradiction.
        \item Conclude Incomparability.
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}
        \item This follows from Problem 1 part 5. 
        \item If $r + \fq, r' + \fq \in \fq'/\fq$ then $r - r' \in \fq \subseteq \fq'$, that is, 
        \[\mathfrak{P} \cap (R/\fp) = \{r + \fp  \mid r + \fq \in \fq'/\fq \} = \{r + \fp \mid r \in \fq' \cap R\} = \{r + \fp \mid r \in \fp\} = \langle 0 \rangle\]
        \item Since $b \in \mathfrak{P}$ we have $b^n+a_1b^{n-1}+\cdots+a_{n-1}b \in \mathfrak{P}$ which implies that $-a_n \in \mathfrak{P}$. But since $a_n \in R/\fp \implies a_n \in \mathfrak{P} \cap (R/\fp) = \langle 0 \rangle \implies a_n = 0$. 
        \item  We have, 
        \begin{align*}
             b(b^{n-1}+a_1b^{n-2}+\cdots+a_{n-1})&=0 \\ 
        \end{align*}
        Since $S/\fq$ is a domain and $b \neq 0$ we must have
        \begin{align*}
             b^{n-1}+a_1b^{n-2}+\cdots+a_{n-1} &=0 \\ 
        \end{align*} 
        Thus giving us a monic equation for $b$ of smaller degree, which is a contradiction since we assumed minimality. Thus $b = 0$. 
        \item Part 4 shows us that $\mathfrak{P}$ has no non-zero elements and thus $\fq' / \fq = \langle 0 \rangle \implies \fq' = \fq$. 
    \end{enumerate}
\end{solution}

\begin{problem}{7}
    \textbf{Dimension is Preserved by Integral Extensions.} 
    
    Let $R\subseteq S$ be an integral extension of nonzero rings.
    \begin{enumerate}
        \item Let
        \[
        \fq_0\subsetneq \fq_1\subsetneq\cdots\subsetneq \fq_n
        \]
        be a chain of prime ideals in $S$. Use Incomparability to show that
        \[
        \fq_0\cap R\subsetneq \fq_1\cap R\subsetneq\cdots\subsetneq \fq_n\cap R
        \]
        is a chain of prime ideals in $R$. Deduce that $\dim(S)\leq \dim(R)$.
        \item Let
        \[
        \fp_0\subsetneq \fp_1\subsetneq\cdots\subsetneq \fp_n
        \]
        be a chain of prime ideals in $R$. Use Lying Over and Going Up to lift it to a chain of prime ideals in $S$. Deduce that $\dim(R)\leq \dim(S)$.
        \item Conclude that $\dim(R)=\dim(S)$.
        \item Use this theorem to compute the Krull dimension of each ring below.
        \begin{multicols}{2}
        \begin{enumerate}
            \item $\K[t^2,t^3]$.
            \item $\K[x,y]/\langle y^2-x^3\rangle$.
            \item $\K[x^2,xy,y^2]$.
            \item $\Z[i]$.
        \end{enumerate}
        \end{multicols}
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}
        \item Since the preimage of prime ideals is prime we know that each $\fq_i \cap R$ is prime, and by Incomparability all containments are strict. Then if  
        \[
        \fq_0\subsetneq \fq_1\subsetneq\cdots\subsetneq \fq_n
        \]
        is the longest chain of prime ideals in $S$, we get a chain of prime ideals in $R$ that has the same length
        \[
        \fq_0\cap R\subsetneq \fq_1\cap R\subsetneq\cdots\subsetneq \fq_n\cap R
        \]
        Hence $\dim(S) \leq \dim(R)$. 

        \item If \[
        \fp_0\subsetneq \fp_1\subsetneq\cdots\subsetneq \fp_n
        \]
        is a chain of prime ideals in $R$. Then by laying over we know that there exist prime ideals 
        \[
        \fq_0, \fq_1, \ldots, \fq_n
        \]
        lying over each corresponding ideal in $R$. Moreover from going over we know that for each pair $\fp_i \subseteq \fp_{i+1}$ such that there exists a $\fq_i$ that lies over $\fp_i$, there exists an prime ideal $\fq_{i+1} \subseteq S$ that lies over $\fp_{i+1}$ and contains $\fq_i$. Thus we get a chain of prime ideals in $S$,
        \[
        \fq_0\subsetneq \fq_1\subsetneq\cdots\subsetneq \fq_n
        \]
        Since the chains are of equal length we have $\dim (S) \geq \dim(R)$. 

        \item Thus, $\dim(S) = \dim(R)$. 
        
        \item \begin{enumerate}
            \item Since $\K[t]$ is integral over $\K[t^2, t^3]$, 
            \[\dim(\K[t^2, t^3]) = \dim (\K[t]) = 1\]
            \item Since $\K[x, y]/\langle y^2 - x^3 \rangle$ is module-finite (and thus integral over) $\K[x]$, 
            \[\dim(\K[x, y]/\langle y^2 - x^3 \rangle) = \dim (\K[x]) = 1\]
            \item Since $\K[x,y]$ is integral over $\K[x^2, xy, y^2]$,
            \[\dim(\K[x, y]) = \dim (\K[x^2, xy, y^2]) = 2\]
            \item Since $\Z[i]$ is integral over $\Z$,
            \[\dim(\Z[i]) = \dim (\Z) = 1\]
        \end{enumerate}
    \end{enumerate}
\end{solution}

\begin{problem}{8}
    \textbf{Closed Sets Under Integral Maps.} 
    
    Let $R\subseteq S$ be integral and let $\phi^{\#}:\Spec(S)\to\Spec(R)$ be the induced map.
    \begin{enumerate}
        \item Let $J\subseteq S$ be an ideal. Show that
        \[
        \phi^{\#}(\V(J))=\V(J\cap R).
        \]
        Hint: One inclusion is direct. For the other, apply Lying Over to $R/(J\cap R)\subseteq S/J$ after checking the kernel carefully.
        \item Deduce that $\phi^{\#}$ is a closed map.
        \item If $R\subseteq S$ is finite and $\kappa$ is algebraically closed, explain why the induced map on affine algebraic sets should be thought of as a finite-to-one map. Make this precise by showing that for each $\fp\in\Spec(R)$, the fibre ring $\kappa(\fp)\otimes_R S$ is finite-dimensional over $\kappa(\fp)$.
    \end{enumerate}    
\end{problem}

\begin{solution}
    \bbni 
    \begin{enumerate}
        \item $\phi^{\#}(\V(J)) \subseteq \V(J\cap R)$
        
        Let $\fp \in \phi^{\#}(\V(J))$. Then there exists some $\fq \in \V(J)$ such that $\phi^{\#}(\fq) =  \fq \cap R = \fp$. Since $J \subseteq \fq$ we must have $J \cap R \subseteq \fq \cap R = \fp$.
        
        $\phi^{\#}(\V(J)) \supseteq \V(J\cap R)$
        
        Let $\fp \in \V(J\cap R)$. Then by going up and laying over we have, 

        \[\begin{tikzcd}
            J \arrow[d] \arrow[r, phantom, "\subseteq"] & \fq \arrow[d]\\ 
            J \cap R \arrow[r, phantom, "\subseteq"] & \fp
        \end{tikzcd}\]
        Thus there exists an $\fq \subset S$ such that $\phi^{\#}(\fq) = \fp$. 

        \item Thus, $\phi^{\#}$ maps closed sets to closed sets and hence is a closed map. 
        
        \item If $R \subseteq S$ is integral and finite we have $S = R[s_1, \ldots, s_n]$. And the following diagram commutes, 
        
        \[\begin{tikzcd}
            R \arrow[r] \arrow[d] & \kappa(\fp)  \arrow[d]\\ 
            S  \arrow[r] & \kappa(\fp) \otimes_R S \\ 
        \end{tikzcd}\]
       Note that every element of $\kappa(\fp) \otimes_R S$ is a finite sum of simple tensors $k \otimes s$. Since $s \in S$, we can write $s = \sum_i r_i s_i$ for $r_i \in R$. Then, 
        \begin{align*}
            k \otimes s &= k \otimes \sum_i r_i s_i\\ 
            &= k  \sum_i r_i \otimes \sum_i s_i\\ 
            &=  \sum_i k r_i \otimes  s_i\\ 
            &=  \sum_i k r_i (1 \otimes  s_i)
        \end{align*}
        Hence $\kappa(\fp) \otimes_R S$ is spanned by simple tensors $(1 \otimes s_1), \ldots, (1 \otimes s_n)$ and thus is finite dimensional over $\kappa(\fp)$. 
    \end{enumerate}
\end{solution}

We now turn to \emph{Going Down}. For a ring map $R\to S$, Going Down means that chains can be lifted in the opposite direction: if $\fp_0\subseteq \fp_1$ in $R$ and a prime $\fq_1$ upstairs has already been chosen over $\fp_1$, then there should exist a prime $\fq_0\subseteq \fq_1$ lying over $\fp_0$.

Going Down is not automatic for integral extensions. There are two especially important settings where it holds. The first is flatness, which is a module-theoretic condition. The second is the classical integral theorem: an integral extension of domains satisfies Going Down when the base is integrally closed. We will not prove these now, but will explore their consequences. 

\begin{theorem}[Flat Going Down]
Let $\phi:R\to S$ be a flat ring map. Suppose $\fp_0\subseteq \fp_1$ are prime ideals of $R$ and $\fq_1\in\Spec(S)$ satisfies $\phi^{-1}(\fq_1)=\fp_1$. Then there exists $\fq_0\in\Spec(S)$ such that
\[
\fq_0\subseteq \fq_1
\qquad\text{and}\qquad
\phi^{-1}(\fq_0)=\fp_0.
\]
\end{theorem}

\begin{theorem}[Going Down for Normal Domains]
Let $A\subseteq B$ be an integral extension of domains. If $A$ is integrally closed in its fraction field, then $A\subseteq B$ satisfies Going Down.
\end{theorem}

\begin{problem}{9}  \textbf{Unpacking Going Down.} 
    
    Let $\phi:R\to S$ be a ring map.
    \begin{enumerate}
        \item Write the definition of Going Up for $\phi$ using a diagram of prime ideals.
        \item Write the definition of Going Down for $\phi$ using a diagram of prime ideals.
        \item Explain why Going Up is about lifting specializations and Going Down is about lifting generalizations.
        \item Prove directly that a localization map $R\to U^{-1}R$ satisfies Going Down. Hint: Use the correspondence between primes of $U^{-1}R$ and primes of $R$ disjoint from $U$.
        \item Give an example of a flat map that is not integral and an integral map that is not flat.
    \end{enumerate}    
\end{problem}
\begin{solution}
    \bbni
    \begin{enumerate}
        \item Going up says, given $\fq_0$ over $\fp_0$ and $\fp_0 \subseteq \fp_1$ we have
         \[\begin{tikzcd}
            \fq_0 \arrow[d, "\cap R", swap]  \arrow[r, dashed, "\subseteq"] & \fq_1 \arrow[d, "\cap R"] \\ 
            \fp_0 \arrow[r, "\subseteq"] & \fp_1 
        \end{tikzcd}\]

        \item Going down says, given $\fq_1$ over $\fp_1$ and $\fp_0 \subseteq \fp_1$ we have
         \[\begin{tikzcd}
            \fq_0  \arrow[d, "\cap R", swap] & \fq_1 \arrow[l, dashed, "\subseteq", swap] \arrow[d, "\cap R"] \\ 
            \fp_0  \arrow[r, "\subseteq"] & \fp_1 
        \end{tikzcd}\]

        \item In $\Spec(S)$ going up enables us to go from $\fq_0$ (over $\fp_0$) to $\fq_0 \subseteq \fq_1$, so $\fq_1 \in \V(\fq_0)$ and going down lets us find $\fq_0$ such that $\fq_1 \in \V(\fq_0)$. 
        
        \item In worksheet 6.1 we showed that a localized ring is a flat module and thus localization satisfies the flat going down theorem. 
        
        \item $\K \to \K[x]$ is flat but not integral and $\Z \to \Z/p\Z$ is integral but not flat. 
    \end{enumerate}
\end{solution}

\begin{problem}{10}
    \textbf{Consequences of Going Down.} 

    \begin{enumerate}
        \item Let $R\to S$ be faithfully flat. Prove that $\Spec(S)\to\Spec(R)$ is surjective. Hint: For $\fp\in\Spec(R)$, show that $\kappa(\fp)\otimes_R S\neq 0$.
        \item Deduce that if $R\to S$ is faithfully flat, then every finite chain of primes in $R$ can be lifted to a chain of primes in $S$. Conclude that $\dim(S)\geq \dim(R)$.
        \item Apply this to $R\to R[x]$ to recover the inequality $\dim(R[x])\geq \dim(R)+1$ for nonzero $R$. Hint: if $\fp_0\subsetneq\cdots\subsetneq\fp_n$ is a chain in $R$, use the explicit chain $\fp_0R[x]\subsetneq\cdots\subsetneq\fp_nR[x]\subsetneq\fp_nR[x]+\langle x\rangle$.
        \item Let $A\subseteq B$ be an integral extension of domains with $A$ integrally closed. Let $\fq\in\Spec(B)$ and $\fp=\fq\cap A$. Use Going Up, Going Down for Normal Domains, and Incomparability to prove
        \[
        \height(\fq)=\height(\fp).
        \]
        \item Specialize the previous part to a finite extension $\K[y_1,\ldots,y_d]\subseteq B$ where $B$ is a domain. What does it say about heights of primes in $B$?
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}
        \item Since $R \to S$ is faithfully flat we know that $\kappa(\fp) \otimes_R S = 0 \iff \kappa(\fp) = 0$, as $\kappa(\fp) \neq 0$ for any prime ideal we know $\kappa(\fp) \otimes_R S \neq 0$. That is $\kappa(\fp) \otimes_R S$ has at least one prime ideal, let this ideal be $\overline{\fq}$. Since pre-images of prime ideals are prime, $\overline{\fq}$ gives us a prime ideal $\fq \in S$. 
        \[\begin{tikzcd}
            S \arrow[r] & \kappa(\mathfrak{p})\otimes_R S \\
            R \arrow[u] \arrow[r] & \kappa(\mathfrak{p}). \arrow[u]
        \end{tikzcd}\]
        
        Since the diagram commutes, the two maps $R \to S \to \kappa(\fp) \otimes_R S$ and $R \to \kappa(\fp) \to \kappa(\fp) \otimes_R S$ must agree. Pulling back $\fq$ along the bottom-right path gives us  $(0) \in \Spec(\kappa(\fp))$, and $(0)$ contracts to $\fp \in \Spec(R)$. Thus $\fq$ lies over $\fp$. Hence every $\fp \in \Spec(R)$ is in the image and thus the map $\Spec(S) \to \Spec(R)$ is surjective. 
        
        % \[\begin{tikzcd}
        %     \Spec(S)  \arrow[d]  & \Spec(\kappa(\fp) \otimes_R S)  \arrow[d] \arrow[l]\\ 
        %     \Spec(R)  & \Spec(\kappa(\fp)) \arrow[l]
        % \end{tikzcd}\]
        
        \item Since $\Spec(S) \to \Spec(R)$ is surjective, given a longest prime chain in $R$, 
        \[\fp_0 \subsetneq \fp_1 \subsetneq \cdots \subsetneq \fp_n\]
        we can find a lift of $\fp_n$, $\fq_n \in \Spec(S)$. Then by going down, we can inductively find the corresponding prime chain of the same length in $S$,
        \[\fq_0 \subsetneq \fq_1 \subsetneq \cdots \subsetneq \fq_n\]
        so $\dim(S) \geq \dim(R)$.

        \item Let 
        \[\fp_0 \subsetneq \fp_1 \subsetneq \cdots \subsetneq \fp_n\] 
        be any chain chain in $R$ and define the ideal laying over $\fp_n$ as $\fq_n = \fp_n R[x]$ since $\fq \cap R = \fp$ and 
        \[R[x] / \fq_n = R/\fp_nR[x] = (R/\fp_n)[x]\]
        which is a domain. Then by part (2) we can inductively find a chain such that 
        \[\fq_0 \subsetneq \fq_1 \subsetneq \cdots \subsetneq \fq_n = \fp_n R[x]\]
        Note that $\fp_n R[x] \subsetneq \fp_n R[x] + \langle x \rangle$ since $x \notin \fp_nR[x]$ because every element of $\fp_n R[x]$ has its coefficients in $\fp_n$ but $x$ has leading coefficient $1$. 
        
        So we have a chain of length $n+1$ in $S$
        \[\fq_0 \subsetneq \fq_1 \subsetneq \cdots \subsetneq \fq_n = \fp_n R[x] \subsetneq \fp_n R[x] + \langle x \rangle\]
        so $\dim(S) \geq \dim(R) + 1$. 

        \item Going down for normal domains gives us that $\height(\fq) \geq \height(\fp)$. To see that $\height(\fq) \leq (\fp)$ we consider a longest chain that ends at $\fq$ in $B$. Restricting it to $A$ gives us a chain that ends at $\fp$. Moreover we know all inclusions in the chain are strict by Incomparability. 
        Hence, $\height(\fq) = \height(\fp)$. 

        \item Since $\K[y_1, \ldots, y_d]$ is a polynomial ring over a field it is a UFD and hence is integrally closed and the extension $\K[y_1, \ldots, y_d] \subseteq B$ is integral. Moreover note that any prime in $\K[y_1, \ldots, y_d]$ has height at most $d$. Then by part 4 for any prime ideal $\fq \in B$ we have, 
        \[\height(\fq) = \height(\fq \cap \K[y_1, \ldots, y_d]) \leq d\]
    \end{enumerate}
\end{solution}

We now move from arbitrary integral extensions to finitely generated algebras over a field. The next theorem is one of the main structural results in commutative algebra. It says that a finitely generated algebra over a field is finite over a polynomial subring.

\begin{theorem}[Noether Normalization]
Let $\K$ be a field and let $A$ be an algebra-finite $\K$-algebra. Then there exist algebraically independent elements $y_1,\ldots,y_d\in A$ such that $A$ is module-finite over the polynomial subring
\[
\K[y_1,\ldots,y_d]\subseteq A.
\]
If $A$ is a domain, then $d=\trdeg_{\K}\Frac(A)$.
\end{theorem}

The proof is an induction on the number of algebra generators. The key step is a change of variables trick: given a nonzero polynomial relation, we choose new variables so that one old variable satisfies a monic equation over the others.

\begin{problem}{11} \textbf{The Monic Change of Variables Trick.} 
    
    Let $0\neq f\in \K[x_1,\ldots,x_n]$. The goal of this exercise is to make a change of variables so that $f$ becomes monic in one variable. Write
    \[
    f=\sum_{\alpha} c_{\alpha}x_1^{\alpha_1}\cdots x_n^{\alpha_n},
    \]
    where $\alpha=(\alpha_1,\ldots,\alpha_n)\in \Z_{\geq 0}^n$, only finitely many $c_\alpha$ are nonzero, and $c_\alpha\in \K$. Choose an integer $N$ larger than every exponent appearing in every monomial of $f$; that is, $N>\alpha_i$ for every exponent vector $\alpha$ appearing in $f$ and every $i$.

    \begin{enumerate}
        \item Define
        \[
        w(\alpha)
        =
        \alpha_n+\alpha_1N+\alpha_2N^2+\cdots+\alpha_{n-1}N^{n-1}.
        \]
        Explain why this is the base-$N$ expansion of an integer whose ``digits'' are $\alpha_n,\alpha_1,\alpha_2,\ldots,\alpha_{n-1}$

      \item Since all these digits are strictly smaller than $N$, prove that if $w(\alpha)=w(\beta)$ then then $\alpha=\beta$. Thus the weights $w(\alpha)$ are distinct for distinct monomials of $f$.

        \item Introduce new variables
        \[
        y_i=x_i-x_n^{N^i}
        \qquad\text{for }1\leq i\leq n-1.
        \]
        Equivalently, $ x_i=y_i+x_n^{N^i}$. Substitute these expressions into a monomial
        \[
        x_1^{\alpha_1}\cdots x_{n-1}^{\alpha_{n-1}}x_n^{\alpha_n}.
        \]
        Show that the unique term obtained by choosing $x_n^{N^i}$ from every factor $(y_i+x_n^{N^i})^{\alpha_i}$ is $x_n^{w(\alpha)}$. All other terms contain at least one $y_i$ and have strictly smaller degree in $x_n$.

        \item Since the weights $w(\alpha)$ are distinct, there is a unique exponent vector $\alpha^{\max}$ appearing in $f$ for which $w(\alpha)$ is largest. Show that, after the substitution, the highest power of $x_n$ appearing in $f$ is
        \[
        c_{\alpha^{\max}}x_n^{w(\alpha^{\max})}.
        \]

        \item Conclude that, after multiplying $f$ by the nonzero scalar $c_{\alpha^{\max}}^{-1}$ we may regard $f$ as a monic polynomial in $x_n$ with coefficients in
        \[
        \K[y_1,\ldots,y_{n-1}].
        \]

        \item Let $A=\K[x_1,\ldots,x_n]/\langle f\rangle$. Prove that the image of $x_n$ in $A$ satisfies a monic polynomial with coefficients in the subalgebra generated by the images of
        \[
        y_1,\ldots,y_{n-1}.
        \]
        Therefore the image of $x_n$ is integral over this subalgebra.

        \item Show $A$ is generated as an algebra over $\K[y_1,\ldots,y_{n-1}]$ by the single integral element $x_{n}$. Deduce that $A$ is module-finite over $\K[y_1,\ldots,y_{n-1}]$. 
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}
        \item Since $\alpha \in \Z$, and all $\alpha_i < N$ we can write a base-$N$ expansion of the integer $(\alpha_n\alpha_1\alpha_2\cdots\alpha_{n-1})_N$ in this manner. 
        
        \item Assume $w(\alpha) = w(\beta)$. Then $(\alpha_n\alpha_1\alpha_2\cdots\alpha_{n-1})_N = (\beta_n\beta_1\beta_2\cdots\beta_{n-1})_N \implies \alpha = \beta$. 
        
        \item We have $\prod_i^n x_i^{\alpha_i} = x_n^{\alpha_n} \prod_i^{n-1} \left (y_i + x_n^{N^i} \right )^{\alpha_i}$. Picking just $x_n^{N^i}$ from every factor gives us, 
        \[ x_n^{\alpha_n} \prod_i^{n-1} \left (x_n^{N^i} \right )^{\alpha_i} = x_n^{\alpha_n} \prod_i^{n-1} x_n^{\alpha_i N^i} = x_n^{\alpha_n + \alpha_1 N^1 + \cdots + \alpha_{n-1}N^{n-1}} = x_n^{w(\alpha)} \]
        By binomial expansion we know that all other terms contain at least one $y_i$ and hence have strictly smaller degree in $x_n$. 

        \item Thus after substitution, the leading term contributed by each monomial $c_\alpha x_1^{\alpha_i} \cdots x_n^{\alpha_n}$ in $f$ is $c_\alpha x^{w_\alpha}$. From part 2 we know that distinct $\alpha$'s have distinct $w(\alpha)$'s, and thus no two monomomials of $f$ contribut leading terms of the same order. Therefore, there exists some $\alpha^{\max}$ whose weight $w(\alpha^{\max})$ is the largest, and thus the highest power of $x_n$ in $f$ is $c_{\alpha^{\max}} x_n^{w(\alpha^{max})}$. 
        
        \item Since $c_\alpha \in \K$, we can multiply the term in the previous part by $c_{\alpha^{\max}}^{-1}$ to ensure the coefficient is one, thus making the polynomial $f$ monic in $x_n$ with coefficients in $\K[y_1, \ldots, y_{n-1}]$. 
        
        \item ${x_n} \in \K[x_1, \ldots, x_n]$ satisfies the monic polynomial 
        \[c_{\alpha^{\max}}^{-1} \cdot f = x_n^{w(\alpha^{\max})} + g_{d-1}(y_1, \ldots, y_{n-1})x_n^{w(\alpha^{\max})-1} + \ldots + g_{0}(y_1, \ldots, y_{n-1}) = 0\]
        with coefficients in $\K[y_1, \ldots y_{n-1}]$. In $A$ we get, 
        \[\overline{x_n} = \overline{c_{\alpha^{\max}}^{-1} \cdot f} = \overline{x_n^{w(\alpha^{\max})}} + \overline{g_{d-1}x_n^{w(\alpha^{\max})-1} }+ \ldots + \overline{g_{0}} = 0\]
        where each $\overline{g_i}$ is the image of $g_i(y_1, \ldots, y_{n-1})$ in $A$. Hence $\overline{x_n}$ is integral over the subalgebra generated by the image of $y_1, \ldots, y_{n-1}$ in $\K[x_1, \ldots, x_n]/\langle f \rangle$. 
        
        \item Note that there is an isomorphism $\K[y_1, \ldots, y_{n-1}][x_n] \iso \K[x_1, \ldots, x_n]$ given by the maps, 
        \[\phi: x_i \mapsto y_i + x_n^{N^i}, \quad x_n \mapsto x_n\]
        \[\psi: y_i \mapsto x_i - x_n^{N^i}, \quad x_n \mapsto x_n\]

        Now we show that the map $\pi: \K[y_1, \ldots, y_{n-1}] \to A$ is injective. Assume for contradiction $p(y_1, \ldots, y_{n-1}) \in \ker(\pi)$. That is, $p(y_1, \ldots, y_{n-1}) \in \langle f \rangle$ or $p(y_1, \ldots, y_{n-1}) = g \cdot f$  for some $g \in \K[x_1, \ldots, x_n]$. But $x_n-\deg(p) = 0$ and $x_n-\deg(g \cdot f) \geq w(\alpha^{\max})$. But this is a contradiction. Hence $\pi$ must be injective. 

        From the isomorphism, we know that every element of $A$ can be represented by $h \in \K[x_1, \ldots, x_n] = \K[y_1, \ldots, y_{n-1}][x_n]$. Since $f$ is monic we get, 
        \[h = f \cdot q + r, \qquad x_n-\deg(r) \leq w(\alpha^{\max})\]
        Since $f \equiv 0$ in $A$, we get $\overline{h} = \overline{r}$ where $x_n-\deg(r) \leq w(\alpha^{\max})$ with coefficients in $\K[y_1, \ldots, y_{n-1}]$. Thus $A$ is spanned by $1, x_n, \ldots, x_n^{w(\alpha^{\max})-1 }$ and hence is module-finite.
    \end{enumerate}
\end{solution}

\begin{problem}{12} \textbf{Proof of Noether Normalization.} 
    
    Let $A=\K[x_1,\ldots,x_n]/I$ be an algebra-finite $\K$-algebra.
    \begin{enumerate}
        \item If $I=\langle 0\rangle$, prove the theorem.
        \item If $I\neq \langle 0\rangle$, choose $0\neq f\in I$. Use the previous exercise to make a change of variables so that $A$ is module-finite over the image of a polynomial ring in $n-1$ variables.
        \item Let $A_1$ denote this image subalgebra. Explain why $A_1$ is algebra-finite over $\K$ using at most $n-1$ algebra generators.
        \item Apply induction to $A_1$ to find a polynomial subring $\K[y_1,\ldots,y_d]\subseteq A_1$ such that $A_1$ is module-finite over $\K[y_1,\ldots,y_d]$.
        \item Prove that $A$ is module-finite over $\K[y_1,\ldots,y_d]$.
        \item Explain why the elements $y_1,\ldots,y_d$ are algebraically independent over $\K$.
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}
        \item If $I = \langle 0 \rangle$, then the $x_i$ are algebraically independent and $A = \K[x_1, \ldots, x_n]$ is itself a polynomial ring and hence is trivially module-finite over itself. 
        \item Assume $I \neq \langle 0 \rangle$. Now let $f$ be a non-zero polynomial in $I$. By Problem 11 we know that $\K[x_1, \ldots, x_n]/\langle f \rangle$ is module-finite over $\K[y_1, \ldots, y_{n-1}]$ for some algebraically independent $y_1, \ldots y_{n-1}$. 
        
        Since $f \in I$ we have $\langle f \rangle \subseteq I$ and we have the quotient map $\K[x_1, \ldots, x_n] /\langle f \rangle \to A$, and hence $A$ is module-finite over the image of $\K[y_1, \ldots, y_{n-1}]$ in $A$. 

        \item Since $A_1$ is the image of a polynomial ring in $n-1$ variables, it has at most $n-1$ algebra-generators over $\K$.  
        
        \item By part 3, $A_1$ has at most $n-1$ generators, so by the induction hypothesis there exists some $d \leq n-1$ and algebraically independant elements $y_1, \ldots, y_d$ such that $A_1$ is module finite over $\K[y_1, \ldots, y_d]$. 
        
        \item Since $A$ is module-finite over $A_1$ and $A_1$ is module-finite over $\K[y_1, \ldots, y_d]$, $A$ is module-finite over $\K[y_1, \ldots, y_d]$ (Worksheet 7.1, Problem 7). 
        
        \item $\K[y_1, \ldots, y_d]$ is a polynomial ring and thus the $y_i$ are algebraically independent over $\K$. 
    \end{enumerate}
\end{solution}

\begin{problem}{13} \textbf{Finding Normalizations by Hand.} 

    For each algebra below, find a polynomial subring over which the algebra is module-finite. When possible, give explicit module generators.
    \begin{multicols}{2}
    \begin{enumerate}
        \item $\K[t^2,t^3]$.
        \item $\K[x,y]/\langle y^2-x^3\rangle$.
        \item $\K[x,y]/\langle xy\rangle$. Hint: Try $x+y$.
        \item $\K[x,y,z]/\langle z^2-xy\rangle$.
        \item $\K[x^2,xy,y^2]\subseteq \K[x,y]$.
        \item $\K[s^2,s^3,t]$.
    \end{enumerate}
    \end{multicols}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}
        \item $\K[t^2, t^3]$ is module-finite over $\K[t^3 - t^2]$ with generators $\{1, t^2, t^4\}$. 
        \item $\K[x,y]/\langle y^2-x^3\rangle$ is module-finite over $\K[x]$ with generators $\{1, y\}$. 
        \item $\K[x,y]/\langle xy\rangle$ is module-finite over $\K[x+y]$ with generators $\{1, x, y\}$. 
        \item $\K[x,y,z]/\langle z^2-xy\rangle$ is module-finite over $\K[x+y, z]$ with generators $\{1, x\}$. 
        \item $\K[x^2,xy,y^2]\subseteq \K[x,y]$ is module-finite over $\K[x^2, y^2]$ with generators $\{1, xy\}$. 
        \item $\K[s^2,s^3,t]$ is module-finite over $\K[s^3 - s^2, t]$ with generators $\{1, s^2, s^4\}$. 
    \end{enumerate}
\end{solution}

\begin{problem}{14} \textbf{Zariski's Lemma from Noether Normalization.} 
    
    Let $A$ be an algebra-finite $\K$-algebra that is also a field.
    \begin{enumerate}
        \item By Noether Normalization, choose $\K[y_1,\ldots,y_d]\subseteq A$ such that $A$ is module-finite over $\K[y_1,\ldots,y_d]$. Explain why this extension is integral.
        \item Use the earlier fact that if $B$ is a field integral over a domain $A_0$, then $A_0$ is a field, to prove that $\K[y_1,\ldots,y_d]$ is a field.
        \item Prove that $d=0$.
        \item Conclude that $A$ is a finite-dimensional vector space over $\K$. Equivalently, every field that is algebra-finite over $\K$ is algebraic and finite over $\K$.
        \item Deduce the weak Nullstellensatz: if $\K$ is algebraically closed and $\fm\subset \K[x_1,\ldots,x_n]$ is maximal, then
        \[
        \fm=\langle x_1-a_1,\ldots,x_n-a_n\rangle
        \]
        for some $a_1,\ldots,a_n\in \K$.
    \end{enumerate}    
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}
        \item Module-finiteness implies integrality hence $A$ is integral over $\K[y_1, \ldots, y_d]$. 
        \item We have an integral extension $\K[y_1, \ldots, y_d] \to A$, and we know that $\K[y_1, \ldots, y_d]$ is a polynomial ring over a field and hence a UFD. Then by Problem 1.2, since $A$ is a field, $\K[y_1, \ldots, y_d]$ is also a field. 
        
        \item Assume $d \neq 0$. Then $y_1 \in \K[y_1, \ldots, y_d]$ doesn't have an inverse. A contradiction. Hence $d$ must be $0$. 
        
        \item Thus $A$ is module-finite over the field $\K$, i.e. is a finite-dimensional vector space over $\K$, i.e. every element of $A$ is algebraic over $\K$. So $A$ is a algebra-finite over $\K$. 
        
        \item Let $\fm \subset \K[x_1, \ldots, x_n]$ be maximal. Then $\K[x_1, \ldots, x_n]/\fm$ is a field. By Noether Normalization $\K[x_1, \ldots, x_n]/\fm$ is algebra-finite over $\K$ and hence a finite algebraic extension of $\K$. Since $\K$ is algebraically closed it has no finite proper extensions and hence $\K[x_1, \ldots, x_n]/\fm \iso \K$. Then $x_i - a_i \in \fm$ and hence $\langle x_1 - a_1, \ldots, x_n - a_n \rangle = \fm$. 
    \end{enumerate}
\end{solution}

We finish by extracting consequences for dimension. Noether Normalization reduces the dimension theory of affine algebras to the dimension theory of polynomial rings, while Lying Over, Going Up, and Incomparability tell us that finite integral extensions preserve lengths of prime chains.

\begin{theorem}[Dimension of Affine Domains]
Let $A$ be an algebra-finite $\K$-algebra that is a domain. Then
\[
\dim(A)=\trdeg_{\K}\Frac(A).
\]
Equivalently, if $\K[y_1,\ldots,y_d]\subseteq A$ is a Noether normalization, then $\dim(A)=d$.
\end{theorem}

\begin{problem}{15} \textbf{Dimension of Affine Domains.} 
    
    Let $A$ be an algebra-finite $\K$-algebra that is a domain.
    \begin{enumerate}
        \item Let $\K[y_1,\ldots,y_d]\subseteq A$ be a Noether normalization. Use dimension preservation for integral extensions to prove $\dim(A)=d$.
        \item Show that $\Frac(A)$ is algebraic over $\K(y_1,\ldots,y_d)$. Hint: First prove that every element of $A$ is algebraic over $\K(y_1,\ldots,y_d)$.
        \item Deduce that $\trdeg_{\K}\Frac(A)=d$.
        \item Conversely, explain why $d$ is determined by $A$ and not by the choice of Noether normalization.
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni 
    \begin{enumerate}
        \item $A$ is integral over $\K[y_1, \ldots, y_d]$. We know that $\dim(\K[y_1, \ldots, y_d]) = d$ and by laying over and going up $\dim(A) \geq d$. Let 
        \[\fq_0 \subsetneq \fq_1 \subsetneq \cdots \subsetneq \fq_n\]
        be a prime ideal chain in $A$. Then by Incomparability we get the chain  
        \[\fq_0 \cap \K[y_1, \ldots, y_n] \subsetneq \fq_1 \cap \K[y_1, \ldots, y_n]  \subsetneq \cdots \subsetneq \fq_n \cap \K[y_1, \ldots, y_n] \]
        Hence every length in $A$ corresponds to a chain of equal length in $\K[y_1, \ldots, y_d]$ and $\dim(A) = d$. 

        \item Since $A$ is integral over $\K[y_1, \ldots, y_d]$ every element of $A$ satisfies some monic polynomial with coefficients in $\K[y_1, \ldots, y_d]$, i.e. every element in $A$ is algebraic over $\K[y_1, \ldots, y_d]$. Since $\K[y_1, \ldots, y_d] \subseteq \K(y_1, \ldots, y_d)$, every element of $A$ is algebraic over it. 
        
        Now note that elements of $\Frac(A)$ are of the form $\frac{a}{b}$ where $a, b \in A$. Since both $a, b$ are algebraic over $\K(y_1, \ldots, y_d)$ and algebraic extensions are closed under field operations $\frac{a}{b}$ is also algebraic over $\K(y_1, \ldots, y_d)$. 

        \item We know that $\trdeg_{\K}\Frac(A) \geq d$ since $y_1, \ldots, y_d$ are algebraically independent over $\K$. By part 2, since $\Frac(A)$ is algebraic over $\K(y_1, \ldots, y_d)$, $\trdeg_{\K} \Frac(A) \leq d$ since every element except $y_1, \ldots y_d$ are algebraic over $\K(y_1, \ldots, y_d)$. Thus, $\trdeg_{\K}\Frac(A)=d$.
 
        \item Because $\trdeg_{\K}\Frac(A) = d$, which depends only on $A$ and $\K$, not on any choices made in the Noether normalization. 
    \end{enumerate}
\end{solution}

\begin{problem}{16}  \textbf{Computations Using Noether Normalization.} 
    
    Compute the Krull dimension of each ring. In each case, try to exhibit a finite extension from a polynomial subring or reduce to irreducible components.
    \begin{multicols}{2}
    \begin{enumerate}
        \item $\K[t^2,t^3]$.
        \item $\K[s^2,s^3,t]$.
        \item $\K[x,y,z]/\langle z^2-xy\rangle$.
        \item $\K[x,y,z]/\langle xy,z\rangle$.
        \item $\K[x,y,z]/\langle xz,yz\rangle$.
        \item $\K[x_1,\ldots,x_n]/\langle x_1,\ldots,x_r\rangle$.
        \item $\Z[x]/\langle x^2+1\rangle$.
        \item $\K[x^2,xy,y^2]$.
    \end{enumerate}
    \end{multicols}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}
        \item Since $\K[t^2, t^3]$ is module-finite over $\K[t^3 - t^2]$ we have $\dim(\K[t^2, t^3]) = \dim(\K[t^3 - t^2]) = 1$. 
        \item Since $\K[s^2, s^3, t]$ is module-finite over $\K[s^3 - s^2, t]$ we have $\dim(\K[s^2, s^3, t]) = \dim(\K[s^3 - s^2, t]) = 2$. 
        \item Since $\K[x, y, z] / \langle z^2 - xy \rangle$ is module-finite over $\K[x + y, z]$ we have $\dim(\K[x, y, z] / \langle z^2 - xy \rangle) = \dim(\K[x+y, z]) = 2$.
        \item Since $\K[x, y, z] / \langle xy, z \rangle$ is module-finite over $\K[x + y]$ we have $\dim(\K[x, y, z] / \langle xy, z \rangle) = \dim(\K[x+y ]) = 1$.
        \item $\K[x, y, z] / \langle xz, yz \rangle = \K[x, y, z] / \langle z \rangle \cap \langle x, y \rangle$. Then, 
        \[\K[x, y, z] / \langle z \rangle \iso \K[x,y] \implies \dim(\K[x, y, z] / \langle xz, yz \rangle) = 2\] 
        or 
        \[\K[x, y, z] / \langle x, y \rangle \iso \K[z] \implies \dim(\K[x, y, z] / \langle xz, yz \rangle) = 1\] 
        Since the Krull dimension is the $\sup$, $\dim(\K[x, y, z] / \langle xz, yz \rangle)= 2$. 
        \item Since $\K[x_1,\ldots,x_n]/\langle x_1,\ldots,x_r\rangle$ is module-finite over $\K[x_{r+1}, \ldots, x_n]$ we have \\
        $\dim(\K[x_1,\ldots,x_n]/\langle x_1,\ldots,x_r\rangle) = \dim(\K[x_{r+1}, \ldots, x_n]) = n - r$. 
        \item Since $\Z[x] / \langle x^2 +1 \rangle$ is module-finite over $\Z$ we have $\dim(\Z[x] / \langle x^2 +1 \rangle) = \dim(\Z) = 1$.
        \item Since $\K[x^2, xy, y^2]$ is module-finite over $\K[x^2, y^2]$ we have $\dim(\K[x^2, xy,  y^2]) = \dim(\K[x^2, y^2]) = 2$. 
    \end{enumerate}
\end{solution}

\begin{problem}{17}  \textbf{Finite Maps to Affine Space.} 

    Let $A$ be an algebra-finite $\K$-algebra and let $P=\K[y_1,\ldots,y_d]\subseteq A$ be a Noether normalization.
    \begin{enumerate}
        \item Show that the induced map $\Spec(A)\longrightarrow \Spec(P)\cong \mathbb{A}^d_{\K}$ is surjective.
        \item For $\fp\in\Spec(P)$, prove that the fibre ring $\kappa(\fp)\otimes_P A$ is finite-dimensional over $\kappa(\fp)$.
        \item Assume $\K$ is algebraically closed and $\fp=\langle y_1-a_1,\ldots,y_d-a_d\rangle$ is a closed point of $\mathbb A^d_{\K}$. Show that the closed points in the fibre over $\fp$ correspond to maximal ideals of the finite-dimensional $\K$-algebra $A/\fp A$.
        \item Prove that a finite-dimensional algebra over a field has only finitely many maximal ideals. Hint: Distinct maximal ideals are comaximal, and the Chinese Remainder Theorem gives a surjection onto a product of fields.
        \item Explain the geometric slogan: Noether normalization gives a finite-to-one projection from an affine algebraic set to affine space of the same dimension.
    \end{enumerate}    
\end{problem}

\begin{solution}
    \bbni 
    \begin{enumerate}
        \item By Laying over every ideal $\fp \in \Spec(P)$ corresponds to an ideal $\fq \in \Spec(A)$ such that $\fq \cap P = \fp$. Hence the map $\Spec(A) \to \Spec(P)$ is surjective. 
        
        \item We proved this in Problem 8. 
        
        \item Since $\K$ is algebraically closed, $\fp$ is maximal in $P$ and $\kappa(\fp) = P/\fp \iso \K$. So the fibre ring is
        \[\kappa(\fp) \otimes_P A = (P/\fp) \otimes_P A \iso A/\fp A\]
        The closed points of the fiber are the maximal ideals of this fiber ring, which are exactly the maximal ideals of $A$ containing $\fp A$ i.e. the maximal ideals of $A/\fp A$. 

        \item Let $\fm_1, \ldots, \fm_r$ be distinct maximal ideals of $A/\fp A$. Since they are distinct maximal ideals, they are pairwise comaximal and by the CRT, there is a surjection
        \[A/\fp A \twoheadrightarrow \prod_{i=1}^r A/\fm_i\]
        Each $\fm_i$ is a maximal ideal of $A/\fp A$, which is a finite-dimensional $\K$-algebra and hence algebra-finite over $\K$. By the Weak Nullstellensatz, $A/\fm_i \cong \K$ for each $i$. Thus we have a surjection
        \[A/\fp A \twoheadrightarrow \prod_{i=1}^r \K\]
        which gives
        \[\dim_\K(A/\fp A) \geq \dim_\K \prod_{i=1}^r \K = r\]
        Since $A/\fp A$ is finite-dimensional over $\K$ by part (2) and $r \leq \dim_\K(A/\fp A)$, $A/\fp A$ has only finitely many maximal ideals.

        \item Noether normalization gives a surjection $\Spec(A) \twoheadrightarrow A_{\K}^d$ where $d - \dim(A)$. Parts (3) and (4) say that each fiber over a closed point is finite and nonempty. So geometrically, every point of $A_{\K}^d$ has finitely many points of $\Spec(A)$ above it and hence Noether normalization gives a finite-to-one projection from an affine algebraic set to affine space of the same dimension.
    \end{enumerate}
\end{solution}
