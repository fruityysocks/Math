\begin{lemma}[Useful Lemma]
    If $\phi: R \to S$ is a ring homomorphism, then it follows from the definition that 
    \begin{itemize}
        \item $\phi(0_R) = 0_S$
        \item $\phi(-a) = -\phi(a)$
        \item $\phi(a^{-1}) = \phi(a)^{-1}$
    \end{itemize}
\end{lemma}

\begin{theorem}
     If $I \subset R$ is an ideal then, 
    \begin{enumerate}
        \item There exists a ring $R/I$ and a surjective ring homomorphism $\pi: R \to R/I$ such that $\ker(\pi) = I$. Moreover $R/I$ and $\pi$ are unique upto isomorphism. 
        \item There is an inclusion preserving bijection 
        \[\{\text{Ideals in $R/I$}\} \leftrightarrow \{\text{Ideals in $R$ containing $I$}\}\]
        \[\overline{J} \leftrightarrow \pi^{-1}(\overline{J})\]
    \end{enumerate}
\end{theorem}

\begin{proof}
    \bbni
    \begin{enumerate}
        \item Construct $R/I$ explicitly. 
        
        \item $I$ gives an equivalence relation $\sim$ on $R$ such that
        
        \[a \sim b \iff a - b \in I \iff a = b + I\]
        
        Show that equivalence classes $R/\sim$ is a ring. 
        
        \[R \to R /I = R/\sim\]
        \[a \mapsto \overline{a} = a + I\]


     Let $\overline J$ be an ideal in $R/I$, and $J$ be an ideal in $R$ and define the maps 
        \[\phi: \overline{J} \to \pi^{-1}(\overline{J}), \quad \psi: J \to \pi(J)\]

        Now we want to show that $\phi \circ \psi = \psi \circ \phi =id$. Consider $\phi \circ \psi(J)$ and $\psi \circ \phi(\overline J)$. Note that
        \[\psi \circ \phi(\overline J) = \pi (\pi^{-1} (\overline J)) = \overline J\]
        since $\pi$ is surjective. Thus $\psi \circ \phi = id$. 

        Now, note that $J \subseteq \phi \circ \psi (J) = \pi^{-1}(\pi(J))$, and since $\overline 0 \in \pi(J)$, $I \subseteq \pi^{-1}(\pi(J))$. An element $x \in \pi^{-1}(\overline J)$ looks like $x + I$, and $\pi^{-1}(\overline J) = J + I$ 
        Moreover since $\overline{0} \in \overline{J}$, we have $\pi^{-1}(\overline{0}) \subseteq \pi^{-1}(\overline{J})$ which implies $I \subseteq J$.  Thus, $\psi \circ \phi(J) = J$ and $\psi \circ \phi = id$.    
    \end{enumerate}
\end{proof}

\begin{definition}
    The spectrum of a ring $R$ is the set 
    \[\Spec(R) = \{\text{Prime Ideals in } R\}\]
    the maximal spectrum of $R$ is the set 
    \[m\Spec(R) = \{\text{Maximal Ideals in } R\}\]
    
    $\Spec(R)$ is where the `geometry lives'. 
    Note that $m\Spec(R) \subset \Spec(R)$. 

    \textbf{Example:} $R = \Z[\sqrt{-5}]$. $2$ is irreducible but not prime. $R/ \langle 2 \rangle = \Z / 2\Z \times \Z/2\Z$. 
    
    \textbf{Example:} $R = \C[x,y] / \langle x^3 - y^2 \rangle $. $x$ is irreducible, but $\langle x \rangle$ is not prime. $R / \langle x \rangle = \C[y]/ \langle y \rangle$ and $y$ is a zero divisor.   
\end{definition}

\begin{theorem}
    Let $R$ be a non-zero ring. \begin{enumerate}
        \item $R$ has a maximal ideal. 
        \item If $I \subsetneq R$ is a proper ideal there exists a prime ideal $P\subseteq R$ such that $I \subset P \subsetneq R$.  
    \end{enumerate}
\end{theorem}

\begin{proof}
    \textbf{Zorn's Lemma:} Let $(P, \leq)$ be a po-set. 
    
    If $P \neq \emptyset$ and if $C = C_1 \leq C_2 \leq C_3 \leq \cdots$ is a chain in $(P, \leq)$ then there exists $B \in P$ such that $C_i \leq B$ for all $i$, then there exists $M \in P$ such that  if $S \leq M$ then $M = S$. 

    Let $P = \{$Proper ideals in $ R\}$ and $\leq$ be inclusion. We claim that $(P, \leq)$ is a poset. 

    \begin{enumerate}
        \item \begin{itemize}
            \item $P \neq \emptyset$ since $\langle 0 \rangle \in P$. 
            \item Given a chain in $P$ 
        \[I_1 \subseteq I_2 \subseteq \cdots\]
        Then, define $B = \bigcup I_j$. Note that $B$ is an ideals since union of nested ideals is an ideal. And $B$ is proper since $1 \notin I_j$ for all $j$, and hence $1 \notin B$. Then there exists some ideal $M \subset R$ such that  $M$ is maximal. 
        \end{itemize}
        \item Follows from (1). Let $R = R/I$. Pre-image of prime ideals is prime. 
    \end{enumerate}
    Thus, $m\Spec(R) \neq \emptyset$ and $\Spec(R) \neq \emptyset$. 
\end{proof}

\section{Nullstellensatz}

\begin{theorem}[Hilbert's Nullstellensatz I] If $\K$ is algebraically closed, then every maximal ideal of $\K[x_1,\ldots,x_n]$ is of the form $\langle x_1-a_1,\ldots,x_n-a_n\rangle$ for some point $(a_1,\ldots,a_n) \in \K^n$.
\end{theorem}  

\begin{theorem}[Hilbert's Nullstellensatz II] If $\K$ is algebraically closed and $V = \V(\cS) \subset \K^n$ is an algebraic set, for some subset $\cS \subset \K[x_1,\ldots,x_n]$, then $\I(V) = \sqrt{\langle \cS \rangle}$. 
\end{theorem}  

\begin{theorem}[Hilbert's Weak Nullstellensatz] If $\K$ is algebraically closed and $I\subset \K[x_1,\ldots,x_n]$ is an ideal then $\V(I)=\varnothing$ if and only if $I = \langle 1 \rangle$.
\end{theorem}

\begin{theorem}[Hilbert's Nullstellensatz III] Let $\K$ be an algebraically closed field. If $V \subset \K^n$ is an algebraic subset then the maps $\I_V$ and $\V_V$ induce inverse, inclusion-reversing bijections:
 \[
\begin{tikzcd}[row sep = .5 em, column sep = 8em]
\left\{ \begin{matrix} 
\text{algebraic subsets} \\
\text{in $V$}
\end{matrix} \right\} \arrow[r, bend left=10, "\I_V"] & \arrow[l,bend left=10,"\V_V"]
\left\{ \begin{matrix} 
\text{radical ideals} \\
\text{in $\K[V]$}
\end{matrix} \right\} 
\end{tikzcd}
\]
\end{theorem}

\section{Localization}

\begin{definition}
    A subset $U \subset R$ is \textbf{multiplicatively closed} if and only if \begin{enumerate}
        \item $1_R \in U$ 
        \item and if $s, u \in U$ then $su \in U$.
    \end{enumerate}

    \begin{example}
    \bbni
       \begin{enumerate}
        \item  $R = \Z$, $U = \{1, 3, 9, \ldots\}$  is multiplicatively closed. 
        \item $R = \Z$, $U= \{1, 3, 9\}$ is not multiplicatively closed. 
        \item $R = \R[x,y]$, $U = \{f \in R \mid f (1,0) \neq 0\}$ is multiplicatively closed. 
        \item $R = \R[x,y]$, $U = \{f \in R \mid f (1,1) \neq 0, \text{ or } f(0,0) \neq 0\}$ is not multiplicatively closed. 
       \end{enumerate}
    \end{example}
\end{definition}

\begin{definition}
    Let $R$ be a ring and $U \subseteq R$ be a multiplicative closed set. The \textbf{localization} of $R$ at $U$ is a ring $U^{-1}R$ together with a ring map $\eta: R \to U^{-1}R$ that is universal in the following sense: 
    \begin{enumerate}
        \item Given a ring map $\psi: R \to T$ such that $\psi(U)\subseteq T^\times$ there exists a unique ring map $\widetilde{\psi}: U^{-1}R \to T$ such that 
        \begin{center}
            \begin{tikzcd}
                R \arrow[r, "\eta"] \arrow[dr, "\psi", swap] & U^{-1}R  \arrow[dashed, d, "\widetilde{\psi}"]\\
                {} & T
            \end{tikzcd}
        \end{center}
    \end{enumerate}
\end{definition}

\textbf{Construction of $U^{-1}R$:}

Consider $R \times U$ as a set $(r_1, u_1)$ where $r_1 \in R, u_1 \in U \, ``='' \frac{r_1}{u_1}$. Define an equivalence relation 
\[(r_1, u_1) \sim (r_2, u_2) \iff t(u_2r_1 - u_1r_2) = 0, \qquad t \in U\]

Note if $R$ is a domain $t = 1_R$.

Define $\frac{r}{u}$ to be the equivalence class of $(r,u)$ in $R \times U / \sim$. 

Set $U^{-1}R = R \times U / \sim = \left\{ \frac{r}{u} \mid r \in R, \, u \in U\right\} / \sim$

Make $U^{-1}R$ a ring with 
\[\fr{r_1}{u_1}\fr{r_2}{u_2} = \frac{r_1r_2}{u_1u_2}, \qquad \frac{r_1}{u_1} + \frac{r_2}{u_2} = \frac{r_1u_2 + r_2u_1}{u_1u_2}\]

\textbf{Additive Identity:} $0/1$, 
\textbf{Multiplicative Identity:} $1/1$

\[\eta: R \to U^{-1}R\]

Then 
\begin{align*}
    \ker(\eta) &= \{r \in R \mid \eta(r) =0\} \\
    &= \{r \in R \mid \frac{r}{1} = \frac{0}{1}\} \\
    &= \{r \in R \mid t(r \cdot 1 - 1 \cdot 0) =0, \, t \in U\} \\
    &= \{r \in R \mid tr  =0, \, t \in U\} 
\end{align*}

If $R$ is a domain, $\ker(\eta) = \langle 0 \rangle \implies \eta$ is injective, unless $0 \in U$.

\begin{remark}
    If $M$ is a $R$-module we say an element $r \in R$ is a nonzerodivisor on $M$  $\iff$ $M \xrightarrow{\times r} M$ is injective $\iff$ $r \notin \Ann(m), \, \forall m \in M$ $\iff$ $(0:_M r) = \{m \in M \mid rm =0\} = 0$
\end{remark}

\begin{definition}
    Let $M$ be a $R$-module. A sequence of elements $x_1, \ldots, x_t \in R$ is a \textbf{regular sequence} on $M$ if and only if 
    \begin{enumerate}
        \item $x_i$ is a nonzerodivisor on $M /\langle x_1, \ldots, x_{i-1} \rangle M$
        \item $M \neq \langle x_1 \ldots x_t \rangle M$ (non-degenracy condition)
    \end{enumerate}
\end{definition}

\begin{lemma}
    If $(R, \fm)$ is a local ring, $M$ is a finitely generated non-zero $R$-module, and $x_1, \ldots, x_t \in \fm$ then the conditions (1) and (2) above are equivalent to just (1). 
\end{lemma}

\begin{proof}
    Nakayama
\end{proof}


\begin{theorem}[Krull's Intersection Theorem]
    Let $R$ be a Noetherian ring, $I \subseteq R$ an ideal, and $M$ a finitely-generated $R$-module. There exists an element $r \in I$ such that \[(1-r) \bigcap_{k=1}^\infty I^kM = 0\]
\end{theorem}

\begin{corollary}
    If $(R, \fm)$ is a local Noetherian ring, $I \subsetneq R$ then 
    \[\bigcap_{k=1}^\infty I^k = 0\]
\end{corollary}


\begin{theorem}
    Let $(R, \fm)$ be a local ring, $R$ is Noetherian, and $M$ is a non-zero finitely generated $R$-module. If $x_1, \ldots, x_t \in \fm$ then $x_1, \ldots, x_t$ is a regular sequence on $M$ if and only if every permutation is a regular sequence on $M$. 
\end{theorem}

\begin{proof}
    Let $x, y$ be a regular sequence on $M$. We want to show that $y, x$ is a regular sequence on $M$.  

    Assume $ym = 0$ for some $m \in M$. Then we have $y \overline{m} = 0 \in M/xM \implies \overline{m} =0 \implies m = xm_1$ for some $m_1$. 

    That is, 
    \[0 = ym = yxm_1 \implies ym_1 = 0 \in M \implies y \overline{m_1} = 0 \in M/xM\]
    Then by inducting on the $m_i$s, we get $m \in \bigcap_{k=1}^\infty \langle x \rangle^k M$. By KIT, this implies $m = 0$, a contradiction, and thus $y$ is a NZD on $M$. 
    
    Now, assume $x\overline{m} = 0 \in M/yM$. Then, $xm  = ym_1 \in M \implies y \overline{m_1} = 0 \in M/xM \implies \overline{m_1} = 0 \implies m_1 = xm_2 \in M$. So in $M$, 
    \[xm = ym_1 = xym_2 \implies m = ym_2 \implies \overline{m} = 0 \in M/yM\] 
    Thus $x$ is a nonzerodivisor on $M/yM$, and $y, x$ is a regular sequence. 
\end{proof}

        &= \inf  \, \{\height (P) \mid P \subseteq I\}
    \end{align*}
\end{definition}

\begin{lemma}
    Let $R$ be a ring
    \begin{enumerate}
        \item $\dim(R) = \sup \, \{ \height (P) \mid P \in \Spec(R)\}$
        \item If $(R, \fm)$ is local, then $\dim(R) = \height(\fm)$
        \item Let $I \subseteq R$, then $\height (I) = \height (\sqrt{I})$
    \end{enumerate}
\end{lemma}

\begin{theorem}
    Let $R$ be a finitely-generated $k$-algebra. If $R$ is an integral domain then for any ideal $I \subseteq R$, 
    \[\height (I) = \dim(R) - \dim(R/I)\]
\end{theorem}

\begin{theorem}[Krull's Principal Ideal Theorem]
    Let $R$ be a Noetherian ring, $x \in R$. If $P \subseteq R$ is a prime ideal minimal over $\langle x \rangle$ then  $\height(P) \leq 1$. 
\end{theorem}

\begin{corollary}
    Let $R$ be a Noetherian ring $I = \langle r_1, \ldots, r_n \rangle \subseteq R$. If $P \subseteq R$ is a prime ideal minimal over $I$ then $\height(P) \leq n$.  
\end{corollary}

\begin{theorem}[Krull's Height Theorem]
    If $R$ is a Noetherian ring and $P \subseteq R$ is prime and minimal over $\langle f_1, \ldots, f_r \rangle$ (among primes, wrt inclusion). Then $\height (P) \leq r$.      
\end{theorem}

\begin{corollary}
    If $(R, \fm)$ is a Noetherian local ring with residue field $k = R/\fm$, 
    \[\dim(R) = \height(\fm) \leq \mu (\fm) = \dim_k (\fm/\fm^2)\]
    where $\mu(\fm)$ is the minimal number of generators of the ideal $\fm$ or the embedding dimension of $R$. 
\end{corollary}

\begin{definition}
    A local ring $(R, \fm)$ is \textbf{regular} if and only if $\fm$ can be generated by exactly $\dim(R)$ many elements $\iff \mu(\fm) = \dim(R)$
\end{definition}

\begin{definition}
    Let $(R, \fm)$ be a local ring or $\dim(R) = d$. A \textbf{system of parameters (SOP)} for $R$ is a sequence of elements $f_1, \ldots, f_d \in \fm$ such that $\sqrt{\langle f_1, \ldots, f_d \rangle} = \fm$. 
\end{definition}

\begin{lemma}
    Let $(R, \fm)$ be local. A sequence $f_1, \ldots, f_d \in \fm$ is a SOP $\iff R/\langle f_1, \ldots, f_d \rangle $ has dimension zero. 
\end{lemma}

\begin{lemma}
    Let $x$ be a nonzerodivisor on a Noetherian ring $R$. 
    \[\dim R/x = \dim R - 1\]
\end{lemma}

\begin{corollary}
    If $x_1, \ldots, x_n$ is a regular sequence on $R$ then 
    \[\dim \left( R / \langle x_1, \ldots, x_n \rangle\right) = \dim R - n\]
\end{corollary}

\begin{definition}
    Let $R$ be a ring, $I \subseteq R$ an ideal, $M$ a module, The depth of $M$ with respect to $I$ is 
    \[\depth_I(M) = \{\text{maximal length of a regular sequence on $M$ contained in $I$}\}\] 
    If $(R, \fm)$ is local, 
    \[\depth(M) = \depth_{\fm} (M)\] 
\end{definition}

\begin{example}
    Depth is not always finite. For instance, if $R = M = \K[x_1, x_2, \ldots]$ where $x_1, \ldots, x_n$ is a regular sequence for all $n$. 
\end{example}

\begin{proposition}
    Let $M$ be a $R$-module, and $x \in R$, then TFAE, 
    \begin{enumerate}
        \item $x$ is a regular sequence on $M$. 
        \item $k(x, M) = 0 \to M \xrightarrow{\times x} M \to 0$ 
        
        Where $H_1(k(x,M)) = 0$ and $H_0(k(x,M)) \neq 0$. In general, 
        \[H_1(k(x,M)) = \ker(M \xrightarrow{\times x} M) = (0:_Mx) \]
        and, 
        \[H_0(k(x,M)) = M/xM \]
    \end{enumerate}

    If $x, y \in R$. 
    \[k(x, y, M) = 0 \to M \xrightarrow{ \begin{bmatrix}
        y \\ -x
    \end{bmatrix}} M \oplus M \xrightarrow{\begin{bmatrix}
        x & y
    \end{bmatrix}} M \to 0\]

    In general we need, 
    \[H_0(k(x,y,m)) = M/(xf + yg \mid f, g \in M) = M/\langle x,y\rangle M \neq 0\]
    \[H_2(k(x,y,m)) = \ker\left(\begin{bmatrix}
        y \\ -x
    \end{bmatrix}\right) = \{m \in M \mid m(-x) = 0, my = 0\}\]
\end{proposition}

\begin{theorem}[Noether]
    Let $R$ be a Noetherian ring if $I \subsetneq R$ then the set $\{\fp \in \Spec(R) \mid \fp$ is minimal over $I\}$ is finite.
\end{theorem}

\begin{lemma}[Prime Avoidance]
    Let $R$ be a ring and $I, J_1, \ldots, J_n$ be proper ideals and let $J_2, \ldots, J_n$ be prime ideals. Then,
    \begin{enumerate}
        \item If $I \subseteq \cup_{i=1}^n J_i$ then $I \subseteq J_i$ for some $i$. 
        \item If $I \not \subseteq J_i$ then $I \not \subseteq \cup_{i=1}^n J_i$. 
    \end{enumerate}
\end{lemma}

\begin{lemma}
    If $(R, \fm)$ is a local Noetherian regular ring then $R$ is an integral domain. 
\end{lemma}

\begin{proof}
    We will show this by induction on the dimension of $R$. 

    \textbf{Base Case:} $\dim(R) = 0$. Then, 
    \[\dim_R(\fm/\fm^2) = \dim(R) = 0\]
    Then as vector spaces $\fm = \fm^2 = \fm \fm $. Then Nakayama's Lemma implies that $\fm = \langle 0 \rangle$, and hence $R$ is a field and thus a domain.
    
    \textbf{Inductive Step:} Assume $\dim(R) = d$, and the claim is true for all rings of $\dim \leq d-1$. 
\end{proof}

\begin{theorem}
    If $(R, \fm)$ is a regular local Noetherian ring then 
    \[\depth_{\fm}(R) = \dim(R)\]
\end{theorem}


\begin{theorem}[ABS] 
    Let $(R, \fm)$ be a Noetherian local ring with residue field $k = R/\fm$. Then TFAE, 
    \begin{enumerate}
        \item $R$ is a regular ring. 
        \item The global dimension of $R$ is finite. 
        \item Every finitely-generated $R$-module has a projective resolution of finite length. 
        \item $k$ has a projective resolution of length equal to $\dim(R)$. 
    \end{enumerate}
\end{theorem}
