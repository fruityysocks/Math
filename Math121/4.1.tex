    Throughout this course, ``ring'' means \textit{commutative} ring with unity. In this worksheet $\K$ will denote a field. A \emph{$R$-module} is an abelian group $(M, +)$ together with a scalar multiplication $R\times M \to M$, $(r,m) \mapsto rm$, satisfying the following:
    \begin{enumerate}[label=(\roman*),ref=\theenumi\roman*]
        \item Associativity: $(rs)m = r(sm)$ for all $r,s \in R$ and $m \in M$,
        \item Distribution over $R$: $(r+s)m = rm + sm$ for all $r,s\in R$ and $m\in M$, 
        \item Distribution over $M$: $r(m+n) = rm+rn$ for all $r\in R$ and $m,n\in M$, and
        \item Identity: $1_{R}m = m$ for all $m\in M$. 
    \end{enumerate}
    Since $M$ is an abelian group, it has a distinguished additive identity, which we normally denote by $0$ or by $0_{M}$ if we need to distinguish it from $0_{R}$. When the base ring is clear from context, we might simply say that $M$ is a module, or we might say that $M$ is a module over $R$. As we will see, modules are the analogue of vector spaces in which the scalars come from a ring instead of a field. They include ideals, quotient rings, abelian groups, vector spaces, and many other naturally occurring algebraic objects.  

    If $M$ is an $R$-module, a subset $N \subseteq M$ is an \emph{$R$-submodule} of $M$ if and only if $rm+sn \in N$ for all $r,s\in R$ and $m,n\in N$. That is, a submodule is a subgroup of $M$ such that the scalar multiplication of $R$ on $M$ restricts to a scalar multiplication on $N$.

    \begin{problem}{1}
        \textbf{Basic Identities.} Let $R$ be a ring and $M$ an $R$-module. Prove the following identities.
        \begin{enumerate}
        \item $r0_{M} = 0_{M}$ for all $r \in R$.
        \item $0_{R}m = 0_{M}$ for all $m \in M$. 
        \item $(-r)m=r(-m)$ for all $r \in R$ and all $m\in M$. 
        \end{enumerate}
    \end{problem}

    \begin{solution}
        \bbni
        \begin{enumerate}
            \item Let $m \in M$. 
            
            \begin{align*}
                r0_M = r(m-m) = rm - rm = 0_M
                % &= r(0_M + 0_M) = r0_M + r0_M  \\ 
                % r0_M - r0_M &= r0_M + r0_M - r0_M  \\
                % (r-r)0_M &= r0_M + (r-r)0_M  \\
                % 0_M &= r0_M + 0_M  = r0_M 
            \end{align*}

            \item Let $r \in R$. 
            \begin{align*}
                0_Rm = (r-r)m = rm - rm = 0_M
            \end{align*}
            
            \item \begin{align*}
                (-r)m &= (0_R -r) m = 0_Rm - rm 
            \end{align*}
            From (ii) we have that $0_Rm = 0_M$, 
            \begin{align*}
                (-r)m &= 0_Rm - rm = 0_M - rm \\ 
                &= 0_Mr - rm = r(0_M -m) \\ 
                &= r(-m)
            \end{align*}
        \end{enumerate}
    \end{solution}

    \begin{problem}{3}
        \textbf{Submodules Generated by Elements.} 
        
        Let $M$ be an $R$-module. Given elements $m_{1},\ldots,m_{t} \in M$, the submodule of $M$ generated by $m_{1},\ldots,m_{t}$ is the set
        \[
        Rm_{1}+Rm_{2}+\cdots+Rm_{t}\coloneqq \left\{r_1m_1+\cdots+r_tm_t \;\middle|\; r_1,\ldots,r_t \in R \right\} \subseteq M. 
        \]
        We say that $m_{1},\ldots,m_{t}$ are \emph{generators} for $M$ if $Rm_{1}+Rm_{2}+\cdots+Rm_{t}=M$.  A submodule $N \subseteq M$ is a \emph{cyclic submodule} if $N=Rm$ for some $m\in M$. 
            \begin{enumerate}
            \item Prove that $Rm_{1}+Rm_{2}+\cdots+Rm_{t}$ is an $R$-submodule of $M$.
            \item Prove that $Rm_{1}+Rm_{2}+\cdots+Rm_{t}$ is the smallest $R$-submodule of $M$ containing $m_{1},\ldots,m_{t}$. 
            \item Prove that a ring $R$ is itself a cyclic $R$-module by finding a generator. 
            \item In the $\Z$-module $\Z/12\Z$, compute the submodules generated by $\overline{8}$ and $\overline{6}$.
            \item Let $R=\K[x]$ and $M=R/\langle x^3 \rangle$. Describe the submodules generated by $\overline{1}$, $\overline{x}$, and $\overline{x}^2$. 
            \end{enumerate}
    \end{problem}

    \begin{solution}
        \bbni

        Let $N = Rm_1 + Rm_2 + \cdots + Rm_t$. 
        \begin{enumerate}
            \item 
            Let $r, s \in R$ and $n, m \in N$. Then, by definition 
            \[m = r_1m_1+\cdots+r_tm_t\] 
            and \[n = s_1m_1+\cdots+s_tm_t\] 
            with $r_1, \ldots r_t, s_1, \ldots s_t \in R$. Then by distributivity and associativity of $R$, and the abelian nature of $M$ we have, 
            \begin{align*}
                pm + qn &=  p\left(r_1m_1 + \cdots + r_tm_t\right) + q\left(s_1m_1 + \cdots + s_tm_t\right)\\
                 &= \left((pr_1)m_1 + \cdots + (pr_t)m_t\right) + \left((qs_1)m_1 + \cdots + (qs_t)m_t\right) \\ 
                 &= \left((pr_1 + qs_1)m_1 + \cdots + (pr_t+qs_t)m_t\right) 
            \end{align*}
            but since $R$ is closed under addition and multiplication $pr_1+qs_1, \ldots, pr_t+qs_t \in R$. Hence, $pm + qn \in N$ and $N$ is a submodule.
            
            \item Let $L$ be the smallest $R$-submodule of $M$ containing $m_1, \ldots, m_t$. Then by minimality of $L$ we have $L \subseteq N$. To show the reverse containment consider $m \in N$. By definition, 
            \[m = r_1m_1+\cdots+r_tm_t\] 
            with $r_1, \ldots, r_t \in R$. But since $m_1, \ldots, m_t \in L$, all $R$-linear combinations of them is in $L$ (by definition of a submodule), hence $m \in L$. Since $m$ was an arbitrary element in $N$ this implies that $N=L$, hence $N$ is the smallest $R$-submodule of $M$ containing $m_1, \ldots, m_t$. 

            \item The ring $R$ is a cyclic $R$-module with the generator $1_R$. 
            
            By definition, 
            \[R \cdot 1_R = \{r \cdot 1_R \mid r \in R\}\]
            But $r \cdot 1_R = r$ for all $r \in R$, i.e., $R \cdot 1_R = R$. 

            \item Consider $\Z/12\Z \cdot \overline{8}$, 
            \[\Z/12\Z \cdot \overline{8} = \{r \cdot \overline{8} \mid r \in \Z /12\Z\} = \{\overline{0}, \overline{8}, \overline{4}\}\]

            Consider $\Z/12\Z \cdot \overline{6}$, 
            \[\Z/12\Z \cdot \overline{6} = \{r \cdot \overline{6} \mid r \in \Z /12\Z\} = \{\overline{0}, \overline{6}\}\]

            \item $R \cdot \overline{1} = \{r \cdot \overline{1} = \overline{r} \mid r \in R\} =  M$
            
            $R \cdot \overline{x} = \{r \cdot \overline{x} = \overline{rx} \mid r \in R\} =  \{\overline{r_1 x} + \overline{r_2 x^2} \mid r_1, r_2 \in R\} = \K \cdot \overline{x}$

            $R \cdot \overline{x}^2 = \{r \cdot \overline{x}^2 = \overline{rx^2} \mid r \in R\} = \K \cdot \overline{x}^2$
        \end{enumerate}
    \end{solution}

    A \emph{homomorphism} of $R$-modules is a group homomorphism $\phi:M \to N$ such that $\phi(rm)=r\phi(m)$ for all $r\in R$ and $m\in M$. Two $R$-modules $M$ and $N$ are \emph{isomorphic} if and only if there exist $R$-module homomorphisms $\phi:M \to N$ and $\psi:N\to M$ such that $\phi\circ \psi = \Id_{N}$ and $\psi\circ \phi=\Id_{M}$. Equivalently, there is a bijective homomorphism of $R$-modules $\phi:M\to N$. We call such maps \emph{isomorphisms} of $R$-modules.  If $\phi:M \to N$ is a homomorphism of $R$-modules, there are two natural submodules associated to it: its \emph{kernel} and its \emph{image}:
    \[
    \ker(\phi) \coloneqq \left\{ m \in M \;\; \big| \;\; \phi(m)=0_{N} \right\}\subseteq M \quad \quad \text{and} \quad \quad \img(\phi) \coloneqq \left\{ \phi(m) \;\; \big| \;\; m \in M\right\} \subseteq N.
    \]
    One should readily check that both the kernel and image of a module homomorphism are submodules. As with linear maps, group homomorphisms, and ring homomorphisms, the kernel measures the failure of injectivity and the image measures the failure of surjectivity.

    Given a submodule $N \subseteq M$, the quotient abelian group $M/N$ inherits an $R$-module structure via
    \[
    r(m+N) \coloneqq rm + N.
    \]
    Well-definedness uses that $N$ is a submodule, not merely a subgroup. The projection $\pi \colon M \twoheadrightarrow M/N$ is a homomorphism with $\ker(\pi) = N$. The quotient $M/N$ is the module obtained by forcing $N = 0$.  Precisely: if $f \colon M \to P$ is a homomorphism with $N \subseteq \ker(f)$, there is a unique $\overline{f} \colon M/N \to P$ with $f = \overline{f} \circ \pi$:
    \[
    \begin{tikzcd}[row sep = 3.5em, column sep = 3.5em]
    M \arrow[r,"f"] \arrow[d,two heads,"\pi"'] & P \\
    M/N \arrow[ur,dashed,"\overline{f}"'] &
    \end{tikzcd}
    \]
    Phrased differently, there is a natural isomorphism 
    \[
    \Hom_R(M/N,\, P) \cong \left\{ f \in \Hom_R(M,P) \mid N \subseteq \ker(f) \right\};
    \] 
    That is, homomorphisms out of $M/N$ are exactly homomorphisms out of $M$ that kill~$N$. This is usually summarized as the First Isomorphism Theorem. 
    
    \begin{theorem}[First Isomorphism Theorem]
    Every homomorphism $\phi \colon M \to N$ of $R$-modules induces an isomorphism
    \[
        M/\ker(\phi) \xrightarrow{\ \sim\ } \img(\phi), \quad m + \ker(\phi) \longmapsto \phi(m).
    \]
    \end{theorem}
    
    In other words, quotienting by the kernel accounts for all the redundancy: every surjection is a quotient map.
    
    \begin{theorem}[Third Isomorphism Theorem]
    Let $N \subseteq M$ be a submodule and $\pi \colon M \twoheadrightarrow M/N$ the projection.  There is an inclusion-preserving bijection
    \[
        \left\{\, \text{submodules of } M \text{ containing } N \,\right\}
        \;\xleftrightarrow{\;1:1\;}\;
        \left\{\, \text{submodules of } M/N \,\right\}
    \]
    given by $L \mapsto L/N = \pi(L)$ with inverse $\overline{L} \mapsto \pi^{-1}(\overline{L})$.  Moreover, for $N \subseteq L \subseteq M$:
    \[
        M/L \;\cong\; (M/N)\big/(L/N).
    \]
    \end{theorem}
    
    The correspondence says that the submodule lattice of $M/N$ is the interval $[N, M]$ in the submodule lattice of~$M$.  The supplementary isomorphism $(M/N)/(L/N) \cong M/L$ is sometimes called the \emph{Third Isomorphism Theorem}: quotienting in two stages is the same as quotienting all at once.

    \begin{problem}{4}
        \textbf{Basic Examples of Module Homomorphisms.} 
        \begin{enumerate}
            \item If $\K$ is a field, explain why a homomorphism of $\K$-modules is the same as a $\K$-linear map. 
            \item Explain why a homomorphism of $\Z$-modules is the same as a homomorphism of abelian groups. 
        \end{enumerate} 
    \end{problem}

    \begin{solution}
        \bbni
        \begin{enumerate}
            \item If $\K$ is a field, we know that a $\K$-module is a $\K$-vector space. Moreover, by definion, a homomorphism $\varphi: N \to M$ where $N, \, M$ are $\K$-modules ($\K$-vector spaces) has the following property: for any $r, s \in \K$ and $a, b \in N$ we have, 
            \[\varphi(ra + sb) = \varphi(ra) + \varphi(sb) = r\varphi(a) + s \varphi(b)\]
            which is the condition defining a $\K$-linear map. 

            \item We proved in Problem 2 that $\Z$-modules are the same as abelian groups, then any $\Z$-moldule homomorphism must preserve the group operation and hence is the same as an abelian group homomorphism. 
        \end{enumerate}
    \end{solution}

    \begin{problem}{5}
        \textbf{Cyclic Modules \& Annihilators.} Let $M$ be an $R$-module. The \emph{annihilator} of an element $m\in M$ is $\Ann_{R}(m)\coloneqq \{r \in R\;\mid\; rm = 0 \}$. 
        \begin{enumerate}
        \item Prove that $\Ann_{R}(m)$ is an ideal of $R$.
        \item What is $\Ann_{R}(0_{M})$?
        \item Consider the multiplication by $m$ function: 
        \[
        \phi_{m}:R \to M \quad \quad \text{given by} \quad \quad r \mapsto rm.
        \]
        Prove that $\phi_{m}$ is a homomorphism of $R$-modules.
        \item Show that $\ker(\phi_m)=\Ann_{R}(m)$ and $\img(\phi_m)=Rm$. 
        \item Prove that a submodule $N \subseteq M$ is cyclic if and only if $N \cong R/I$ for some ideal $I \subseteq R$. 
        \end{enumerate}
    \end{problem}

    \begin{solution}
        \bbni
        \begin{enumerate}
            \item To show that $\Ann_R(m)$ is an ideal of $R$ we need to check that it is a subgroup under addition, and is closed under multiplication from elements in $R$. 
            \begin{enumerate}
                \item \textbf{Identity:} Note that $0 \in \Ann_R(m)$ since $0\cdot m = 0$. 
                \item \textbf{Closure under Addition and Inverses: }If $a,b \in \Ann_R(m)$, then
                    \[
                    (a-b)m = am - bm = 0 - 0 = 0
                    \]
                    so $a-b \in \Ann_R(m)$. 
                    
                Hence $\Ann_R(m)$ is a subgroup of $(R,+)$.

                \item Let $r \in R$ and $a \in \Ann_R(m)$. Then
                    \[
                    (ra)m = r(am) = r\cdot 0 = 0
                    \]
                    so $ra \in \Ann_R(m)$. 
            \end{enumerate}
            Hence $\Ann_R(m)$ is an ideal of $R$. 

            \item $\Ann_R(0_M) = \{r \in R \mid r0_M = 0\} = R$. That is, $\Ann_R(0_M)$ is the whole ring. 
            
            \item Let $a, b \in R$. Consider \[\phi_m(a+b) = (a+b)m = am + bm = \phi_m(a) + \phi_m(b)\] Hence $\phi_m$ is a group homomorphism. And, for any $r, a \in R$ we have, 
            \[\phi_m(ra) = (ra)m = r(am) = r\phi_m(a)\] 
            Hence, $\phi_m$ is a homomorphism of $R$-modules. 

            \item $\ker(\phi_m) = \{r \in R \mid rm = 0\}$ which is precisely the definition of $\Ann_R(m)$. 
            
            And $\im(\phi_m) = \{rm \mid r \in R\} = Rm$. 

            \item ($\implies$)
            
            Assume submodule $N \subseteq M$ is cyclic. Then $N = R \cdot m$ for some $m \in M$. From part (4) we know that $\ker(\phi_m) = \Ann_R(m)$ and $\im(\phi_m) = Rm$, that is by the first isomorphism theorem, 
            \[N = Rm = \im(\phi_m) \iso R / \ker(\phi_m) = R/\Ann_R(m)\]

            where we know that $\Ann_R(m)$ is an ideal in $R$ (part 1) 
            
            ($\impliedby$) 

            Assume $N \cong R/I$ for some ideal $I \subseteq R$. Then there exists some isomorphism of $R$-modules $\varphi: R/I \to N$. Note that every element in $R/I$ can be written as, $r(1 + I)$  for some $r \in R$ and therefore every element of $N$ is of the form
            \[\varphi(r(1 + I) )= r \varphi(1 + I) \] 
            Let $\varphi(1+I) = n \in N$. That is every element is of the form $rn$ for some $r \in R$, then $N \subseteq Rm$. Moreover since $m \in N$ and $N$ is closed under scalar multiplication we have $Rm \subset N$. Hence, $N$ is cyclic.  
        \end{enumerate}
    \end{solution}

    Let $\{M_i\}_{i\in I}$ be a family of $R$-modules indexed by a set $I$. The \emph{direct product} is
    \[
    \prod_{i\in I}M_i=\left\{(m_i)_{i\in I}\mid m_i\in M_i\right\},
    \]
    with componentwise addition and scalar multiplication. The \emph{direct sum} is the submodule
    \[
    \bigoplus_{i\in I}M_i=\left\{(m_i)_{i\in I}\in \prod_{i\in I}M_i\mid m_i=0 \text{ for all but finitely many } i\right\}.
    \]
    For each $j \in I$ there is a projection map $\pi_j$ and an inclusion map $\iota_j$:
    \[
    \pi_j:\prod_{i\in I}M_i\to M_j,
    \quad
    ((m_i))_{i\in I}\longmapsto m_j,
    \] and \[
    \iota_j:M_j\to \bigoplus_{i\in I}M_i,
    \quad
    m\longmapsto (0,\ldots,0,m,0,\ldots).
    \]
    The product is characterized by maps to the factors, while the direct sum is characterized by maps from the summands. These are the universal properties that make products and sums so useful.

    \begin{problem}{6}
        \textbf{Direct Sums and Products of Modules.} 
        
        Let $\{M_{i}\}_{i\in I}$ be a family of $R$-modules indexed by a set $I$. 
        \begin{enumerate}
        \item Prove that both $\prod_{i\in I} M_{i}$ and $\bigoplus_{i\in I}M_{i}$ are $R$-modules. 
        \item Explain why, if $I$ is finite, $\prod_{i \in I} M_{i} = \bigoplus_{i \in I} M_i$. 
        \item Consider the $\Z$-modules $M_{n}=\Z/n\Z$ for $n\geq 1$. Describe explicitly an element of $\prod_{n=1}^{\infty}M_{n}$ that is not an element of $\bigoplus_{n=1}^{\infty}M_{n}$.
        \item Let $M_i=R$ for all $i\in \N$. Show that the submodule of $\prod_{i\in \N}R$ generated by the standard basis vectors $e_1,e_2,e_3,\ldots$ is $\bigoplus_{i\in \N}R$, not the whole product, provided $R\neq 0$.
        \item For two modules $M$ and $N$, write down the inclusion maps $\iota_M,\iota_N$ and projection maps $\pi_M,\pi_N$ for $M\oplus N$. Verify the following identities
        \[
        \pi_M \circ \iota_M=\Id_M,
        \quad 
        \pi_N \circ \iota_N=\Id_N,\]
        \[
        \pi_M \circ \iota_N=0,
        \quad
        \pi_N \circ \iota_M=0,
        \, 
        \text{and}
        \, 
        \iota_M\circ \pi_M+\iota_N\circ \pi_N=\Id_{M\oplus N}.
        \]
        \end{enumerate}
    \end{problem}

    \begin{solution}
        \bbni
        \begin{enumerate}
            \item Let $f, g, h \in \prod_{i \in I} M_i$ and $r, s \in R$. We can see that $\prod_{i \in I} M_i$ is an abelian group under componentwise addition because each $M_i$ is an abelian group under addition by definition. Now we check the scalar multiplication: 
            \begin{enumerate}
                \item \textbf{Closure:} Since $r \cdot f(i) \in M_i$ for all $i$, $r \cdot f \in \prod_{i \in I} M_i$. 
                \item \textbf{Associativity: } Since $(rs) f(i) = r(s f(i))$ for all $i$, $(rs)f  = r(sf)$. 
                \item \textbf{Distributivity over Module Addition:} Since for all $i$ we have, \[r(f(i) + g(i)) = r(f(i)) + r(g(i))\] we get 
                \[r(f + g) = rf + rg\]
                \item  \textbf{Distributivity over Ring Addition:} Since for all $i$ we have, \[(r+s)f(i) = r(f(i)) + s(f(i))\] we get 
                \[(r+s)f = rf + sf\]
                \item \textbf{Unitality:} Since for all $i$ we have $1_R \cdot f(i) = f(i)$ we get 
                \[1_R \cdot f = f\]
            \end{enumerate}
             
            Since $\bigoplus_{i \in I} M_i \subseteq \prod_{i \in I} M_i$ it suffices to show that $\bigoplus_{i \in I} M_i$ is a submodule. Let $f, g \in \bigoplus_{i \in I} M_i$ and $r \in R$.

            \begin{enumerate}
                \item \textbf{Non-empty:} $0 \in \bigoplus_{i \in I} M_i$
                \item \textbf{Closure under Addition: } Since both $f$ and $g$ have only finitely many non-zero entries, their sum also has finitely many non-zero entries and hence $f + g \in \bigoplus_{i \in I} M_i$. 
                \item \textbf{Closure under Scalar Multiplication:} Since both $f$ has only finitely many non-zero entries, and $ r \cdot 0_M  = 0$ the product $r \cdot f$ also has finitely many non-zero entries and hence $r \cdot f \in \bigoplus_{i \in I} M_i$. 
            \end{enumerate}

            \item We know that $\bigoplus_{i\in I}M_i \subseteq \prod_{i \in I}M_i$, so we only need to show the opposite containment. But if $I$ is finite, every element $f \in \prod_{i \in I} M_i$ only has finitely many non-zero entries and hence by definition $f \in \bigoplus_{i\in I}M_i$. Thus, $\bigoplus_{i\in I}M_i = \prod_{i \in I}M_i$. 
            
            \item Note that $1 \in \Z/nZ$ for all $n \geq 1$. Then the element $(1, 1, 1, \ldots) \in \prod_{n = 1}^\infty M_n$ but it is not in $\bigoplus_{n = 1}^\infty M_n$ because it has infinitely many non-zero entries. 
            
            \item An element of the submodule generated by $\{e_i\}$ is a finite $R$-linear combination, and hence has only finitely many non-zero entries and hence is in $\bigoplus_{i \in \N}R$. Since we are not allowed infinite sums elements like $(1, 1,1, \ldots) = \sum_{i = 1}^\infty e_i$ are not in the generated submodule. 
            
            \item We have, 
            \[\pi_M : M \bigoplus N \to M, \qquad  \pi_M(m, n) = m \]
            \[\pi_N : M \bigoplus N \to N, \qquad  \pi_N(m, n) = n \]

            And, 

            \[\iota_M : M \to M \bigoplus N , \qquad  \iota_M(m) = (m,0) \]
            \[\iota_N : N \to M \bigoplus N , \qquad  \iota_N(n) = (0, n) \]

            Then, 
            \[\pi_M \circ \iota_M (m) = \pi_M(0, m) = m = \id_M(m),\]
            \[\pi_N \circ \iota_N (n) = \pi_N(0, n) = n = \id_N(n),\]
            \[\pi_M \circ \iota_N (n) = \pi_M(0, n) = 0,\]
            \[\pi_N \circ \iota_M (m) = \pi_N(m, 0) = 0,\]
            \[(\iota_M\circ \pi_M+\iota_N\circ \pi_N) ((m, n))= \iota_M (m) + \iota_N(n) = (m, 0) + (0,n) = (m, n) = \Id_{M\oplus N}(m, n).\]
        \end{enumerate}
    \end{solution}

    \begin{problem}{7}
        \textbf{Universal Properties of Products and Sums.} 
        
        Let $\{M_{i}\}_{i\in I}$ be a family of $R$-modules indexed by $I$. 
        \begin{enumerate}
        \item Let $L$ be an $R$-module and suppose that for every $i\in I$ we are given a homomorphism $\phi_i:L\to M_i$. Prove that there exists a unique homomorphism $\phi:L\to \prod_{i\in I}M_i$ such that the following diagram commutes for all $i\in I$:
        \[
        \begin{tikzcd}[row sep = 3.5em, column sep = 4.5em]
        L \arrow[r, dashed, "\phi"] \arrow[dr, swap, "\phi_i"] & \displaystyle \prod_{i \in I} M_{i} \arrow[d, "\pi_i"] \\
        & M_{i}
        \end{tikzcd}.
        \]
        \item Let $L$ be an $R$-module and suppose that for every $i\in I$ we are given a homomorphism $\psi_i:M_i\to L$. Prove that there exists a unique homomorphism $\psi:\bigoplus_{i\in I} M_{i} \to L$ such that the following diagram commutes for all $i\in I$:
        \[
        \begin{tikzcd}[row sep = 3.5em, column sep = 4.5em]
        \displaystyle \bigoplus_{i\in I} M_{i} \arrow[r, dashed, "\psi"] & L\\
        M_{i} \arrow[u, swap, "\iota_{i}"] \arrow[ur, swap, "\psi_{i}"]&
        \end{tikzcd}.
        \]
        \item Deduce natural isomorphisms of $R$-modules
        \[
        \Hom_R\left(L, \; \prod_{i\in I}M_i\right)\cong \prod_{i\in I}\Hom_R(L,\; M_i)
        \] and \[
        \Hom_R\left(\bigoplus_{i\in I}M_i,\; L\right)\cong \prod_{i\in I}\Hom_R(M_i,\; L).
        \]
        \item Explain why a family of maps $\psi_i:M_i\to L$ does not generally define a map $\prod_{i \in I} M_i\to L$. Use the case $M_i=L=\Z$ and $\psi_i=\Id_{\Z}$ to illustrate the problem.
        \item When $I$ is finite, explain why the two universal properties describe the same module. This is why finite direct sums are also finite direct products.
        \end{enumerate}

    \end{problem}

    \begin{solution}
        \bbni
        \begin{enumerate}
            \item \textbf{Existence:} We define $\phi: L \to \prod_{i \in I} M_i$ as follows 
            \[\phi(l) = (\phi_1(l), \ldots, \phi_i(l), \ldots)\]
            This is a well-defined homomorphism because each $\phi_i$ is a homomorphism. 

            \textbf{Uniqueness:} Let $\psi: L \to \prod_{i \in I}M_i$ be another such homomorphism. Then we have $\pi_i \circ \psi(l) = \phi_i(l) = \pi_i \circ \phi(l)$ for all $i \in I$ and $l \in L$. Since $\pi_i$ is well-defined, this means $\psi = \phi$. 

            \item \textbf{Existence:} We define $\psi:  \bigoplus_{i \in I} M_i \to L$ as follows 
            \[\psi(f) = \sum_{i \in I} \psi_i(f(i))\]
            Note that this sum is well-defined since for all $f \in \bigoplus_{i \in I} M_i$ only finitely many $f(i) \neq 0$. This is a homomorphism since, 
            \begin{align*}
                \psi ( f + g ) &= \sum_{i \in I} \psi_i(f(i) + g(i))  \\ 
                &= \sum_{i \in I} \psi_i(f(i)) + \psi_i(g(i)) \\ 
                &= \sum_{i \in I} \psi_i(f(i)) + \sum_{i \in I} \psi_i(g(i))  \\
                &= \psi(f) + \psi(g)
            \end{align*}
            And, 
            \begin{align*}
                 \psi ( r\cdot f ) &= \sum_{i \in I} \psi_i(r \cdot f(i))  \\ 
                &= \sum_{i \in I} r \cdot \psi_i( f(i)) \\ 
                &= r  \sum_{i \in I}  \psi_i( f(i)) \\ 
                &= r \cdot \psi(f) 
            \end{align*}


            \textbf{Uniqueness:} Let $\phi: \bigoplus_{i\in I} M_i \to L$ be another homomorphism such that $\phi \circ \iota_i = \psi_i$ for all $i \in I$. For any $f \in \bigoplus_{i \in I} M_i$, since $f$ has only finitely many non-zero entries we can write $f = \sum_{i \in I} \iota_i(f(i))$, and so
            \[\phi(f) = \sum_{i \in I} \phi(\iota_i(f(i))) = \sum_{i \in I} \psi_i(f(i)) = \psi(f)\]
            
            Since $f$ was arbitrary, $\phi = \psi$.


            \item From (1) and (2) we can see that there exists a bijection between homomorphisms $\phi \leftrightarrow (\phi_i)$ and $\psi \leftrightarrow (\psi(i))$. Then we define homomorphisms: 
            
            \[\Phi: \Hom_R\left(L, \prod_{i \in I} M_i\right) \to \prod_{i \in I}\Hom_R(L, M_i), \quad \Phi(\phi) = (\pi_i \circ \phi)_{i \in I}\]

            \[\Psi: \Hom_R\left(\bigoplus_{i \in I} M_i, L\right) \to \prod_{i \in I}\Hom_R( M_i, L), \quad \Psi(\phi) = (\psi \circ \iota_i)_{i \in I}\]

            Since the universal properties say that every family $(\phi_i)$ and $(\psi_i)$ come from unique $\phi$ and $\psi$ respectively, the homomorphisms $\Phi$ and $\Psi$ are bijective. 


            \item Let $M_i = L = \Z$ and $\psi_i = \id_\Z$. Then $f = (1, 1, 1, \ldots)$ is an element in $\prod_{i \in I} M_i$. Note that the number of non-zero entries in $f$ can be infinite. Then, the map \[\psi(f) = \sum_{i \in I} \id_\Z(f(i)) = \sum_{i \in I} 1\] is not defined because it describes an infinite sum. Hence, a family of maps $\psi_i:M_i\to L$ does not generally define a map $\prod_{i \in I} M_i\to L$. 
            
            \item This is clear from Problem 6 (2). 

        \end{enumerate}
    \end{solution}

    Over a field, every module (i.e.\ vector space) has a basis, and a linear map is determined by its values on a basis. Over a general ring, modules need not have bases at all---but those that do are of fundamental importance. The goal of this section is to set up the theory of free modules and to understand precisely how they differ from vector spaces.
    
    Let $M$ be an $R$-module. A subset $B \subseteq M$ \emph{generates} (or \emph{spans}) $M$ if every element of $M$ can be written as a finite $R$-linear combination of elements of $B$, i.e.\ if
    \[
    M = \left\{\sum_{i=1}^n r_i m_i \;\bigg|\; n\geq 0,\; r_i\in R,\; m_i\in B\right\}.
    \]
    We say $B$ is \emph{linearly independent} if, whenever $m_1,\ldots,m_n\in B$ are distinct and $\sum_{i=1}^n r_i m_i = 0$, we have $r_i=0$ for all $i$. A \emph{basis} for $M$ is a subset that is both generating and linearly independent, or equivalently, a subset $\{b_s\}_{s\in S}$ such that every element of $M$ can be written \emph{uniquely} as a finite $R$-linear combination $\sum_s r_s b_s$.
    
    Over a field, these notions interact exactly as in undergraduate linear algebra: every spanning set contains a basis, every linearly independent set extends to one, and all bases have the same cardinality. Over a general ring, the first two statements can fail; in particular, many modules do not admit bases at all. A module that does admit a basis is called \emph{free}. Given a set $S$, the \emph{free $R$-module on $S$} is
    \[
    R^{(S)}\coloneqq \bigoplus_{s\in S}R.
    \]
    Concretely, $R^{(S)}$ consists of all functions $S\to R$ with finite support. It has a distinguished element $e_s$ for each $s\in S$, where $e_s$ has a $1$ in the $s$-coordinate and $0$ elsewhere. Every element of $R^{(S)}$ can then be written uniquely as a finite sum
    \[
    \sum_{s\in S}r_se_s,
    \]
    so $\{e_s\mid s\in S\}$ is a basis, called the \emph{standard basis}. When $S=\{1,\ldots,n\}$ we write $R^n$ instead of $R^{(S)}$.
    
    The key feature of a free module is that a homomorphism out of it is completely determined by where the basis goes. Precisely: for every function of sets $\alpha\colon S\to M$ (where $M$ is any $R$-module), there is a unique $R$-module homomorphism $\widetilde{\alpha}\colon R^{(S)}\to M$ such that $\widetilde{\alpha}(e_s)=\alpha(s)$ for all $s\in S$. In categorical language, the functor $S\mapsto R^{(S)}$ is left adjoint to the forgetful functor from $R$-modules to sets. This universal property characterizes $R^{(S)}$ up to unique isomorphism and is the reason free modules play a role analogous to that of vector spaces---they are the modules we can map out of ``for free.''

    \begin{problem}{8}
        \textbf{Free Modules and Their Universal Property.}
        \begin{enumerate}
        \item Verify directly that every element of $R^{(S)}$ can be written uniquely as a finite $R$-linear combination of the standard basis elements $e_s$.
        \item Prove the universal property of $R^{(S)}$ stated above.
        \item Deduce that there is a natural isomorphism
        \[
        \Hom_R(R^{(S)},M)\cong \prod_{s\in S}M.
        \]
        Why is the right side a product rather than a direct sum?
        \item Show that an $R$-module $M$ is generated by $m_1,\ldots,m_n$ if and only if there is a surjective homomorphism $R^n\twoheadrightarrow M$ sending $e_i$ to $m_i$.
        \item Conclude that every finitely generated $R$-module is a quotient of a finite free module.
        \item More generally, show that every $R$-module is a quotient of a free $R$-module.
        \item For $M=R/I$, write $M$ as a quotient of a free module and identify the kernel.
        \item For $R=\K[x]$ and $M=R/\langle x^3\rangle$, describe $M$ as a module with one generator and one relation.
        \end{enumerate}
    \end{problem}

    \begin{solution}
        \bbni 
        \begin{enumerate}
            \item Let $f \in R^{(S)}$. Then by definition $f(s) \neq 0$ for only finitely many $s \in S$. So we can write $f = \sum_{s \in S} f(s) \cdot e_s = \sum_{s \in S} r_s \cdot e_s$ for some $r_s \in R$, which is a finite sum and hence every element of $R^{(s)}$ can be written uniquely as a finite $R$-linear combination of the standard basis elements. 
            
            \textbf{Uniqueness:} Suppose $f = \sum_{s \in S} r'_s e_s = \sum_{s \in S} r_s e_s$. Then comparing the $s$-th element gives us $r'_s e_s = r_s e_s \implies r'_s = r_s$. Hence the linear combination is unique. 

            \item \textbf{Existence:} Let $\alpha: S \to M$ be a set map. For each $s \in S$, define $\psi_s: R \to M$ by $\psi_s(r) = r \cdot \alpha(s)$. This is an $R$-module homomorphism since
            \[
            \psi_s(r + r') = (r+r')\alpha(s) = r\alpha(s) + r'\alpha(s) = \psi_s(r) + \psi_s(r')
            \]
            and $\psi_s(rr') = rr'\alpha(s) = r\psi_s(r')$. Since $R^{(S)} = \bigoplus_{s \in S} R$, by the universal property of the direct sum there exists a unique $R$-module homomorphism $\widetilde{\alpha}: R^{(S)} \to M$ such that $\widetilde{\alpha} \circ \iota_s = \psi_s$ for all $s \in S$. In particular, $\widetilde{\alpha}(e_s) = \psi_s(1_R) = \alpha(s)$, so the diagram
            \[
            \begin{tikzcd}[row sep = 3.5em, column sep = 4.5em]
            R^{(S)} \arrow[r, dashed, "\widetilde{\alpha}"] & M\\
            S \arrow[u, "\iota_s"] \arrow[ur, swap, "\alpha"]&
            \end{tikzcd}
            \]
            commutes for all $s \in S$.

            \textbf{Uniqueness:} Any $R$-module homomorphism $\phi: R^{(S)} \to M$ satisfying $\phi(e_s) = \alpha(s)$ for all $s \in S$ must satisfy
            \[
            \phi\!\left(\sum_{s \in S} r_s e_s\right) = \sum_{s \in S} r_s \phi(e_s) = \sum_{s \in S} r_s \alpha(s) = \widetilde{\alpha}\!\left(\sum_{s \in S} r_s e_s\right).
            \]
            Since every element of $R^{(S)}$ is a finite $R$-linear combination of the $e_s$, we conclude $\phi = \widetilde{\alpha}$.


            \item By the universal property, every homomorphism $\phi: R^{(s)} \to M$ is uniquely determined by $\phi(e_s) \in M$ for each $s \in S$. Then the map, 
            \[\Hom_R(R^{(s)}, M) \to \prod_{s \in S} M, \qquad \phi \mapsto (\phi(e_s))_{s\in S}\]
            is a bijection. There is no finiteness restriction on how many $\phi(e_s) \neq 0$, hence the right-hand side must allow non-zero infinite tuples and hence is $\prod_{s\in S}M$. 


            \item ($\implies$) 
            
            Assume $M$ is a $R$-module generated by $m_1, \ldots, m_n$. Then from part (3) we know there exists a homomorphism $\phi: R^n \to M$ that sends $e_i \mapsto m_i$ for all $i \in \{1, \ldots, n\}$. This homomorphism is surjective because all elements of $M$ are finite linear combinations of $m_i$. 
            
            ($\impliedby$)

            Assume there exists a surjective homomorphism $\phi: R^n \to M$. Then for any $m \in M$, surjectivity gives some $x \in R^n$ with $\phi(x) = m$ and writing $x = \sum r_i e_i$ gives $m = \phi(\sum r_i e_i) = \sum r_i \phi(e_i)$. Hence, $M$ is generated by $\{\phi(e_1), \ldots, \phi(e_n)\}$. 

            \item Let $M$ be a finitely-generated $R$-module. Then from part (5) we know there exists some surjective homomorphism $\phi: R^n \twoheadrightarrow M$ for some $n \in N$. Then by the first isomorphism theorem we have an induced isomorphism, 
            \[\overline{\phi}: R^n / \ker(\phi) \iso M\]

            \item Let $M$ be a $R$-module, and define $S = M$. Then there exists a homomorphism $\phi: R^{(M)} \to M$ that sends $e_m \mapsto m$. Again, by the first isomorphism theorem there is an induced homomorphism $\overline{\phi}$, 
            \[\overline{\phi}: R^{(M)} / \ker(\phi) \iso M\]

            \item Let $S = \{1\}$. Then we have $\phi: R^{(S)} \twoheadrightarrow R / I = M$ with $\ker(\phi) = I$. 
            
            \item $M$ is generated by $\overline{1} \in M$ and obeys the relation $x^3 = 0$. 
        \end{enumerate}    
    \end{solution}

    \begin{problem}{9}
        \textbf{Free Modules Are Not Quite Vector Spaces.}
    \begin{enumerate}
        \item Show that $\Z/n\Z$ is not a free $\Z$-module for any $n>1$.
        \item More generally, if $R$ is an integral domain and $0\neq I\subsetneq R$ is an ideal, show that $R/I$ is not a free $R$-module.
        \item In the $\Z$-module $\Z$, show that $\{2,3\}$ is a generating set but not a basis.
        \item In the $\Z$-module $\Z$, show that $\{2\}$ is linearly independent but does not generate $\Z$.
        \item Let $R=\Z/6\Z$. Show that the element $\overline{2}\in R$ is nonzero but not part of a basis for the free $R$-module $R$.
        \item Explain why the previous examples show that bases over rings behave differently from bases over fields.
        \end{enumerate}
    \end{problem}

    \begin{solution}
        \bbni
        \begin{enumerate}
            \item Suppose for contradiction that $\Z/n\Z$ is a $\Z$-module with basis $B$. Then $B$ must be linearly independent, meaning whenever $m_1, \ldots, m_k \in B$, are distinct and $\sum_{i=1}^k r_i m_i = 0$ we have $r_i = 0$ for all $i$. But for any $m \in B \subseteq \mathbb{Z}/n\mathbb{Z}$ we have $n \cdot m = 0$ in $\mathbb{Z}/n\mathbb{Z}$ with $n \neq 0 \in \mathbb{Z}$ contradicting linear independence. Hence $\mathbb{Z}/n\mathbb{Z}$ admits no basis and is not a free $\mathbb{Z}$-module.
            
            \item  Suppose for contradiction that $R/I$ is a free $R$-module with basis $B$. Since $I \neq 0$, pick some $0 \neq a \in I$. Then for any $m \in B \subseteq R/I$ we have $ a \cdot m = 0 \in R/I$, with $a \neq 0$ thus contradicting linear independence. Hence $R/I$ admits no basis and is not a free $R$-module.
            
            \item $\{2, 3\}$ is a generating set since $3-2 = 1$ and $1$ is a generator for $\Z$. But $\{2, 3\}$ is not a basis $3 \cdot 2 + (-2) \cdot 3 = 0$, where $3, -2 \neq 0$. Thus voilating linear independence. Hence, $\{2, 3\}$ is not a basis.
            
            \item $\{2\}$ is linearly independent because if $r \cdot 2 = 0$ then $r = 0$ since $\Z$ is an integral domain. But $\{2\}$ is not a generating set because no finite linear combination of $\{2\}$ gives us $1$. 
            
            \item $\overline{2} \neq 0$ since $2 \not \equiv 0 \pmod 6$, and $\overline{2} \cdot \overline{3} = 0$ and hence $\overline{2}$ fails linear independence and cannot be part of a basis. 
            
            \item Over a field every linearly-independent set extends to a basis and every spanning set contains a basis. The previous examples show that both properties fail over general rings. 
        \end{enumerate}
    \end{solution}

    \begin{problem}{10}
        \textbf{Matrices and Maps Between Finite Free Modules.}
        \begin{enumerate}
        \item Let $A=(a_{ij})$ be an $m\times n$ matrix with entries in $R$. Define a homomorphism
        \[
        \phi_A:R^n\to R^m,
        \quad
        e_j\longmapsto \sum_{i=1}^m a_{ij}e_i.
        \]
        Prove that every homomorphism $R^n\to R^m$ arises uniquely in this way.
        \item Under this correspondence, prove that composition of homomorphisms corresponds to multiplication of matrices.
        \item When $R=\K$ is a field, explain how this recovers the usual matrix description of linear maps between finite-dimensional vector spaces.
        \item Let $A=\begin{pmatrix}2&0\\0&3\end{pmatrix}$ and view $A$ as a homomorphism $\Z^2\to \Z^2$. Describe $\coker(A)=\Z^2/\img(A)$.
        \item Let $R=\K[x]$ and consider multiplication by $x$ as a map $R^2\to R^2$. Write its matrix with respect to the standard basis, and compute its cokernel.
        \end{enumerate}
    \end{problem}


    \begin{solution}
        \bbni
        \begin{enumerate}
            \item \textbf{Existence:} By the universal property of free modules, a homomorphism $\phi: R^n \to R^m$ is determined by where it sends the basis vectors $e_1, \ldots, e_n$. Then, for all $j$, $\phi(e_j) \in R^m$ can be written as 
            \[\phi(e_j) = \sum_{i=1}^{m}a_{ij}e_i\]
            for some $a_{ij} \in R$. Then we define $A = (a_{ij})$ to be a $m \times n$ matrix. Then $\phi = \phi_A$ by construction. 

            \textbf{Uniqueness:} The matrix $A$ is unique since each entry $a_{ij}$ is uniquely determined as a linear combination of the basis elements $e_i$, hence no two matrices give the same homomorphism. 

            Hence every homomorphism $R^n \to R^m$ arises uniquely in this way. 


            \item Let $\phi: R^n \to R^p$, $\psi: R^p \to R^m$ be two homomorphisms with corresponding matricies $A = (a_{kj}), B =(b_{ik})$ respectively. Then for each basis vector $e_j \in R^n$, 
            \[(\psi \circ \phi)(e_j) = \psi \left(\sum_{k=1}^{p}a_{kj}e_k\right) = \sum_{k=1}^{p}a_{kj}\psi(e_k) = \sum_{k=1}^{p}a_{kj}\sum_{i=1}^{m}b_{ik}e_i = \sum_{i=1}^m \left(\sum_{k=1}^p b_{ik}a_{kj}\right)e_i\]
            The coefficient of $e_i$ is $\sum_{k=1}^p b_{ik}a_{kj}$,  which is exactly the $(i,j)$-entry of the matrix product $BA$. Since $\psi \circ \phi$ is determined by its values on the basis vectors, we conclude that $\psi \circ \phi$ corresponds to $BA$.


            \item Let $R = \K$ be a field. Then a homomorphism $\phi: \K^n \to \K^m$ is a $\K$-linear map and the correspondence $\phi \to A$ is the usual representation of a linear map by its matrix with respect to the standard bases. 
            
            \item Note that $A(e_1) = (2, 0)$ and $A(e_2) = (0,3)$. Then we have, 
            \[\img(A) = \{(2a, 3b) \mid a, b \in \Z\} = 2\Z \oplus 3\Z\]
            Hence, $\coker(A) = \Z^2/\img(A) = \Z/2\Z \oplus \Z/3\Z $

             \item Let $A$ be the matrix that corresponds to multiplication by $x$. Then, 
            $A = \begin{pmatrix}
                x & 0 \\ 
                0 & x 
            \end{pmatrix}$, and $\coker(A) = R^2/\img(A) = \K \bigoplus \K$. 
        \end{enumerate}
    \end{solution}

    For two $R$-modules $M$ and $N$, we write $\Hom_R(M,N)$ for the set of $R$-module homomorphisms $M\to N$. Since $R$ is commutative, $\Hom_R(M,N)$ is itself an $R$-module: addition and scalar multiplication are defined pointwise by
    \[
    (\phi+\psi)(m)=\phi(m)+\psi(m),
    \qquad
    (r\phi)(m)=r\phi(m).
    \]
    This construction is one of the most important ways of building new modules from old ones. There is already a great deal to understand: Hom out of a free module is easy, Hom out of a quotient records annihilators, and Hom interacts cleanly with direct sums and products through their universal properties.

    \hrulefill

    \newpage 
    
    \begin{problem}{11}
        \textbf{$\Hom_R(M,N)$ as a Module.}
        \begin{enumerate}
        \item Prove that $\Hom_R(M,N)$ is an abelian group under pointwise addition. What is the zero element?
        \item Prove that the formula $(r\phi)(m)=r\phi(m)$ makes $\Hom_R(M,N)$ into an $R$-module.
        \item Show that evaluation at $1\in R$ defines an isomorphism of $R$-modules
        \[
        \Hom_R(R,M)\cong M.
        \]
        \item More generally, prove that
        \[
        \Hom_R(R/I,M)\cong \left\{m\in M\mid Im=0\right\}.
        \]
        The isomorphism should again be given by evaluation at $\overline{1}\in R/I$.
        \item Deduce that $\Hom_R(R^n,M)\cong M^n$.
        \item Prove that $\End_R(R)\cong R$ as rings, where the ring structure on $\End_R(R)$ is given by addition and composition.
        \item If $I\subseteq R$ is an ideal, identify $\Hom_R(R/I,R)$ in terms of an annihilator ideal of $R$.
        \end{enumerate}
    \end{problem}

    \begin{solution}
        \bbni
        \begin{enumerate}
           \item Let $\phi, \psi \in \Hom_R(M, N)$. Then we define, 
           \[(\phi+\psi)(m) := \phi(m) + \psi(m) \qquad \forall m \in M\]
           Then since $\phi, \psi$ are homomorphisms and addition in $N$ is commutative, for $m, m' \in M$ we have, 
           \begin{align*}
                (\phi+\psi)(m+m') &= \phi(m + m') + \psi(m + m') \\ 
                &= \phi(m) + \phi(m') + \psi(m) + \psi(m') \\ 
                &= \phi(m) + \psi(m) + \phi(m') + \psi (m') \\ 
                &= (\phi + \psi)(m) + (\phi + \psi) (m')
           \end{align*}
           Also note that associativity and commutativity are inherited from $N$, and the additive inverse of $\phi$, $(-\phi)(m) := -\phi(m)$ is well-defined. The zero element is the zero map, $0(m) := 0_N$ for all $m \in M$,
           \[(f+0)(m) = f(m) + 0(m) = f(m) + 0_N = f(m)\]
           
           Hence, $\Hom_R(M,N)$ is an abelian group under pointwise addition. 

           \item We know that $\Hom_R(M,N)$ is an abelian group, and the formula $(r\phi)(m)=r\phi(m)$ shows that $\Hom_R(M, N)$ is closed under scalar multiplication,
           \[(r\phi)(sm) = r\phi(sm) = rs\phi(m) = sr \phi(m) = s(r\phi)(m)\]
           All other module properties follow, showing that it is a $R$-module.

           \item Define a map $\eta: \Hom_R(R, M) \to M$, where $\phi \mapsto \phi(1) =m$ for some $m \in M$. This map is injective since the homomorphism is completely determined by where $1$ goes, so if $\phi(1) =\psi(1) \implies \phi = \psi$. 
           
           To show that this map is a surjection we must show that there exists a homomorphism $\phi_m \in \Hom_R(R, M)$ such that $\phi_m(1) = m$ for all $m \in M$. So, we define 
           \[\phi_m(r) = rm \quad \forall r \in R\]
           and show that $\phi_m$ is a homomorphism: 
           \begin{itemize}
            \item $\phi_m(s \cdot r) = (s \cdot r) m = s (rm) = s\phi_m(r)$
            \item $\phi_m(s + r) = (s + r)m = sm + rm = \phi_m(s) + \phi(r)$
           \end{itemize}
           So for every $m \in M$ there exists a homomorphism $\phi_m$ that maps to it under $\eta$. 

           \item Similarly, we define a map $\eta: \Hom_R(R/I, M) \to \left\{m\in M\mid Im=0\right\}$, where $\phi \mapsto \phi(\overline{1}) = \phi(1 + I)$. This map is well-defined since for any $a \in I$ we have, 
           \[a \phi(1 + I) = \phi(a + aI) = \phi(\overline{0}) = 0 \]
           
           This map is injective since the homomorphism is completely determined by where $\overline{1}$ goes, so 
           \[\eta(\phi) = \eta(\psi) \implies \phi(1 + I) = \psi(1 +I) \implies \phi = \psi\] 
           
           To show that this map is a surjection we define 
           \[\phi_m(r + I) = rm \quad \forall r \in R\]
           and show that $\phi_m$ is a well-defined homomorphism: 
           \begin{itemize}
            \item Let $s + I = r +I$ then, $s - r = 0$ and 
            \[(s-r)m = 0 \implies sm = rm \implies \phi_m(s + I) = \phi_m(r+I) \]
            \item $\phi_m(s (r+I)) = \phi (sr + I) = srm = s\phi(r+I)$
            \item $\phi_m((s + r) + I) = (s + r)m = sm + rm = \phi_m(s) + \phi(r)$
           \end{itemize}
           So for every $m \in M$ such that $mI = 0$ there exists a homomorphism $\phi_m$ that maps to it under $\eta$. 

           \item 
        \end{enumerate}    
    \end{solution}

% Many natural constructions in algebra are \emph{bilinear}: matrix multiplication, the dot product, and the evaluation pairing between a module and its dual. In each case, the map is additive in each variable separately, but not additive as a function of both variables together---so it is not a homomorphism from the product module. The tensor product is the device that resolves this tension. It provides a module $M\otimes_R N$ that converts bilinear maps \emph{out of} $M\times N$ into honest linear maps out of $M\otimes_R N$.
 
% Let $R$ be a commutative ring with identity, and let $M$, $N$, and $P$ be $R$-modules.  An \emph{$R$-bilinear map} from $M\times N$ to $P$ is a function
% \[
% B\colon M\times N\to P
% \]
% satisfying the following three conditions for all $r\in R$, $m,m'\in M$, and $n,n'\in N$:
% \begin{enumerate}[label=(\roman*),ref=\theenumi\roman*]
% \item $B(m+m',\,n)=B(m,n)+B(m',n)$,
% \item $B(m,\,n+n')=B(m,n)+B(m,n')$,
% \item $B(rm,\,n)=rB(m,n)=B(m,\,rn)$.
% \end{enumerate}
% In other words, $B$ is $R$-linear in each argument when the other is held fixed. A \emph{tensor product} of $M$ and $N$ is a pair $(M\otimes_R N,\;\tau)$ consisting of an $R$-module $M\otimes_R N$ and a bilinear map
% \[
% \tau\colon M\times N\to M\otimes_R N,
% \qquad
% (m,n)\longmapsto m\otimes n,
% \]
% that is \emph{universal} in the following sense: for every $R$-module $P$ and every $R$-bilinear map $B\colon M\times N\to P$, there exists a \emph{unique} $R$-module homomorphism $\widetilde{B}\colon M\otimes_R N\to P$ such that $B=\widetilde{B}\circ\tau$.
% \[
% \begin{tikzcd}
% M\times N \arrow[rr,"\tau"] \arrow[dr,"B"'] & & M\otimes_R N \arrow[dl,dashed,"\widetilde{B}"] \\
% & P &
% \end{tikzcd}
% \]
% The content of this property is that $\tau$ is the ``most general'' bilinear map out of $M\times N$: every other one factors through it, and does so in exactly one way. One important consequence is that to define a linear map \emph{out of} a tensor product, it suffices to specify its values on the generators $m\otimes n$ and verify that the assignment is bilinear---the universal property then guarantees a unique, well-defined homomorphism. This strategy is used repeatedly in the exercises below.

% The universal property tells us what a tensor product must \emph{do}, but not that one exists. Here is a direct construction. Begin with the free $R$-module $F$ on the set $M\times N$; its basis consists of formal symbols $[m,n]$, one for each pair $(m,n)\in M\times N$. Let $K\subseteq F$ be the submodule generated by all elements of the forms
% \begin{align*}
% [m+m',\,n]-[m,n]-[m',n],\\
% [m,\,n+n']-[m,n]-[m,n'],\\
% [rm,\,n]-r\,[m,n],\\
% [m,\,rn]-r\,[m,n],
% \end{align*}
% for all $r\in R$, $m,m'\in M$, and $n,n'\in N$.  Set $M\otimes_R N=F/K$, and write $m\otimes n$ for the image of $[m,n]$ in the quotient. By construction, the map $(m,n)\mapsto m\otimes n$ is bilinear, and every element of $M\otimes_R N$ is a finite $R$-linear combination of such ``simple tensors.''
 
% The identities that $m\otimes n$ satisfies are therefore precisely the bilinearity relations:
% \begin{align*}
% (m+m')\otimes n &= m\otimes n+m'\otimes n,\\
% m\otimes(n+n') &= m\otimes n+m\otimes n',\\
% (rm)\otimes n &= r(m\otimes n)=m\otimes(rn).
% \end{align*}
