\begin{problem}{1.4.1}
    Show the following properties for the complex exponential: 
    \begin{enumerate}
        \item $e^{z+2\pi i} = e^z$
        \item $e^{z+w} = e^ze^w$
        \item $e^{z-w} = \frac{e^z}{e^w}$
        \item $e^{-w} = \frac{1}{e^w}$
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}
        \item Let $z = x + iy$. Then, 
        \begin{align*}
            e^{z+2\pi i} &= e^{x + iy + 2 \pi i} = e^{x + i(y + 2\pi)} \\ 
            &= e^x \cos(y+ 2\pi) + ie^x \sin (y + 2\pi) \\
            &=e^x (\cos(y)\cos(2\pi) - \sin(y)\sin(2\pi)) + ie^x (\sin(y)\cos(2\pi) + \cos(y)\sin(2\pi)) \\ 
            &=e^x (\cos(y) \cdot 1 - \sin(y) \cdot 0) + ie^x (\sin(y)\cdot 1 + \cos(y) \cdot 0) \\
            &= e^x\cos(y) + ie^x\sin(y) \\ 
            &= e^{x+iy} \\ 
            &= e^z
        \end{align*}
        Hence $e^{z + 2\pi i} = e^z$ holds. 

        \item Let $z = x + iy$ and $w = x' + iy'$. Then, 
        \begin{align*}
            e^{z+w} &= e^{x + iy + x' + iy'} = e^{(x+x') + i(y+y')} \\ 
            &= e^{x+x'} \cos(y+ y') + ie^{x+x'} \sin (y + y') \\
            &=e^{x+x'} (\cos(y)\cos(y') - \sin(y)\sin(y')) + ie^{x+x'} (\sin(y)\cos(y') + \cos(y)\sin(y')) \\ 
        \intertext{From Real Analysis we know that $e^{x+x'} = e^x e^{x'}$. So we get,}
         e^{z+w} &=e^{x}e^{x'} \cos(y)\cos(y') - e^{x}e^{x'}\sin(y)\sin(y') + i e^{x}e^{x'}\sin(y)\cos(y') + ie^{x}e^{x'} \cos(y)\sin(y') \\ 
         &= e^x  \cos(y) \left( e^{x'} \cos(y') + i e^{x'}\sin(y')\right) + e^x\sin(y) \left(i e^{x'} \cos(y') - e^{x'}\sin(y')\right) \\ 
         &= e^x  \cos(y) \left( e^{x'} \left(\cos(y') + i \sin(y')\right)\right) + e^x\sin(y) \left( e^{x'} \left(i\cos(y') - \sin(y')\right)\right) \\ 
         &= e^x\cos(y) \cdot e^w + e^x\sin(y) \left( ie^{x'} \left(\cos(y') + i\sin(y')\right)\right) \\ 
         &= e^x\cos(y) \cdot e^w + e^x\sin(y) \cdot i e^{w} \\
         &= e^w \left(e^x\cos(y) + ie^x\sin(y)\right) \\ 
         &= e^w \cdot e^z 
        \end{align*}
        Hence $e^{z + w} = e^z \cdot e^w$ holds. 

        \item Let $z = x + iy$ and $w = x' + iy'$. Then, 
        \begin{align*}
            e^{z-w} &= e^{x + iy - x' - iy'} = e^{(x-x') + i(y-y')} \\ 
            &= e^{x-x'} \cos(y-y') + ie^{x-x'} \sin (y - y') \\
            &=e^{x-x'} (\cos(y)\cos(y') + \sin(y)\sin(y')) + ie^{x-x'} (\sin(y)\cos(y') - \cos(y)\sin(y')) \\ 
        \intertext{From Real Analysis we know that $e^{x-x'} = \frac{e^x }{e^{x'}}$. So we get,}
         e^{z-w} &=e^{x}e^{-x'} \cos(y)\cos(y') + e^{x}e^{-x'}\sin(y)\sin(y') + i e^{x}e^{-x'}\sin(y)\cos(y') - ie^{x}e^{-x'} \cos(y)\sin(y') \\ 
         &= e^x  \cos(y) \left( e^{-x'} \cos(y') - i e^{-x'}\sin(y')\right) + e^x\sin(y) \left(i e^{-x'} \cos(y') + e^{-x'}\sin(y')\right) \\ 
         &= e^x  \cos(y) \left( e^{-x'} \left(\cos(y') - i \sin(y')\right)\right) + e^x\sin(y) \left( e^{-x'} \left(i\cos(y') + \sin(y')\right)\right) \\ 
         &=e^x  \cos(y) e^{-x-iy} + e^x\sin(y) \left( ie^{-x'} \left(\cos(y') - i \sin(y')\right)\right) \\ 
         &= e^{-x-iy}(e^x\cos(y) + e^x\sin(y)) = e^{-w} \cdot e^z \\
         &= \frac{e^z}{e^w}
        \end{align*}
        Hence $e^{z - w} = \frac{e^z}{e^w}$ holds. 

        \item This follows from (3) if we let $z = 0$. 
        \[e^{-w} = e^{0 - w} = \frac{e^0}{e^z} = \frac{1}{e^z}\]
    \end{enumerate}
\end{solution}

\begin{problem}{1.4.9}
    Observe that the $n$th roots of $1$ are given by $\{1, \omega, \cdots, \omega^{n-1}\}$, where $\omega = e^{2\pi i/n}$. Assuming $n > 1$, show that $1+\omega+\cdots+\omega^{n-1} = 0$. Prove a similar result for the $n$th-roots of $-1$. 
\end{problem}

\begin{solution}
    \bbni
    Let $p = 1+\omega+\cdots+\omega^{n-1}$. We can see that $p$ is a geometric series. 
    Then, 
    \begin{align*}
        p &= 1+\omega+\cdots+\omega^{n-1} \\ 
        p &= \frac{1 - \omega^n}{1 - \omega} \\ 
        \intertext{Note that $\omega ^ n = 1$. So we get,} 
        p &= \frac{1 - 1}{ 1 - \omega} = 0
    \end{align*}
    Hence, $p = 0$. 

    Note that $n-th$ roots of $-1$ is the set $\left \{\zeta_{2n}^{(2k+1)}; 0 \leq k \leq n-1 \right \}$ where $\zeta_{2n}$ is the $2n$-th root of unity. That is, 
    \[\zeta_{2n}^{2k+1} = (e^{2\pi i / 2n})^{2k+1} = (e^{\pi i / n})^{2k+1} = e^{(2k+1)\pi i/n}\] 
    Let $\sigma = e^{\pi i /n}$ and $p' = \sigma + \sigma^3 + \sigma ^5 +  \cdots + \sigma^{2n-1}$. Again this is a geometric series. Then, 
    \begin{align*}
        p' &= \sigma + \sigma^3 + \sigma ^5 +  \cdots + \sigma^{2n-1} \\ 
         &= \sigma \left ( 1 + \sigma^2 + \sigma ^4 +  \cdots + \sigma^{2(n-1)} \right ) \\ 
         &= \sigma \left ( 1 + e^{2\pi i /n} + (e^{2\pi i /n}) ^2 +  \cdots + (e^{2\pi i /n})^{(n-1)} \right ) \\ 
         &= \sigma \cdot 0 = 0   
    \end{align*}
\end{solution}

\begin{problem}{2.2.3}
    Suppose that $D$ is a domain and $f \in H(D)$. Show that if $f$ is real-valued in $D$, then $f$ is constant. 
\end{problem}

\begin{solution}
        \bbni
        $D$ is a connected open subset of $\C$ and $f$ is complex differentiable on $D$. We know that 
        \[f(x + iy) = u(x,y) + iv(x,y)\]
        Since $f$ is real valued in $D$, $v(x,y) = 0$ everywhere on $D$, that is 
        \[v_x = v_y = 0\]
        Then since the Cauchy-Riemann equations hold we have, 
        \[u_x = v_y = 0, \quad u_y = -v_x = 0\]
        Hence, $f'(z) = f_x(z) = u_x + i v_x = 0$, and since $z$ was an arbitrary element in $D$ this holds for all $z$. By a result shown in class, since $D$ is a domain and $f'(z) = 0$ for all $z \in D$, $f$ is constant on $D$. 
\end{solution}
