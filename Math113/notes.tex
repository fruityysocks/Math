\begin{definition}
    If $X$ is a topological space, then a curve in $X$ is a continuous function $\gamma: [a,b] \subset \R \to X$. We call $[a,b]$ the parameter space and write $\gamma^\star$ for the image $\gamma([a,b])\subset X$. If $\gamma(a) = \gamma(b)$ we call $\gamma$ closed, and we also have $\gamma(s) \neq \gamma(t)$ when $a \leq s < t < b$, then we say $\gamma$ is a simple closed curve. 

    A simple closed curve is called a \textbf{Jordan Curve}. 

    If $X \subset \C$ we write, $\gamma(t) = x(t) + iy(t)$. 
    We let $\gamma'(t) = \lim_{h\to 0} \frac{1}{h} (\gamma(t+h) - \gamma(t))$ if it exists. This happens iff $x'(t)$ and $y'(t)$ exist, and then 
    \[\gamma'(t) = x'(t) + iy'(t)\]
\end{definition}

\begin{definition}
    $\gamma$ is smooth if $\gamma'(t)$ exists and is non-zero for all $t \in [a,b]$. 
\end{definition}

\textbf{Examples:} 

\begin{enumerate}
    \item Consider $\gamma_1(t) = t + it^2$ if $t \in [-1,1]$ (smooth) and $\gamma_2(t) = t^3 + it^6$ for $t \in [-1,1]$ (not smooth $\gamma_2(0,0) = 0$). 
    \item  $\gamma(t) = t^3 + it^2$, $t \in [-1, 1]$. That is $y = x^{2/3}$ (not smooth). 
\end{enumerate}

\begin{theorem}{Jordan Curve Theorem.}
    If $\gamma$ is a Jordan Curve, then $\C \setminus \gamma^\star$ consists of two domains each of which has $\gamma ^\star$ as its boundary. 
\end{theorem}

\begin{remark}
    $\gamma^\star$ is compact, it is contained in some $D_R(0)$ and hence exactly one of the components of $\C \setminus \gamma^\star$ of $\C \gamma ^\star$ is bounded and we call that the interior of $\gamma$. 
\end{remark}

\begin{definition}
    A curve $\gamma: [a, b] \to \C$ is called a path or a contour if there is a partition 
    \[P = \{a = t_0 < t_1 < \cdots < t_n =b\}\]
    such that $\gamma'(t)$ is continuous on $[t_{k-1}, t_k]$ for $1 \leq k \leq n$. If $D \subset \C$ then we say $\gamma$ is a path in $D$ if $\gamma^\star \subset D$. 
\end{definition}

\begin{definition}
    Let $\gamma : [a, b] \to \C$ be a curve. Say $\gamma(t) = x(t) + iy(t)$.
    \[\int_a^b \gamma(t) dt = \int_a^b x(t) dt + i \int_a^b y(t) dt\]
    Then if $\gamma'$ is continuous then 
    \[\int_a^b \gamma'(t) dt = x(b) - x(a) + i(y(b) - y(a))\]
\end{definition}

\begin{proposition}
    Suppose $\gamma$ and $\sigma$ are curves defined on $[a,b]$. Then, 
    \begin{enumerate}
        \item $\int_a^b (\gamma(t) + \sigma(t)) dt = \int_a^b \gamma(t) dt + \int_a^v \sigma(t) dt$
        \item If $\alpha \in \C$ then $\int_a^b \alpha \gamma(t) dt = \alpha \int_a^b \gamma(t) dt$. 
        \item $\left| \int_a^b \gamma(t) dt \right| \leq \int_a^b |\gamma(t)| dt$. 
    \end{enumerate}
\end{proposition}

\begin{proof}

    \begin{enumerate}
        \item Straightforward. 
        \item Is routine if $\alpha \in \R$. If $\alpha = a + ib$ we can multiply it out and use the definition. 
        \item Let $\tau$ be a unimodular constant such that 
        \[\left| \int_a^b \gamma(t) dt \right| = \tau \int_a^b \gamma(t)dt\]
        Now, 
        \begin{align*}
            \left| \int_a^b \gamma(t) dt \right| &=  \int_a^b \tau \gamma(t)dt \\ 
            &= \text{Re}\left( \int_a^b \tau \gamma(t)dt \right) \\ 
            &= \int_a^b \text{Re}\left( \tau \gamma(t)\right)dt \\ 
            &\leq \int_a^b |\tau \gamma(t)|dt \\ 
            &= \int_a^b | \gamma(t)|dt \\ 
        \end{align*}
    \end{enumerate}
\end{proof}

\begin{remark}

\begin{theorem}{Homer's Cauchy Theorem.}
    Let $\gamma(t) =a + re^{it}$ with $r > 0$ and $t \in [0,2\pi]$. Then if $n \in \Z$, we have that $\int_\gamma(z-a)^n dz = \begin{cases}
        0, & n\neq -1 \\
        2\pi i, & n = -1
    \end{cases}$

    If $\sigma(t) = e^{\alpha}$ with $\alpha \in \C$ then $\sigma'(t) =\alpha e^{\alpha t}$. If $\neq -1$, then, 
    \begin{align*}
        \int_\gamma(z-a)^n dz &= \int_0^{2\pi}(re^{it})^n rie ^{it} dt \\
        &= r^{n+1} i \int_0^{2\pi}e^{i(n+1)t} dt\\ 
        &= ir^{n+1} \frac{e^{i(n+1)t}}{i(n+1)} \big \vert_0^{2\pi} = 0
    \end{align*}
\end{theorem}

\begin{lemma}[{3.20}]
    Suppose $\gamma: [a,b] \to \C$ is a path and that $\phi: [c,d] \to [a,b]$ is differentiable with $\phi(c) = a$ and $\phi(d) = b$. Let $\sigma(t) = \gamma(\phi(t))$. Then $\sigma$ is path defined on $[c,d]$ which is equivalent to $\gamma$. 
\end{lemma}
\begin{proof}
    Check $\sigma'(t) = \gamma'(\phi(t))\phi'(t)$. 

    If $f \in C(\gamma_1^\star)$, then 

    \begin{align*}
        \int_\sigma f(z) dz &= \int_c^d f(\sigma(t)) \sigma'(t) dt \\ 
        &= \int_c^d f(\gamma(\phi(t))) \gamma'(\phi(t))\phi(t) dt \\ 
        \intertext{Let $u = \phi(t)$, $du = \phi'(t)$}
        \int_\sigma f(z) dz &= \int_{\phi(c)}^{\phi(d)} f(\gamma(u))\gamma'(u) du \\
        &= \int_a^b f(\gamma(u)) \gamma'(u) du \\
        &= \int_\gamma f(z) dz
    \end{align*}
\end{proof}

\begin{lemma}[3.7]
    Suppose $\gamma:[a,b] \to D$ is a path in a domain $D$ and $f \in H(D)$. Suppose that $f'$ is continuous and $\gamma'$ is continuous then $\sigma(t) = f(\gamma (t))$ is a path on $[a,b]$ such that 
    \[\sigma'(t) = f'(\gamma(t))\gamma'(t)\]
\end{lemma}

\begin{proof}
    Because $f'$ is continuous, we just need to show that the following equation holds, 
    \[\sigma'(t) = f'(\gamma(t))\gamma'(t).\]
    Let $t \in (a,b)$. Then, 
    \begin{align*}
        \sigma'(t) &= \lim_{h \to 0} \frac{1}{h} \left( \sigma(t + h) - \sigma(t)\right) \\ 
        &= \lim_{h \to 0} \frac{1}{h} \left( f( \gamma(t + h)) - f(\gamma(t))\right) \\ 
        &= \lim_{h \to 0} G(h) \frac{\gamma(t+h) - \gamma(t)}{h}
    \intertext{when,}
    G(h) &= \begin{cases}
        \frac{f(\gamma(t+h)) - f(\gamma(t))}{\gamma(t+h)-\gamma(t)}, & \gamma(t+h) \neq \gamma(t) \\ 
        f'(\gamma(t)), & \gamma(t+h) = \gamma(t) \\ 
    \end{cases}
    \end{align*}
    It is not hard to see that $G$ is continuous ar $0$. Now we are done. 
\end{proof}

\begin{theorem}[Fundamental Theorem for Path Integrals]
    Suppose $D$ is a domain and that $f: D \subset \C \to \C$ is continuous. Suppose that $F \in H(D)$ and $F'(z) = f(z), \forall z \in D$. Then if $\gamma: [a,b] \to D$ is any path in $D$, 
    \[\int_\gamma f(z) dz = F(\gamma(b)) - F(\gamma(a))\]
\end{theorem}

\begin{proof}
    We have
    \begin{align*}
        LHS &= \int_a^b f(\gamma(t))\gamma(t) dt \\ 
        &= \sum_{k=1}^n \int_{t_{k-1}}^{t_k} f(\gamma(t))\gamma'(t) dt \\ 
    \end{align*}
    where $\gamma'$ is continuous on $[t_{k-1}, t_k]$. But, 
    \[\frac{d}{dt} \left(t \to F(\gamma(t))\right) = f(\gamma(t)) \gamma'(t) = \sum_{k=1}^nF(\gamma(t_k)) - F(\gamma(t_{k-1}))\]
\end{proof}

\begin{definition}
    Let $\gamma: [a,b] \to \C$ is a path, then the length of $\gamma$ is 
    \[\ell(\gamma) = \int_a^b |\gamma'(t)| dt\]

    See Remark 3.2.7. 
\end{definition}

\begin{proposition}
    Suppose $\gamma:[a,b] \to \C$ is a path and $F \in C(\gamma^\star)$. 
    \begin{enumerate}
        \item [(a)] If $|f(z)| \leq M$ for all $z \in \gamma^\star$, then 
        \[\left| \int_\gamma f(z) dz\right| \leq M \ell(\gamma)\] 

        \item[(b)] If $(f_n) \subset C(\gamma^\star)$ and if $f_n \to f$ uniformly on $\gamma^\star$, then 
        \[\lim_{n\to \infty} \int_\gamma f_n(z) dz = \int_\gamma f(z) dz\]
    \end{enumerate}
\end{proposition}

\begin{proof}
    \begin{enumerate}
        \item [(a)] \begin{align*}
            \left| \int_\gamma f(z) dz \right| &= \left| \int_a^b f(\gamma(t)) \gamma'(t) dt\right| \\ 
            & \leq \int_a^b \left| f(\gamma(t))\right| |\gamma(t)| dt \\ 
            &\leq M \int_a^b |\gamma'(t)|dt \\ 
            &= M \ell(\gamma)
        \end{align*}

        \item[(b)] Let $\epsilon > 0$. Then $\exists N$ such that for all $n \geq N$ 
        \[|f_n(z)  - f(z)| < \frac{\epsilon}{\ell(\gamma) +1}, \quad \forall z \in \gamma^\star\]
        Now using part (a) 
        \begin{align*}
            \left|\int_\gamma f(z) dz - \int_\gamma f_n(z) dt\right| &= 
             \left|\int_\gamma (f(z) dz - f_n(z) )dz\right| \leq \frac{\epsilon}{\ell(\gamma ) + 1} \ell(\gamma) < \epsilon
        \end{align*}
    \end{enumerate}
\end{proof}

\begin{theorem}[3.31]
    Supppose that $\gamma$ is a path and that $f \in C(\gamma^\star)$. Let $D = \C \setminus \gamma^\star$. Define, 
    \[F(z) = \int_\gamma \frac{f(w)}{w-z} dz, \qquad \forall z \in D\]
    Then $F$ is analytic on $D$. For $n \geq 0$, 
    \[F^{(n)}(z) = n! \int_\gamma \frac{f(w)}{(w-z)^{n+1}} dw\]
    Moreover, the radius of convergence of the Taylor series in $F$ about $a \in D$ is at least $\sup \{R : D_R(a \subset D)\}$. 
\end{theorem}

\begin{proof}
    Fix $a \in D$. Since $D$ is open, $\exists R > 0$ such that $D_R(a) \subset D$. If $z \in D_R(a)$ and $w \in \gamma^\star$ then, 
    \[\left|\frac{z-a}{w-a}\right| \leq \frac{|z -a|}{R}  < 1\]

    Hence, 
    \begin{align*}
        \frac{1}{w-z} &= \frac{1}{w-a-(z-a)} = \frac{1}{w-a} \frac{1}{1 - \frac{z-a}{w-a}} \\ 
        &= \frac{1}{w-a} \sum_{n=0}^\infty \left( \frac{z-a}{w-a}\right)^n \\ 
        &= \sum_{n=0}^\infty \left( \frac{(z-a)^n}{(w-a)^{n+1}}\right)  
    \end{align*}
    Note that if $w = \gamma^\star$, then $| w- a| \geq R$ and $\exists M$ such that $|f(w)| \leq M$, for all $w \in \gamma^\star$. Hence, 
    \[\left| f(w)\frac{(z-a)^n}{(w-a)^{n+1}}\right| \leq \frac{M}{R} \left(\frac{|z-a|}{R}\right)^n\]
    This for fixed $z \in D_R(a)$, 
    \[\frac{f(w)}{w-z} = \sum_{n=0}^\infty f(w) \frac{(z-a)^n}{(w-a)^{n+1}}\]
    uniformly in $w$ by Weierstrass $m$-test. 
    
    Thus $\forall z \in D_R(a)$, we have 
    \begin{align*}
        F(z) &= \int_\gamma\frac{f(w)}{w-z} dw \\ 
        &= \int_\gamma \left( \sum_{n=0}^\infty f(w) \frac{(z-a)^n}{(w-a)^{n-1}}\right) dw \\
        &= \sum_{n=0}^\infty \left( \int_\gamma\frac{f(w)}{(w-a)^{n-1}} dw\right) (z-a)^n 
    \end{align*}
    Thus $F$ is analytic and 
    \[\frac{F^{(n)}(a)}{n!} = \int_\gamma \frac{f(w)}{(w-a)^{n+1}} dw\]
\end{proof}
