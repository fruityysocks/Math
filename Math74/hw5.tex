\begin{problem} {1}
    Compute the simplicial homology for $\RP^2$ using a $\Delta$-complex structure on it. 
\end{problem}

\begin{solution}
    \bbni

    $\RP^2$ as a $\Delta$-complex is formed by two $0$-simplices, three $1$-simplices and two $2$-simplices. 

     \[
    \begin{tikzcd}
    v_0   & v_1  \arrow[d,"a"] \arrow[l,"b", swap]\\
    v_1 \arrow[u,"a"] \arrow[r,swap,"b"] \arrow[ur,"c"{yshift = -.05in}] \arrow[ur, phantom, "\scriptstyle U"{yshift=0.1in, xshift=-.2in}] & v_0  \arrow[ul, phantom, "\scriptstyle L", very near start]
    \end{tikzcd}
    \]
    

    We have, 
    \begin{itemize}
        \item $\Delta_0(\RP^2) = \Z \langle v_0 \rangle \oplus \Z \langle v_1 \rangle$ 
        \item $\Delta_1(\RP^2) = \Z \langle a \rangle \oplus \Z \langle b \rangle \oplus \Z \langle c \rangle $ 
        \item $\Delta_2(\RP^2) = \Z \langle U \rangle \oplus \Z \langle L \rangle$ 

    \end{itemize}

    \[0 \xrightarrow{\d_3} \Delta_2(\RP^2) \xrightarrow{\d_2} \Delta_1(\RP^2) \xrightarrow{\d_1} \Delta_0 (\RP^2) \xrightarrow{\d_0} 0\]

    The homologies are, 
     \begin{itemize}
        \item $H_0^\Delta(\RP^2) = \frac{\ker(\d_0)}{\im(\d_1)} =  \frac{\Z\langle v_0 \rangle \oplus \Z\langle v_1 \rangle}{\langle v_0 - v_1\rangle} \iso \Z\langle v_0 \rangle \iso \Z$
        
        \item $H_1^\Delta(\RP^2) = \frac{\ker(\d_1)}{\im(\d_2)} =  \frac{\Z\langle a -b  \rangle \oplus \Z \langle c \rangle}{ \Z\langle a - b - c, \, a - b + c \rangle}$
        
        Let $x = a - b$, $y = c$. Then we have, 
        \[x + y = 0, \quad x - y = 0 \implies 2x = 2 y = 0\] 
        i.e. both $x$ and $y$ have order $2$. And, 
        \[x - y = 0 \implies x = y\]
        So,  
        \[H_1^\Delta(\RP^2) = \frac{\Z\langle a -b  \rangle \oplus \Z \langle c \rangle}{ \Z\langle a - b - c, \, a - b + c \rangle} \iso \frac{\Z\langle x \rangle }{2x} \iso \Z/2\Z\]

        
        \item $H^\Delta_2(\RP^2) = \frac{\ker(\d_2)}{\im(\d_3)} = \frac{\{0\}}{\{0\}} \iso \{0\} $

        \item $H_n^\Delta = 0$ if $n \geq 3$. 
    \end{itemize}
\end{solution}

\begin{problem} {2}
    Prove directly that singular homology is an invariant of homeomorphism class of the space. Here `directly' means without referring to the fact that singular homology is an invariant of homotopy equivalence class of the space.
\end{problem}

\begin{solution}
    \bbni

    Let $f: X \to Y$ be a homeomorphism. Then for each singular $n$-simplex $\sigma^n: \Delta^n \to X$ we define $f'_n(\sigma^n) = f \circ \sigma^n$, i.e. $f'(\sigma^n): \Delta^n \to Y$. We will show that the following diagram commutes,
    \[\begin{tikzcd}
        C_n(X) \ar[r, "\d_n"]  \ar[d, "f'_{n}"] & C_{n-1}(X)  \ar[d, "f'_{n-1}"]\\ 
        C_n(Y) \ar[r, "\d_n"] & C_{n-1}(Y)  
    \end{tikzcd}\]
    So, 
    \begin{align*}
        f_{n-1}' \circ \d_{n} (\sigma^n) &= f_{n-1}' \left(\sum_i (-1)^i \sigma^n  \mid_{[v_0, \ldots, \widehat{v_i}, \ldots, v_n]} \right) = \sum_i (-1)^i f \circ \sigma^n  \mid_{[v_0, \ldots, \widehat{v_i}, \ldots, v_n]} \\ 
        &= \sum_i (-1)^i (f \circ \sigma^n) \mid_{[v_0, \ldots, \widehat{v_i}, \ldots, v_n]} \\ 
        &= \d_n(f \circ \sigma^n) = \d_n \circ f_n' (\sigma^n )
    \end{align*}
    Note that since $X \cong Y$ the map $f^{-1}: Y \to X$ induces the inverse chain maps
     \[\begin{tikzcd}
        C_n(Y) \ar[r, "\d_n"]  \ar[d, "(f^{-1})'_{n}"] & C_{n-1}(Y)  \ar[d, "(f^{-1})'_{n-1}"]\\ 
        C_n(X) \ar[r, "\d_n"] & C_{n-1}(X)  
    \end{tikzcd}\]
    Thus making the chain complexes isomorphic. Now show that the induced maps $\widetilde{f}: H_n(X) \to H_n(Y)$, and $\widetilde{f^{-1}}: H_n(Y) \to H_n(X)$ on the singular homology groups are isomorphisms. 
    \begin{itemize}
        \item \textbf{Well-defined:} Assume $[\sigma^n_\alpha] = [\sigma_\beta^n] \in H_n(X)$. Then, $\sigma^n_\alpha - \sigma_\beta^n \in \im(\d_{n+1})$, that is there exists $\tau \in C_{n+1}(X)$ such that 
        \[\sigma^n_\alpha - \sigma_\beta^n = \d_{n+1}(\tau) \implies f'_n(\sigma_\alpha^n) - f'_n(\sigma_\beta^n) = f_n'(\d_{n+1}(\tau)) = \d_{n+1}(f'_{n+1}(\tau))\]
        Hence, $[f'_n(\sigma_\alpha^n)] = [f'_n(\sigma^n_\beta)] \in H_n(Y)$
        \item \textbf{Isomorphism:} For $\sigma^n \in C_n(X)$ and $\tau^n \in C_n(Y)$, we have
        
        \[\widetilde{f^{-1}} \circ \widetilde{f} ([\sigma^n]) = [(f^{-1})_n' \circ f'_n(\sigma^n)] = [\sigma^n] = \id_{H_n(X)}([\sigma^n])\]
        and 
        \[\widetilde{f} \circ \widetilde{f^{-1}} ([\tau^n]) = [f'_n \circ (f^{-1})'_n([\tau^n])] = [\tau^n] = \id_{H_n(Y)}([\tau^n])\] 
    \end{itemize}

   Hence $\widetilde{f}$ is an isomorphism and the singular homology is an invariant of the homeomorphism class of the topological space. 
\end{solution}

\begin{problem}{3}
    Compute the singular homology groups of a point.
\end{problem}

\begin{solution}
    \bbni

    Let $x_0$ be a point. Then the constant map $\sigma_n: \Delta^n \to X$ gives us a $n$-simplex for all $n$ since constant maps are continuous. Since the restriction of the constant map is the constant map we get, 
    \[\d_n(\sigma_n) = \sum_{i=0}^n (-1)^i \sigma_{n-1} = \sigma_{n-1} \sum_{i=0}^n (-1)^i = \begin{cases*}
        0 ,& n \text{ odd} \\ 
        \sigma_{n-1}, & n \text{ even} \\ 
    \end{cases*}\]

    Thus we get, 
    \[\cdots \xrightarrow{\d_3} C_2(x_0) \xrightarrow{\d_2} C_1(x_0) \xrightarrow{\d_1} C_0(x_0) \xrightarrow{\d_0} 0 \]
    \[\cdots \xrightarrow{0} \Z \xrightarrow{\iso} \Z \xrightarrow{0} \Z \xrightarrow{} 0 \]
    Then for odd $n$ we have, 
    $\ker(\d_n) = \Z$ and $\im(\d_{n+1}) = \im(\id_{C_{n-1}}) = \Z$, so
    \[H_n(x_0) = \frac{\ker(\d_n)}{\im(\d_{n+1})} = \frac{\Z}{\Z} = \{0\}\]
    For even $n > 0$, we have
    $\ker(\d_n) = 0$,  so
    \[H_n(x_0) = \frac{\ker(\d_n)}{\im(\d_{n+1})} = \{0\}\]
    For $n = 0$, we have,
    \[H_0(x_0) = \frac{\ker(\d_0)}{\im(\d_{1})} = \frac{\Z}{0} = \Z\]

    Hence, the singular homology groups of a point are $0$ for $n > 0$ and $\Z$ for $n = 0$. 
\end{solution}

\begin{problem} {4}
     Compute the homology groups of the $\Delta$-complex $X$ obtained from $\Delta^n$ by identifying all faces of the same dimension. Thus $X$ has a single $k$ simplex for each $k \leq n$.
\end{problem}

\begin{solution}
    \bbni
    
    We get 
    \[\cdots \xrightarrow{\d_3} \Delta^2(X) \xrightarrow{\d_2} \Delta^1(X) \xrightarrow{\d_1} \Delta^0 (X) \xrightarrow{\d_0} 0\]
    where for $k \leq n$,
    \[\d_k(\sigma_k) = \sum_i(-1)^i\sigma_{k-1} = \sigma_{k-1}\sum_i(-1)^i = \begin{cases*}
        0, & k \text{ odd} \\ 
        \sigma_{k-1}, & k \text{ even}
    \end{cases*}\]
    and for $k > n$ $\d_k(\sigma_k) = 0$. 
    Then the homology groups are as follows, 
    \begin{itemize}
        \item $H_0(X) = \ker(\d_0)/\im(\d_1) = \Z/0 \cong \Z$
        \item If $k$ odd, $k < n$
        \[H_k(X) = \ker(\d_k)/\im(\d_{k+1}) = \Z/\Z \cong 0\]
        \item If $k = n$, $n$ odd
        \[H_n(X) = \ker(\d_n)/\im(\d_{n+1}) = \Z/0 \cong \Z\]
        \item If $k$ even, $k \leq n$
        \[H_k(X) = \ker(\d_k)/\im(\d_{k+1}) = 0/0 \cong 0\]
        \item If $k > n$
        \[H_k(X) = \ker(\d_k)/\im(\d_{k+1}) = 0/0 \cong 0\]
    \end{itemize}
\end{solution}

\begin{problem}{5}
    Compute $H_i(D^n, S^{n-1})$ for all $i$ and $n$. 
\end{problem}

\begin{solution}
    \bbni
    
    We have the long exact sequence,
    \[\cdots \xrightarrow{} H_i(S^{n-1}) \xrightarrow{i_\star} H_i(D^n) \xrightarrow{\pi_\star} H_i(D^n, S^{n-1}) \xrightarrow{\d_i} H_{i-1} (S^{n-1}) \xrightarrow{i_\star} H_{i-1}(D^n) \xrightarrow{} \cdots\]
    Since $D^n$ is contractible and homology is an invariant of the homotopy equivalence class of the space, we have $H_i(D^n) = 0$ for $i > 0$ and $H_0(D^n) = \Z$. For $i > 1$, the sequence reduces to,
    \[\cdots \xrightarrow{} H_i(S^{n-1}) \xrightarrow{i_\star} 0 \xrightarrow{\pi_\star} H_i(D^n, S^{n-1}) \xrightarrow{\d_i} H_{i-1} (S^{n-1}) \xrightarrow{i_\star} 0 \xrightarrow{} \cdots\]

    Then by exactness, $\im(\pi_\star) = 0 = \ker(\d_i)$ and $\ker(i_\star) = H_{i-1}(S^{n-1}) = \im(\d_i) = \d_i(H_i(D^n, S^{n-1}))$, so $\d_i$ is an isomorphism and for $i > 1$,
    \[H_{i-1}(S^{n-1}) \cong H_i(D^n, S^{n-1})\]

    Since the $\Delta$-complex structure of $S^{n-1}$ is one $0$-cell attached to one $(n-1)$-cell we have, $H_{n-1}(S^{n-1}) = H_0(S^{n-1}) = \Z$ and $H_i(S^{n-1}) = 0$ otherwise. So for $ i \neq 1$ we have, 
    \[H_i(D^n, S^{n-1}) \cong H_{i-1} (S^{n-1}) = \begin{cases*}
        \Z & i = n \\
        0 & otherwise
    \end{cases*}\]

    And for $i = 1$ we have the following exact sequence,
    \[\cdots \xrightarrow{i_\star} 0 \xrightarrow{\pi_\star} H_1(D^n, S^{n-1}) \xrightarrow{\d_1} \Z \xrightarrow{i_\star} \Z \xrightarrow{\pi_\star} \cdots\]
    Then we have, $\im(\d_1) = \ker(i_\star) = 0$ and $\im(\pi_\star) = 0 = \ker(d_1)$. That is, $H_1(D^n, S^{n-1})$ must be the trivial group.

    Hence, finally 
    \[H_i(D^n, S^{n-1}) = \begin{cases*}
        \Z & i = n \\
        0 & otherwise
    \end{cases*}\]
\end{solution}

\begin{problem}{6}
    Show that if $A$ is a retract of $X$ then the map $H_n(A)\to H_n(X)$ induced by the inclusion $A \subset X$ is injective. Now show that there is no retraction from $D^n$ to $S^{n-1}$, for any $n$.
\end{problem}

\begin{solution}
    \bbni

    We know that chain maps between chain complexes induce homomorphisms between the homology groups of the respective complexes. If $A \subseteq X$ is a retract, then we have $i \circ r = \id_A \implies (i \circ r)_\star = (\id_A)_\star \implies i_\star$ is injective. So, $H_n(X) \to H_n(A) \to H_n(X)$ is injective.

    Assume there is a retraction from $D^n$ to $S^{n-1}$. Then $H_{n-1}(S^{n-1}) \xrightarrow{i_\star} H_{n-1}(D^n)$ is an injection. But $H_{n-1}(S^{n-1}) = \Z$ and $H_{n-1}(D^n) = 0$, a contradiction! Hence there is no retraction from $D^n$ to $S^{n-1}$ for any $n$. 
\end{solution}

\begin{problem}{7}
    Let $f : (X,A) \to (Y,B)$ be a map such that both $f : X \to Y$ and the restriction $f : A \to B$ are homotopy equivalences.

    \begin{enumerate}
        \item Show that
        \[
            f_* : H_n(X,A) \to H_n(Y,B)
        \]
        is an isomorphism for all $n$.

        \item For the case of the inclusion
        \[
            f : (D^n,S^{n-1}) \hookrightarrow (D^n,D^n-\{0\}),
        \]
        show that $f$ is not a homotopy equivalence of pairs --- there is no
        \[
            g : (D^n,D^n-\{0\}) \to (D^n,S^{n-1})
        \]
        such that $fg$ and $gf$ are homotopic to the identity through maps of pairs.
        (Observe that a homotopy equivalence of pairs $(X,A) \to (Y,B)$ is also a homotopy equivalence for the pairs obtained by replacing $A$ and $B$ by their closures.)
    \end{enumerate}
\end{problem}

\begin{solution}
    \bbni
    \begin{enumerate}
        \item Since $f$ is a homotopy equivalence we get the following commutative diagram (Corollary 2.11), where each sequence is long exact, 
        \[\begin{tikzcd} 
            \cdots \ar[r] & H_i(A) \ar[r,"i_*"] \ar[d,"f_*"', "\cong"] & H_i(X) \ar[r,"\pi_{1_*}"] \ar[d,"f_*"', "\cong"] & H_i(X,A) \ar[r,"\partial_i"] \ar[d,"f_*"] & H_{i-1}(A) \ar[r] \ar[d,"f_*"', "\cong"] & H_{i-1}(X) \ar[d,"f_*"', "\cong"] \ar[r] & \cdots \\ 
            \cdots \ar[r] & H_i(B) \ar[r,"j_*"] & H_i(Y) \ar[r,"\pi_{2_*}"] & H_i(Y,B) \ar[r,"\partial_i"] & H_{i-1}(B) \ar[r] & H_{i-1}(Y) \ar[r] & \cdots 
        \end{tikzcd}
        \]
        Since the sequences are exact and we have the four isomorphisms induced by $f$, by the five lemma we know that $f_*$ is also an isomorphism. Since $i$ was arbitrary, this holds for all homologies and hence $ f_* : H_n(X,A) \to H_n(Y,B)$ is an isomorphism for all $n$.
       

        \item For the sake of contradiction assume $f$ is a homotopy equivalence of pairs. Then there exists its homotopy inverse $g: (D^n, D^n - \{0\}) \to (D^n, S^{n-1})$ and we have 
        \[fg \simeq \id_Y \qquad gf \simeq \id_X\]
        Since these homotopies are continuous maps and map $D^n - \{0\} \to D^n - \{0\}$ and $S^{n-1} \to S^{n-1}$ they must also map $\overline{D^n - \{0\}} \to \overline{D^n - \{0\}}$ and $\overline{S^{n-1}} \to \overline{S^{n-1}}$. Since $\overline{D^n - \{0\}} = D^n$ and $\overline{S^{n-1}} = S^{n-1}$ we get another map of pairs, 
        \[f: (D^{n}, S^{n-1}) \to (D^n, D^n)\]
        By part (1) we know that $H_i(D^n, S^{n-1}) \iso H_i(D^n, D^n)$ for all $i$. In particular, we know from Problem 5 that $H_n(D^n, S^{n-1}) = \Z$. But $H_n(D^n, D^n) = 0$, a contradiction! 
    \end{enumerate}
\end{solution}

\begin{problem}{8}
     Show that chain homotopy of chain maps is an equivalence relation.
\end{problem}

\begin{solution}
    \bbni

    If $P$ is a chain homotopy between the chain maps $f$ and $g$, we know that, 
    \[\d P + P \d = g - f\]
    Let $P, Q$ be chain homotopies between $f,g$ and $g,h$. Then, 
    \begin{enumerate}
        \item \textbf{Reflexive:} Let $P = 0$. Then, 
         \[\d 0 + 0 \d = 0 =  f - f\] 
        Hence, $f \simeq_C f$. 
        \item \textbf{Symmetric:} Let $P^{-1} = - P$. Then we have, 
         \[\d (-P) + (-P) \d = - (\d P + P \d) = - (g - f) = f - g\]
        Hence if $f \simeq_C g$, $g \simeq_C f$. 
        \item \textbf{Transitive:} To show that $f \simeq_C h$, we let $R = P + Q$. 
        Then,  
        \[\d (P +Q ) + (P + Q) \d = g - f + h -g = h - f \]
        Hence if $f \simeq_C g$ and $g \simeq_C h$, we have $f \simeq_C h$. 
    \end{enumerate}
\end{solution}
