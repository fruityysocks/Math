\begin{problem}{2}
    Let $P: E \to B$ be a covering map, with $B$ connected. Then if $p^{-1}(b_0)$ has $k$-many elements fr some finite number $k$, $p^{-1}(b)$ also has $k$-many elements for every $b \in B$. 
\end{problem}

\begin{problem}{3}
    If $p: E \to B$ is a covering map with $E$ path-connected and $B$ simply-connected. Then $\phi$ is a homeomorphism.
\end{problem}

\begin{problem}{4}
     If $p: E \to B$ and $p': E' \to B'$ are covering maps, then so is $p \times p': E \times E' \to B \times B'$, where $p \times p'$ is defined in an obvious manner.
\end{problem}

\begin{problem}{5}
    Show that if a subspace $A$ is a deformation retraction of a space $X$, and another subspace $B$ is a deformation retraction of $A$, then $B$ is a deformation retraction of $X$.
\end{problem}

\begin{problem}{6}
    If a space retracts onto a subspace $A$, then the homomorphism on the fundamental group inducted by the inclusion $\iota : A \hookrightarrow X$ is injective. If $A$ is a deformation retract of $X$, then $\iota_\star$ is an isomorphism. 
\end{problem}

\begin{problem}{7}
    Show that if a space $X$ deformation retracts to a point $x \in X$ , then for each neighborhood $U$ of $x$ in $X$ there exists a neighborhood $V \subset U$ of $x$ such that the inclusion map $V \hookrightarrow U$ is nullhomotopic (nullhomotopic means homotopic to a constant map).
\end{problem}
