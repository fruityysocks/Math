\section{Topological Spaces $\&$ Homeomorphisms}

\begin{definition}
    A \textbf{topology} on a set $X$ is a collection $J$ of subsets of $X$ having the following properties. 
    \begin{enumerate}
        \item $\emptyset, X \in J$ 
        \item The union of the elements of any sub-collection of $J$ is in $J$. 
        \item The intersection of finitely many elements in $J$ is in $J$. 
    \end{enumerate}
    A set for which a topology has been specified is said to be a topological space. We think of it as a pair $(X, J)$. 
\end{definition}

\begin{definition}
     Elements of $J$ are called \textbf{open sets} of the topological space $(X, J)$.
\end{definition}

\begin{definition}
     A subspace $Y$ of a topological space $(X, J)$ is a subset $Y \subset X$ which is given the \textbf{subspace topology} where the open sets of $Y = \{U \cap Y : U \in J\}$
\end{definition}

\begin{definition}
    A space $X$ is \textbf{path connected} if every pair of points of $X$ can be joined by a path in $X$. 
\end{definition}

\begin{definition}
    Let $(X_1, J_1)$ and $(X_2, J_2)$ be topological spaces. A map $f: (X_1, J_1) \to (X_2, J_2)$ is said to \textbf{continuous} if $f^{-1} (U) \in J_1$ for all $U \in J_2$.    
\end{definition}

\begin{definition}
     Let $(X, J)$ and $(X', J')$ be topological spaces. Then they are \textbf{homeomorphic} if $\, \exists$ a continuous bijection $f: (X, J) \to (X', J')$. So that $f^{-1}$ is also continuous. 
\end{definition}

\begin{example}
    Show that any two open intervals of $\R$ are homeomorphic (where we put subspace topology on the open intervals). 
\end{example}    

\begin{solution}
    Let $(a,b)$ and $(c,d)$ be open intervals and define $f: (a,b) \to (c,d)$ as follows: 
    \[f(x) = c + (x-a) \frac{d-c}{b-a}\]
    Then, $f$ is continuous since it is a linear function in $x$, bijective because it is strictly increasing (injective) and has a continuous inverse. 
\end{solution}

\begin{remark}
    Number of path-connected topologies is preserved by continuous maps. For instance, $[0,1]$ is not homeomorphic to $(0,1)$, since $[0,1] \setminus \{0\} = (0,1]$ is path-connected but $(0, 1) \setminus \{\text{pt}\} = (0, \text{pt}) \cup (\text{pt}, 1)$ which has two path-connected components. 
    Similarly, $\R$ is not homeomorphic to $\R^2$ since $\R \setminus \{\text{pt}\}$ gives us two disconnected components, but $\R^2 \setminus \{\text{pt}\}$ is path-connected.    
\end{remark}

\begin{example}
    \bbni 
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item Is $\R^2$ homeomorphic to $\R^3$? 
        \item Is $S^2 ( = x^2 + y^2 + z^2  = 1)$ homeomorphic to $T^2 = S^1 \times S^1$?
        \item Is $S^2$ homeomorphic to $S^1 \times [a,b)$? 
    \end{enumerate}
\end{example}

\begin{solution}
    \bbni 
    \begin{enumerate}[label=(\alph*),ref=\theenumi\alph*]
        \item No, since  $\pi_1(\R^2 \setminus \{\text{pt}\}) = \pi_1(S^1) = \Z$ and $\pi_1(\R^3 \setminus \{\text{pt}\}) = \pi_1(S^2) = 0$
        \item No, since $\pi_1(S^2) = 0$ and $\pi_1(T^2) = \Z \oplus \Z$. 
        \item No, since $\pi_1(S^2) = 0$ and $\pi_1(S^1 \times [a,b)) = \pi_1(S^1) = \Z$. 
    \end{enumerate}
\end{solution}
    
\begin{definition}
    If $f$ and $f'$ are continuous maps from spaces $X$ to $Y$ we say that $f$ is homotopic to $f'$ if $\exists$ a continuous map $F: X \times [0,1]  \to Y$ such that $F(x, 0) = f(x)$ and $F(x, 1) = f'(x)$.  $F$ is called the \textbf{homotopy} between $f$ and $f'$, and we write $f \simeq f'$.
\end{definition}

\begin{definition}
    If $X$ and $Y$ are topological spaces, we can define a \textbf{product topology} on $X \times Y$ by saying that a basis consists of the subsets $U \times V$ as $U$ ranges over open sets in $X$ and $V$ ranges over open sets in $Y$.
\end{definition}

\begin{definition}
    Define a \textbf{path} $f$ as $f: [0,1] \to X$ with $f(0) = x_0$ and $f(1) = x$. 
\end{definition}

\begin{definition}
    Two paths $f, f': [0,1] \to X$ are said to be path homotopic if there exist a continuous map $F: I \times I \to X$ such that $\forall s \in I$, 
    \[ F(s, 0) = f(s), \qquad F(s, 1) = f'(s)\]
    and $\forall t \in I$, 
    \[ F(0, t) = x_0, \qquad F(0, t) = x\]

    Then we write $f \underset p \simeq f'$. 
\end{definition}


\begin{example}
    Let $f: X \to \R^2$ and $g: X \to \R^2$ then $f \simeq g$. 
\end{example}

\begin{proof}
     Want to find $F$ such that $F: X \times I \to \R^2$, $F(x, 0) = f(x)$, and $F(x,1) = g(x)$ for all $x \in X$. The following map (straight-line homotopy) works. 
    \[F(x,t) = (1-t)f(x) + tg(x)\]
\end{proof}

\begin{lemma}
     The relations $\simeq$ and $\simeq_p$ are equivalence relations. 
\end{lemma}

\begin{proof}
    \bbni

    Homotopy is an equivalence relation: 
    \begin{enumerate}
        \item \textbf{Reflexive:} $f \simeq f$ by $F(s,t) = f(s)$. 
        \item \textbf{Symmetric:} Let $f \simeq g$ by the homotopy $F(s,t)$. Then $g \simeq f$ is homotopic by the homotopy $G(s,t) = F(s,1-t)$. 
        \item \textbf{Transitive:} Let $f \simeq g$ by $F(s,t)$ and $g \simeq h$ by $G(s,t)$. We can show that $f \simeq h$ by \[H(s,t) = \begin{cases}
            F(s, 2t), & 0 \leq t \leq 1/2 \\ 
            G(s, 2t-1), & 1/2 \leq t \leq 1 
        \end{cases}\]
        Then $H(s, 0) = F(s, 0) = f(s)$, and $H(s, 1) = G(s, 1) = h(s)$. Hence, $f \simeq h$. 
    \end{enumerate}

    Path-Homotopy is an equivalence relation: 
    \begin{enumerate}
        \item \textbf{Reflexive:} $f \simeq_p f$ by $F(s,t) = f(s)$. 
        \item \textbf{Symmetric:} Let $f \simeq_p g$ by the homotopy $F(s,t)$. Then $g \simeq_p f$ is homotopic by the homotopy $G(s,t) = F(s,1-t)$. 
        \item \textbf{Transitive:} Let $f \simeq_p g$ by $F(s,t)$ and $g \simeq_p h$ by $G(s,t)$. We can show that $f \simeq_p h$ by, 
        \[H(s,t) = \begin{cases}
            F(s, 2t), & 0 \leq t \leq 1/2 \\ 
            G(s, 2t-1), & 1/2 \leq t \leq 1 
        \end{cases}\]
        Where 
        \begin{itemize}
            \item $H(s, 0) = F(s, 0) = f(s)$
            \item $H(s, 1) = G(s, 1) = h(s)$ 
            \item $H(0, t) = x_0$ and $H(1, t) = x_1$. 
        \end{itemize}
        Hence, $f \simeq_p h$. 
    \end{enumerate}
    Thus path-homotopies are equivalence relations and we write $[f]$ to denote the path-homotopy equivalence class of $f$. 
\end{proof}

\begin{lemma}[Pasting Lemma]
    Let $X = A \cup B$ be a topological space with $A, B \subset X$ are closed. Let $f: A \to Y, g: B \to Y$ be continuous and $f(x) = g(x), \, \forall x \in A \cap B$, then 
    \[h(x) = \begin{cases*}
        f(x), \, x \in A \\ 
        g(x), \, x \in B
    \end{cases*}\]
    is continuous.
\end{lemma}

\begin{definition}
     Let $f: I \to X$ be a path from $x_0$ to $x_1$ and $g: I \to X$ be a path from $x_1$ to $x_2$, then we define a new path 
    \[f \star g: I \to X\]
    from $x_0$ to $x_2$ by 
    \[f \star g (t) = \begin{cases*}
        f(2t), \, t \in [0, 1/2] \\ 
        g(2t-1), \, t \in [1/2, 1] \\ 
    \end{cases*}\]
    where $\star$ is \textbf{composition of paths}. 

    Some Properties of $`\star'$. 
    \begin{enumerate}
        \item $`\star'$ is well-defined operation in the path homotopy equivalence class, 
        \[[f] \star [g] = [f \star g]\]
        \item Identity: 
        \[[f] \star [e_{x_1}] = [f], \quad [e_{x_0}] \star [f] = [f]\]

        where $e_{x_t}$ is the constant path based at $x \in X$ and $t \in [0,1]$

        \item Inverse: 
        \[[f]\star[\overline{f}] = [e_{x_0}], \quad [\overline f] \star [f] = [e_{x_1}]\]
        where $\overline{f}(t) = f(1-t)$. 

        \item Associative: 
        \[[f] \star ([g]\star[h]) = ([f]\star [g]) \star [h]\]
    \end{enumerate}
\end{definition}

\begin{proof}
    \bbni
    \begin{enumerate}
        \item Let $f \simeq_p f'$ by $F$ and $g \simeq_p g'$ by $G$, then we can show that $f \star g \simeq_p f' \star g'$ by defining $H: I \times I \to X$ such that 
        \[H(s,t) = \begin{cases*}
            F(2s, t), \, s \in [0, 1/2] \\ 
            G(2s -1, t), \, s \in [1/2, 1]
        \end{cases*}\]

        \item $f \star e_{x_1} \simeq_p f$ by $H : I \times I \to X$ such that 
        \[H(s,t) = \begin{cases*}
            f \fr{2 \cdot s}{2-t}, \, s \in \left [0, \frac{2-t}{2} \right ] \\ 
            x_1, \, s \in \left [\frac{2-t}{2},1 \right ] \\ 
        \end{cases*}\]

        \item To see $e_{x_0} \simeq_p f \star \overline f$
        define a path $\alpha(t)$ that starts at $x_0$ and ends at $f(t)$. Then the path-homotopy is given by $\alpha(t) \star \overline{\alpha}(t)$. 

        \[H(s,t) = \begin{cases*}
            f(2st), \, s \in [0, 1/2] \\ 
            f(2t(1-s)), \, s \in [1/2, 1]
        \end{cases*}\]

        \item By well-definedness, 
        \[[f] \star ([g]\star[h]) = [f] \star [g \star h] = [f \star g \star h] = [f \star g] \star h = ([f]\star [g]) \star [h]\]
    \end{enumerate}
\end{proof}

\begin{remark}
    Note that $\star$ operation on the set of path-homotopy equivalence classes of paths in the space $X$ forms a groupoid structure.
\end{remark}

\section{The Fundamental Group}

\begin{definition}
    Let $X$ be a space and $x_0 \in X$ be a point in $X$ we say that a path $f: I \to X$ is a \textbf{loop} based at $x_0$ if $f(0) = f(1) = x_0$
\end{definition}

\begin{definition}
    The set of all path-homotopy equivalence classes of loops based at $x_0$ forms a group under $\star$ called the \textbf{fundamental group} of $x$ based at $x_0$, denoted as $\pi_1(X, x_0)$

    \begin{itemize}
        \item Identity: $[e_{x_0}]$
        \item Inverses: $[f] \to [\overline{f}]$
    \end{itemize}

    \begin{example}
        $\pi_1(\R^n, (0,0)) \equiv 0  \equiv \langle e_{\text{origin}} \rangle$
        \begin{proof}
        Take any two loops $f(t), g(t)$ based at the origin. Then consider the straight-line homotopy,
        \[H(s,t) = f(s) (1-t) + tg(s)\]
        where, 
        \begin{itemize}
            \item $H(s, 0) = f(s)$
            \item $H(s, 1) = g(s)$
            \item $H(0, t) = f(0)(1-t) + tg(0) = (0,0)  = H(1, t)$
        \end{itemize}
        Hence, it is a path-homotopy. 
    \end{proof}
    \end{example}
\end{definition}

\begin{definition}
    Let $\alpha$ be a path from $x_0$ to $x_1$, where $x_0, x_1 \in X$. We define a map $\widehat{\alpha}: \pi_1 (X, x_0) \to \pi_1(X, x_1)$ such that 
    \[\widehat{\alpha}([f]) := [\overline{\alpha}] \star [f] \star [\alpha] =[\overline{\alpha}\star f \star \alpha]  \]
\end{definition}
\begin{proposition}
    $\widehat{\alpha}$ is well-defined and an isomorphism. 
\end{proposition}

\begin{proof}
    Let $f \simeq_p f'$. We need to check that 
    \begin{align*}
    [\overline{\alpha}] \star [f] \star [\alpha]  = [\overline{\alpha}] \star [f'] \star [\alpha]  
    \\ 
    \iff [\overline{\alpha} \star f \star \alpha]  = [\overline{\alpha} \star f \star \alpha] 
    \\
    \iff \overline{\alpha} \star f \star \alpha \simeq_p \overline{\alpha} \star f' \star \alpha
    \end{align*}
    Since $\star$ is well-defined, $[f] \star [\alpha] = [f'] \star [\alpha]$,  $\widehat{\alpha}$ is well-defined


    Let $[f], [g] \in \pi_1(X, x_0)$. 
    \begin{align*}
        \widehat{\alpha}([f]) \star \widehat{\alpha} ([g]) &= [\overline{\alpha}] \star [f] \star [\alpha] \star  [\overline{\alpha}] \star [g] \star [\alpha] \\ 
        &=  [\overline{\alpha}] \star [f] \star [e_{x_0}] \star [g] \star [\alpha] \\
        &=  [\overline{\alpha}] \star [f] \star [g] \star [\alpha] \\ 
        &= \widehat{\alpha}([f] \star [g])
    \end{align*}
    Hence $\widehat{\alpha}$ is a homomorphism. To check that $\widehat{\alpha}$ is an isomorphism, we see that $\widehat{\overline{\alpha}} (\widehat{\alpha}) ([f]) = [f]$ for all $[f] \in \pi_1(X, x_0)$.  
\end{proof}

\begin{corollary}
    The fundamental group of $X$ based at $x_0$ only depends on the path-connected component of $x_0 \in X$ upto isomorphism. 
\end{corollary}

\begin{definition}
    Let $h: X \to Y$ be a continuous map such that $h(x_0) = y_0$, $x_0 \in X, y_0 \in Y$. Then define $h_\star: \pi_1(X, x_0) \to \pi_1(Y, y_0)$ by $h_\star([f]) = [h \circ f]$. This is a \textbf{push-forward} map. 
\end{definition}

\begin{proposition}
    $h_\star$ is a well-defined homomorphism. 
\end{proposition}

\begin{proof}
    Let $f \simeq_p f'$. We need to check that $h \circ f \simeq_p h \circ f'$. 
    Let $H: I \times I \to X$ be the path homotopy between $f$ and $f'$. Consider $h \circ H: I\times I \to Y$. 
    \begin{align*}
        h\circ H(s, 0) &= h \circ f \\ 
        h\circ H(s, 1) &= h \circ f' \\ 
        h\circ H(0,t) &= h \circ f(0) \\ 
        h\circ H(1,t) &= h \circ f(1) = h \circ f(0) 
    \end{align*}

    Hence $h \circ H$ is the required path-homotopy. 

    Consider $[f], [g] \in \pi_1(X, x_0)$. We want to show that 
    $h_\star([f]) \star h_\star([g]) = h_\star([f] \star [g])$. 
    \begin{align*}
        h_\star([f])\star h_\star([g])  = [h \circ f] \star [h \circ g] 
    \end{align*}
    Note that it suffies to show that $h(f \star g) = (h \circ f) \star (h \circ g)$. Now, 
    \begin{align*}
        h(f \star g)(s) &= \begin{cases} 
            (h \circ f)(2s), & s\in \left[0,\frac{1}{2}\right]\\
            (h \circ g)(2s-1), & s\in \left[\frac{1}{2}, 1\right]\\
        \end{cases} \\
            &=( h\circ f) \star (h\star g) 
    \end{align*}
    Hence proved. 
\end{proof}

\begin{theorem}
    If $h: (X, x_0) \to (Y, y_0)$ and $k: (Y, y_0) \to (Z, z_0)$ then 
    \begin{enumerate}
        \item [(a)] $(k \circ h)_\star = k_\star \circ h_\star$
        \item [(b)] If $\id: (X, x_0) \to (X, x_0)$ is the identity map then $(\id)_\star$ is the identity $\pi_1(X, x_0) \to \pi_(X, x_0)$. 
    \end{enumerate}
\end{theorem}

\begin{proof}
    \bbni

    \begin{enumerate}
        \item [(a)] We have, 
        \begin{align*}
            (k \circ h)_\star ([f]) &= \left[(k \circ h) \circ (f)\right] \\
            &= [k(h(f))] \\ 
            &= [k_\star(h_\star([f]))]\\
            &= (k_\star \circ h_\star) ([f])
        \end{align*}
        \item [(b)] $(\id)_\star [f] = [\id \circ f] = [f]$
    \end{enumerate}
\end{proof}

\begin{corollary}
    If $h: X\to Y$ is a homeomorphism then $\pi(X, x_0) \cong \pi(Y, h(x_0)$ via $h_\star$. 
\end{corollary}

\begin{definition}
    Let $P: E\to B$ be a surjective map. A open set $U \subseteq B$ is said to be \textbf{evenly covered} by $P$ if $P^{-1} (U) = \underset \alpha \bigsqcup V_\alpha$, where $V_\alpha$ are open sets of $E$ for all $\alpha$ and $P|_{V_\alpha}$ is a homeomorphism of $V_\alpha$ onto $U$. 
\end{definition}

\begin{definition}
    Let $P: E \to B$ be surjective if every point $b \in B$ has a neighbourhood $U \subset B$ such that $U$ is evenly covered by $P$. Then we say that $P$ is a \textbf{covering map} and $E$ is a \textbf{covering space }of $B$.

    \begin{example}
        \bbni

        \begin{enumerate}
        \item $\id: X \to X$. 
        \item $X \times \{1, 2, \cdots, n\} \overset P \to X$ where $P(x, i)  = x$.
        \end{enumerate} 
    \end{example}
\end{definition}

\begin{theorem}
    The map $P: \R \to S^1$ such that $x \mapsto (\cos(2\pi x), \sin(2 \pi x))$ is a covereing map. 
\end{theorem}

\begin{proof}
    

    Let $U_1 = S^1 \setminus \{(1,0)\}$ and $U_2 = S^1 \setminus \{(-1, 0)\}$. Then it suffices to show that both of these sets is evenly covered. We have,
    \[P^{-1}(U_1) = \sqcup_{n \in \Z} (n, n+1) =: V_n \]
    And, 
    \[P^{-1}(U_2) = \sqcup_{n \in \Z} \left( n - \frac{1}{2}, n + \frac{1}{2} \right) =: W_n \]
    

    Note that $P$ is injective on any interval of length less than $1$, because if $P(x) = P(y)$ then $(\cos(2\pi x), \sin(2 \pi x)) = (\cos(2\pi y), \sin(2 \pi y)) \iff x - y \in \Z$ but that is not possible if $x - y < 1$. $P \mid_{V_n}$ is surjective onto $U_1$ by IVT and is continuous because $P$ is continuous. It has a continuous inverse since it is a continuous bijection from an open interval onto an open subset of $S^1$. 
    
    Also, the invervals $(n, n+1)$ are pairwise disjoint for each $n$. A similar argument applies to $W_n$. Hence, both $U_1$ and $U_2$ are evenly covered and $P$ is a covering map. 
\end{proof}

\begin{definition}
    If a path-connected space $X$ has $\pi_1(X, x_0) \iso 0$ then we say that $X$ is \textbf{simply-connected}.
\end{definition}

\begin{lemma}
    If $X$ is simply-connected, then any two paths in $X$ with the same initial point and end point are path-homotopic. 
\end{lemma}

\begin{proof}
    $f \star \overline{g} \simeq_p e_{x_0}$ and $f \simeq_p f \star e_{x_1} \simeq_p f \star (\overline{g} \star g) \simeq_p (f \star \overline{g}) \star g \simeq_p  e_{x_0} \star g \simeq_p g.$
\end{proof}

\begin{definition}
    Let $p: E \to B$ be a surjective map, and $f: X\to B$ be a map. Then a map $\widetilde{f}: X \to E$ is called \textbf{lifting} of $f$ to $E$ if $p \circ \widetilde{f} = f$. 

    \begin{center}
        \begin{tikzcd}
            {} & E \arrow[d, "p"] \\ 
            X \arrow[r,"f", swap] \arrow[dashed, "\widetilde{f}"]{ur} & B
        \end{tikzcd}
    \end{center}
\end{definition}

\begin{lemma}[Path Lifting Lemma]
    Let $p: E\to B$ be a covering map with $p(e_0) = b_0$. Any path $f:[0,1] \to B$ beginning at $b_0$ can be lifted uniquely to a path $\widetilde{f}: [0,1] \to E$ beginning at $e_0$.
\end{lemma}

\begin{proof}
    Cover $B$ by open sets $U$, each of which is evenly covered by $p$. We can find a subdivision (partition) defined by $s_0, \ldots, s_n \in [0,1]$ such that $f([s_i, s_{i+1}])$ is contained entirely inside a evenly covered $U$ for all $i$. Now we will define $\widetilde{f}$ inductively. Suppose that $\widetilde{f}(0) = e_0$ and it has been defined for $0 \leq s \leq s_i$ for some $i$. We will now define $\widetilde{f}$ at $[s_i, s_{i+1}]$. Suppose that $f([s_i, s_{i+1}]) \subset U$ where $U$ is evenly covered open set of $B$. This implies that $p^{-1}(U) = \sqcup_\alpha V_\alpha$. Now, $\widetilde{f}(s_i)$ lies in one such $V_\alpha$ say $V_0$. 

    Define $\widetilde{f}$ on $[s_i, s_{i+1}]$ as $\widetilde{f}(s) := (p |_{V_0})^{-1}(f(s))$ for all $s \in [s_i, s_{i+1}]$. 
    By pasting lemma $\widetilde{f}$ is continuous on $[0,s_{i+1}]$ and we argue inductively to define $\widetilde{f}$ for $[0,1]$. 
    
    By induction, we can again suppose that for another lift $\widetilde{f'}: [0,1] \to E$ of $f$ with $\widetilde{f'}(0) = e_0$, it agrees with $\widetilde{f}$ of $0 \leq s \leq s_i$ for some $i$ \[\widetilde{f'}(s) = \widetilde{f}(s), \qquad \forall s \in [0, s_i]\]
    Now we would like to show that 
    \[\widetilde{f'} (s) = \widetilde{f} (s), \qquad \forall s \in [s_i, s_{i+1}] \]
    \[\widetilde{f'} (s) =  (p|_{V_0})^{-1} \circ f (s), \qquad \forall s \in [s_i, s_{i+1}] \]
    $\widetilde{f'}(s_i)$ lies in $V_0$ since $\widetilde{f'}(s_i) = \widetilde{f}(s_i)$. 
    $\implies \widetilde{f'}([s_i, s_{i+1}])$ lies in $V_0$. 
    
    $p \circ \widetilde{f'} = f$ on $[s_1, s_{i+1}]$. But $\widetilde{f'}([s_i, s_{i+1}]) \subset V_0$ and $P|_{V_0}$ is a homeomorphism. 

    Then for all $s \in [s_i, s_{i+1}]$
    \[\implies \widetilde{f'}(s) = (p|_{V_0})^{-1} \circ f (s) = \widetilde{f}(s)\]
\end{proof}

\begin{lemma}
    Let $P: E \to B$ be a covering map with $P(e_0) = b_0$. Let $f, g$ be two paths in $B$ from $b_0$ to $b_1$. Let $\widetilde{f}, \widetilde{g}$ be their lifts beginning at $e_0$. Then if $f \simeq_p g \implies \widetilde{f}$ and $\widetilde{g}$ end at the same point and they are path homotopic. 
\end{lemma}

\begin{proof}
    Assume $f \simeq_p g$ with the path homotopy $F : I \times I \to B$. Then Lemma $2.15$ implies that $F$ lifts to $\widetilde{F}: I \times I \to E$ with $\widetilde{F}(0,t) = e_0$.

    Then we can say that $\widetilde{F}(1, t) = e_1$ for some $e_1 \in P^{-1}(b_1), \, \forall t \in I$. 
    By uniqueness of path lifting, $f, g$ have unique lifts to $E$, beginning at $e_0$. We also know that $\widetilde{F}(s,0)$ and $\widetilde{F}(s,1)$ are lifts of $f, g$ both beginning at $e_0$. 

    Then $\widetilde{F}(s,  0) = \widetilde{f}$ and $\widetilde{F}(s,1) = \widetilde{g}$ for all $s \in I$. Hence $\widetilde{F}(1,0) = \widetilde{F}(1,1) \implies \widetilde{f}(1) = \widetilde{g}(1)$, and $\widetilde{f} \simeq \widetilde{g}$. 
\end{proof}

\begin{theorem}
    $\pi_1(S^1, (1,0) = b_0) \iso (\Z, +)$
\end{theorem}

\begin{proof}
    \bbni
    
    \begin{center}
        \begin{tikzcd}
            {} & \R \arrow[d, two heads, "p"] \\ 
            I \arrow[ur, "\widetilde{f}"] \arrow[r, "f", swap] & S^1
        \end{tikzcd}
    \end{center}

    Consider the convering map

    \[\phi: \pi_1 (S^1, b_0) \to \Z, \quad \phi([f]) = f(1)\] 
    By previous lemma, $\phi$ is well-defined. 

    Now we show that $\phi$ is surjective. Let $n \in \Z$. Define $\widetilde{f}$ to be the path in $\R$ which starts at $0$ and ends at $n$. Then by definition, $(P \circ \widetilde{f})$ will start at $(1,0)$ and end at $(1,0)$ where $\widetilde{f}$ is by definition the lift of $(P \circ \widetilde{f})$ beginning at $0$. So $\phi([P \circ \widetilde{f}]) = n$. 

    To show that $\phi$ is injective suppose that $\phi([f]) = \phi([g]) = n$ for some $[f], [g] \in \pi_1(S^1, b_0)$ we need to show that $[f] = [g] \in \pi_1(S^1, b_0)$. 

    Now let $\widetilde{f}$ and $\widetilde{g}$ be two lifts of $f$ and $g$ respectively beginning at $0$. Moreover, since $\R$ is simply connected and $\widetilde{f}(0) = \widetilde{g}(0) = 0$, $\widetilde{f}(1) = \widetilde{g}(1) = n$, $\widetilde{f} \simeq_p \widetilde{g}$,  then $P \circ \widetilde{F}$ will be the path-homotopy between $f$ and $g$. This implies $[f] = [g]$. 

    Now we want to show that $\phi$ is a group homomorphism. Let $[f], [g] \in \pi_1(S^1, b_0)$ such that $\phi([f]) = n, \, \phi([g]) = m$ for $n, m \in \Z$. Now we want to show that $\phi([f] \star [g]) = \phi([f \star g]) = n+m$. 
    We need to find a lift of $f \star g$ beginning at $0$ and ending at $n+m$. 

    Let $\widetilde{f}, \widetilde{g}$ be lifts of $f, g$ respectively, $\widetilde{f}(0) = \widetilde{g}(0) = (0)$. 

    Define a path on $\R$ by 
    
    \[h(s) = \begin{cases}
        \widetilde{f}(2s),  &s \in \left[0, \frac{1}{2}\right] \\ 
        n + \widetilde{g}(2s-1),  &s \in \left[\frac{1}{2}, 1\right] \\ 
    \end{cases}\]

    Then $h$ is a path beginning at $0$ and ending at $n+m$. Now consider, 
    \[P(h(s))= \begin{cases}
        P \circ \widetilde{f}(2s),  &s \in \left[0, \frac{1}{2}\right] \\ 
        P \circ (n + \widetilde{g}(2s-1)) = P \circ \widetilde{g}(2s-1),  &s \in \left[\frac{1}{2}, 1\right] \\ 
    \end{cases} \]
    This implies that $h(s)$ is a lift of $f \star g(s) \implies \phi([f \star g]) = h(1) = n + m$
\end{proof}

\end{remark}

\begin{proof}
    \bbni
    \begin{enumerate}
        \item [(a)] For $[\gamma] \in \pi_1(B, b_0)$, let $\gamma' : I \to E$ be the unique lift of $\gamma$ starting at $e_0$. Define
        \[
            \phi: \pi_1(B, b_0) \to p^{-1}(b_0), \quad \phi([\gamma])
            = \gamma'(1).
        \]
        Note $\gamma'(1) \in p^{-1}(b_0)$ since $p(\gamma'(1)) = \gamma(1)
        = b_0$, so $\phi$ is well-defined.

        
        % Moreover, if $\gamma \simeq \delta$ rel $\{0,1\}$, then by the
        % homotopy lifting lemma their lifts satisfy $\gamma' \simeq \delta'$
        % rel $\{0,1\}$, so $\gamma'(1) = \delta'(1)$ and $\phi$ is
        % independent of representative.

        To see surjectivity, let $e_\alpha \in p^{-1}(b_0)$. Since $E$ is path-connected, there exists a path $\sigma_\alpha: I \to E$ such that $\sigma_\alpha(0) = e_0$ and $\sigma_\alpha(1) = e_\alpha$. 
        
        Then $\gamma = p \circ \sigma_\alpha$ is a loop in $B$ based at $b_0$, since 
        \[\gamma(0) = (p \circ \sigma_\alpha) (0) = p(e_0) = b_0 \quad \text{ and } \quad \gamma(1) = (p \circ \sigma_\alpha) (1) = p(e_\alpha) = b_0\]
       
        By uniqueness of lifts, $\sigma_\alpha$ is the unique lift of $\gamma$ starting at $e_0$, so $\phi([\gamma]) = \sigma_\alpha(1) = e_\alpha$. We can similarly define a unique lift for each $e_\alpha$ and thus $\phi$ is surjective.
        
        \item [(b)] 
        By part (a) we already know that $\phi$ is surjective. To see that $\phi$ is injective, let $[f], [g] \in \pi_1(B, b_0)$ such that $\phi([f]) = \phi([g]) = e_\alpha$ for some $e_\alpha \in p^{-1}(b_0)$, and let $f', g'$ be the lifts of $f, g$ starting at $e_0$. 
         
        Then $f'(0) = g'(0) = e_0$ and $f'(1) = g'(1)= e_\alpha$. Since $E$ is simply connected, $\pi_1(E, e_0) \cong \{e\}$, i.e. any two paths with the same initial and final points in $E$ are path-homotopic. Thus, $[f'] = [g']$, let $F'$ be the homotopy between $f'$ and $g'$. Then we have
        \[
            F'(s,0) = f'(s) \quad \text{and} \quad F'(s,1) = g'(s).
        \]
        Applying $p$ to both sides,
        \[
            (p \circ F')(s,0) = p(f'(s)) = f(s) \implies p (f') = f
        \]
        \[
            (p \circ F')(s,1) = p(g'(s)) = g(s)\implies p (g') = g
        \]
        Therefore $p \circ F' : I \times I \to B$ is a homotopy between $f$
        and $g$.
        Hence, $[f] = [g]$, $\phi$ is injective, and thus bijective.
    \end{enumerate}
\end{proof}

\begin{theorem}
    \[\pi_1(\mathbb{R}^2 \setminus \{(0,0)\}, b_0) \cong \mathbb{Z} \cong \pi_1(S^1), \quad b_0 \in S^1.\]
\end{theorem}

\begin{proof}

    \[
    J : (S^1, b_0) \to (\mathbb{R}^2 \setminus \{(0,0)\}, b_0)
    \]
    induces an isomorphism
    \[
    J_* : \pi_1(S^1, b_0) \to \pi_1(\mathbb{R}^2 \setminus \{(0,0)\}, b_0).
    \]

    Consider
    \[
    r : \mathbb{R}^2 \to S^1, \quad r(x) = \frac{x}{\|x\|}.
    \]

    Observe that $r \circ J = \mathrm{id}$, thus
    \[
    r_* \circ J_* = \mathrm{id}.
    \]
    Thus $J_*$ is injective.

    Choose $[f] \in \pi_1(\mathbb{R}^2 \setminus \{(0,0)\}, b_0)$. Recall that,
    \[
    J_* \circ r_*([f])
    \]
    is explicitly $J \circ r \circ f$. Let
    \[
    g(s) = \frac{f(s)}{\|f(s)\|}.
    \]

    Next show $g \simeq_p f$. Define
    \[
    F : I^2 \to \mathbb{R}^2 \setminus \{(0,0)\}
    \]
    such that
    \[
    F(s,t) = t \cdot \frac{f(s)}{\|f(s)\|} + (1 - t) f(s).
    \]

    \[
    t \cdot \frac{1}{\|f(s)\|} + (1 - t) = 0
    \]
    implies $\|f(s)\|$ negative or $t \notin [0,1]$.

    Thus $F$ is well-defined, since $f \simeq_p g$,
    \[
    J_* \circ r_*([f]) = [f].
    \]

    Thus surjection.
\end{proof}

\section{Retractions}

\begin{definition}
    If $A$ is a subspace of $X$ s.t. $\exists$ a map
    \[r : X \to A\]
    with $r(a) = a$ for all $a \in A$, then we say $A$ is a retract of $X$ and $r$ is a retraction map from $X$ to $A$.
\end{definition}

\begin{definition}
     Let $A$ be a subspace of $X$. $A$ is a deformation retract of $X$ if there exists a homotopy
    \[H : X \times I \to X\]
    such that
    \[H(x,0) = x \quad \forall x \in X,\]
    \[H(x,1) \in A,\]
    \[H(a,t) = a.\]
\end{definition}

\begin{theorem}
    Let $A$ be a deformation retract of $X$. Then $\iota: (A, x_0) \hookrightarrow (X,x_0)$ induces an isomorphism in $\pi_1(A, x_0)$ to $\pi_1(X,x_0)$. If $A$ in a retract of $X$, then $\iota$ is injective. 
\end{theorem}

\begin{proof}
    \bbni
    \begin{enumerate}
        \item[(a)] Let $r$ be the map that retracts $X$ to $A$, and $r(x_0) = a_0$ in $A$. Note that the composition $r \circ \iota$ is the identity on $A$. Now consider the homomorphism $(r \circ \iota)_\star$. We have, 
        \[(r \circ \iota)_\star = r_\star \circ \iota_\star = (\id_A)_\star = \id_{\pi_1(A, a_0)}\]

        Let $\iota_\star([f]) = \iota_\star([g])$. Then,
        \begin{align*}
            &r_\star(\iota_\star([f])) = r_\star(\iota_\star[g]) \\ 
            \implies & (\id_A)_\star([f]) = (\id_A)_\star([g]) \\ 
            \implies & [f] = [g]
        \end{align*}
        Hence $\iota_\star$ is injective. 

        \item [(b)] Let $H$ be the deformation retraction of $X$ to $A$, and $H(x_0, 1) = a_0$ in $A$. From part (a) we know that $\iota_\star$ is injective. So we only need to show that $\iota_\star$ is surjective. 
        
        \noindent
        We define the continuous map $r = H(x, 1)$. Consider the map $\iota \circ r$ which maps $X \to A \to X$.  This map is homotopic to the identity map by $H$ (by definition),
        \[\iota \circ r \simeq \id_X\] 
        Then $(\iota \circ r)_\star = \iota_\star \circ r_\star \simeq (\id_X)_\star = \id_{\pi_1(X, x_0)}$ (Lemma 58.1). Since $\im(r_\star) = \pi_1(A, a_0)$, $\iota_\star$ is surjective.
        Thus, we have an isomorphism $\pi_1(A, a_0) \cong \pi_1(X, x_0)$. 
    \end{enumerate}
\end{proof}

\begin{theorem}[Brouwer-Fixed Point Theorem]
    Every map $h: D^2 \to D^2$ has a fixed point, i.e. $h(x) = x$ for at least one $x \in D^2$. 
\end{theorem}

\begin{proof}
    Suppose towards contradiction $h(x) \neq x$ for all $x \in D^2$. Define a function $\gamma: D^2 \to S^1$ as follows: 
    consider the straight line starting at $h(x)$ and going to $x$. Define $r(x)$ as the intersection of that line with $\delta D^2 = S^1$. Now $r$ is a continuous function, and $r|_{S^1} = Id_{S^1}$. That is, $r$ is a retraction of $D^2$ to $S^1$. 
    
    Since for the inclusion $\iota: S^1 \hookrightarrow D^2$ we should have $\pi_1(S^1, (1,0)) \xrightarrow{\iota_\star} \pi_1(D^2, (1,0))$ is an injection which gives a contradiction. 
\end{proof}

\begin{theorem}[Borsuk-Ulam Theorem]
    Every map $f: S^2 \to \R^2$ has a pair of anti-podal points ($x$ is an antipode of $-x$, $x \in S^2$) in $S^2$ such that $f(x) = f(-x)$. 
\end{theorem}

\begin{proof}
    Suppose that this is false. Then $f(x) \neq f(-x)$ for all $x \in S^2$. We define $g: S^2 \to S^1$ by 
    \[g(x) = \frac{f(x) - f(-x)}{\|f(x) - f(-x)\|}\]
    Define a loop $\eta$ by the equator that lies in the $X-Y$ plane of $S^2$. 
    \[\eta(S) = (\cos(2\pi s), \sin(2\pi s), 0), \, s \in [0,1]\]
    and let $h: I \to S^1$ defined by $h := g \circ \eta$. Note that since $g(x) = -g(-x)$ we have $h(s + \frac{1}{2})  = - h(s)$ for all $s \in \left[0, \frac{1}{2}\right]$. 

    \begin{align*}
        g(\cos(2\pi (s + 1/2)), \sin(2\pi (s + 1/2)), 0) &= g(-\cos(2 \pi s),\, - \sin (2 \pi s), 0)) \\ 
        &= -g(\cos(2 \pi s), \sin (2 \pi s), 0)
    \end{align*}
    By path-lifting lemma, we can lift $h$ to a path in $\R$, say $\widetilde{h}$ starting at $0$ such that 
    \[\widetilde{h} \left(s + \frac{1}{2}\right) = \widetilde{h}(s) + \frac{q(s)}{2}\]
    for some odd integer values $q(s)$. But now, ${q}(s) = 2(\widetilde{h}\left(s + \frac{1}{2}\right) - \widetilde{h}(s))$ and hence it is also a continuous function. This implies $q(s)$ is a constant map (since the domain of $q$ is conected), i.e. $\widetilde{h} \left(s + \frac{1}{2}\right) = \widetilde{h}(s) + \frac{q}{2}$. 

    \[\widetilde{h}(1) = \widetilde{h}\left(\frac{1}{2}\right) + \frac{q}{2} = \widetilde{h}(0) + q\]

    This implies $\widetilde{h}(1) =q$ and hence $[h] \in \pi_1(S^1, (1,0))$ is not homotopic to the identity. But since $\eta$ is nullhomotopic in $S^2$, $g \circ \eta$ is homotopic to a constant map $[h] = 0 \in \pi_1(S^1, (1,0))$. A contradiction! 
\end{proof}

\section{Homotopy Equivalences}

\begin{theorem}
    $\pi_1(X \times Y, x_0 \times y_0) \iso \pi_1(X, x_0) \times \pi_1(Y, y_0)$ for any pair of topological spaces $X$ and $Y$. 
\end{theorem}

\begin{proof}
    Let $P_1: X\times Y \to X$ and $P_2: X \times Y \to Y$ be the respective projections. 
    \[\phi: \pi_1(X \times Y, x_0 \times y_0) \to \pi_1(X, x_0) \times \pi_1(Y, y_0)\]
    where
    \[\phi([f]) = (P_1)_\star[f] \times (P_2)_\star[f]\]
    
    \textbf{$\phi$ is surjective:} Let $[g] \in \pi_1(X, x_0)$ and $[h] \in \pi_1(Y, y_0)$. Then define $[g] \times [h]: I \to X \times Y$ by $s \mapsto g(s) \times h(s)$. Then $\phi([g\times h]) = [g] \times [h]$. 
    
    \textbf{$\phi$ is injective:} Suppose that $[f] \in \pi_1(X \times Y, x_0 \times y_0)$ such that $\phi([f]) = (P_1)_\star([f]) \times (P_2)_\star([f]) = 0 = ([e_{x_0}] \times [e_{y_0}])$. Then $P_1 \circ f \simeq_p e_{x_0}$ and $P_2 \circ f \simeq_p e_{y_0}$ by homotopies $F_t$ and $G_t$. 

    Then, $F_t \times G_t: I \to X \times Y$ gives $f \simeq_p e_{x_0 \times y_0} \implies [f] =0$.  
\end{proof}

\begin{definition}
    We say $f: X \to Y$ is a homotopy equivalence if there exists  $g: Y \to X$ such that $f \circ g \simeq id_Y$ and $g \circ f \simeq id_X$. Then we say that $X$ is homotopy equivalent to $Y$ and write it as $X \simeq_h Y$.
\end{definition}

\begin{remark}
    Homotopy equivalence is an equivalence relation on topological spaces. 
\end{remark}

\begin{proposition}
    Suppose that $X$ deformation retracts to $A$. Then $X \simeq_h A$. 
\end{proposition}

\begin{proof}
    Let $H: X \times I \to A$ be the deformation retraction, and $\iota: A \hookrightarrow X$ be the inclusion map. Then let $r = H(x, 1)$ for all $x \in X$. Now note that, 
    \[r \circ \iota \simeq \id_A\]
    And, by definition of the deformation retraction $H$ gives us, 
    \[\iota \circ r \simeq \id_X\]
    
    Thus, $X \simeq_h A$. 
\end{proof}

\begin{remark}
    Note not all homotopy equivalence come from deformation retractions. 
\end{remark}

\begin{remark}
    $\pi_1$ is not only an invariant of homeomorphism class of a topological space but also an invariant of the homotopy equivalence class of the space. 
\end{remark}

\begin{lemma}
    If $\phi_t: X\to Y$ be a homotopy and $h$ is the path given by $\phi_t(x_0)$ for a choice of $x_0 \in X$, then we have $(\phi_0)_\star = \widehat{\overline{h}}((\phi_1)_\star)$ on $\pi_1(X, x_0)$. That is, 
    \begin{center}
        \begin{tikzcd}
            \pi_1(X,x_0) \arrow[dr, "(\phi_0)_\star", swap] \arrow[r, "(\phi_1)_\star"] & \pi_1(Y, \phi_1(x_0)) \arrow[d, "\widehat{\overline{h}}"]\\ 
            {} & \pi_1(Y, \phi_0(x_0)) 
        \end{tikzcd}
    \end{center}
\end{lemma}

\begin{proof}
    Let $[f] \in \pi_1(X, x_0)$. Let $h_t$ be a reparameterization of the path $h$ such that \\ $h_t(s) := h(st)$. In particular, $h_t$ is a path from $\phi_0(x_0)$ to $\phi_t(x_0)$. Now the composition, $h_t \star \phi_t \circ f \star \overline{h}_t$ gives a path homotopy between 
    \[h_0 \star \phi_0 \circ f \star \overline{h}_0 = e_{x_0}\star \phi_0 \circ f \star \overline{e}_{x_0} \simeq \phi_0 \circ f\]
    and  
    \[h_1 \star \phi_1 \circ f \star \overline{h}_1 = h \star \phi_0 \circ f \star \overline{h} \implies (\phi_0)_\star = \widehat{\overline{h}}((\phi_1)_\star)\]
\end{proof}

\begin{theorem}
    Let $\phi: X \to Y$ be a homotopy equivalence. Then $\phi_\star: \pi_1(X, x_0) \to \pi_1(Y, \phi(x_0))$ is an isomorphism. 
\end{theorem}

\begin{proof}
    Since $\phi$ is a homotopic equivalence, there exists a $\psi: Y \to X$ such that $\phi \circ \psi \simeq \id_Y$ and $\psi \circ \phi \simeq \id_X$. Consider, 
    \[\pi_1(X, x_0) \xrightarrow{\phi_\star} \pi_1(Y, \phi(x_0)) \xrightarrow{\psi_\star} \pi_1(X, \psi(\phi(x_0))) \xrightarrow{\phi_\star} \pi_1(Y, \phi(\psi(\phi(x_0))))\]
    Note that $\psi_\star \circ \phi_\star = (\psi \circ \phi)_\star = \widehat{\overline{h}}\circ (\id_X)_\star$ for some path $h$ in $X$ (by the previous lemma). Tha implies  $ \psi_\star \circ \phi_\star = \widehat{\overline{h}} \implies \psi_\star$ is surjective since $\widehat{\overline{h}}$ is an isomorphism. 

    $\psi_\star$ is injective because 
    \[\phi_\star \circ \psi_\star = \widehat{\overline{h'}} \circ (\id_Y)_\star\] 
    for some path $h'$ in $Y$ (by the previous lemma). 

    Hence, $\phi_\star = (\psi_\star)^{-1} \circ \widehat{\overline{h}}$ which is an isomorphism. 
\end{proof}

\begin{theorem}
    Define $p: S^n \to \RP^n$, $x \mapsto [x]$. Then $p$ is a covering map. 
\end{theorem}

\begin{proof}
    First show that $p$ is an open map (i.e. $U$ open $\implies p(U)$ open). Let $V \subset S^n$ be an open set ($S^n$ has subspace topology  from $\R^{n+1})$. Consider `antipodal map', 
    \[a: S^n \to S^n, \quad x \mapsto -x\]
    Then $a$ is a homeomorphism. Thus its inverse is continuous, and $a(V)$ is open in $S^n$. Now since $p^{-1}(p(V)) = V \cup a(V)$, which is open in $S^n$, $p$ is an open map (using definition of quotient topology). 

    Now we will show $p$ is a covering map. Let $y \in \R P^n$. Pick $x \in p^{-1}(y)$. Choose a small neighborhood $U$ of $x$ in $S^n$ such that $U$ contains no antipodal points. Then $p|_U: U \to p(U)$ is a homeomorphism (since $p$ is open and continuous and a bijection).

    Similarly, $p|_{a(U)}: a(U) \to p(a(U))$ is also an homeomorphism. Note $p(a(U)) = p(U)$ which is an open set around $y \in \RP^n$. So each point in $\RP^n$ is evenly covered by an open set in $\RP^n$. Hence, $p$ is a covering map (specifically a $2$-fold covering). 

    Then $\pi_1(\RP^n, x_0) \leftrightarrow p^{-1}(x_0) \implies |\pi_1(\RP^n, x_0)| = 2 \implies  |\pi_1(\RP^n, x_0)| = \Z/2\Z $. 
\end{proof}

\begin{proposition}
    Let $h: S^1 \to Y$ where $Y$ is some topological space, then $h$ is homotopic to a constant map iff $h$ can be extended to a continuous map $D^2 \to Y$. 
\end{proposition} 
\begin{proof}
    ($\implies$)

    Let $H: S^1 \times I \to Y$ be a homotopy from $h$ to a constant map $S^1 \to x_0$. Define $\pi: S^1 \times I \to D^2$ by $\pi(x,t) = (1-t)x$. Then $\pi\mid_{S^1 \times [0,1)}$ is a bijection to $D^2 \setminus \{0\}$ and it maps $S^1 \times \{1\}$ to $0 \in D^2$. 
    
    Since $S \times I $ is compact and $D^2$ is Hausdorff, $\pi$ is a closed map. Now $H$ induces a map $g: D^2 \to Y$ such that $g \circ \pi = H$, given by 
    \[g(x) = \begin{cases}
        H(\pi^{-1}(x)) & x \neq 0 \\ 
        y_0 & x = 0
    \end{cases}\]
    The map $g$ is continuous since $\pi$ is a quotinent map. Thus the map $g$ is the desired extension of $h$ for if $x \in S^1$ then, 
    \[g(x) = g(\pi(x,0)) = H(x,0) = h(x)\]

    ($\impliedby$)

    Let $g: D^2 \to Y$ be a continuous extension of $h$, and define $F: S^1 \times I \to Y$ by 
    \[F(x,t) = g((1-t)x)\]
    Then $F$ is a homotopy between $h$ and a constant map.  
\end{proof}

\begin{proposition}
    Let $p: (\widetilde{X}, \widetilde{x}_0) \to (X, x_0)$ be a covering map. Then the map $p_*$ is injective. Moreover, $p_*(\pi_1(\widetilde{X}, \widetilde{x}_0))$ consists of homotopy classes of loops in $X$ that lift to loops in $\widetilde{X}$ based at $\widetilde{x}_0$.
\end{proposition}

\begin{proof}
    Let $[f] \in \pi_1(\widetilde{X}, \widetilde{x}_0)$ and suppose $p_*([f]) = 0 \in \pi_1(X, x_0)$. Then there exists a path homotopy $H$ between $p \circ f$ and the constant loop $c_{x_0}$. By the homotopy lifting lemma, we can lift this to a homotopy between $f$ and the trivial loop $c_{\widetilde{x_0}}$ in $\widetilde{X_0}$. Thus, $[f] \simeq_p 0$, and $p_{*}$ is injective. 
    
    Moreover, loops at $x_0$ lifting to loops at $\widetilde{x_0}$ certainly represent elements of $p_*(\pi_1(\widetilde{X}, \widetilde{x}_0))$, and conversely, a loop representing an element of $p_*(\pi_1(\widetilde{X}, \widetilde{x}_0))$ is homotopic to a loop having sucha lift, so by homotopy lifting, the loop itself must have such a lift. 
\end{proof}

\begin{proposition}
    Given $X$ and $\widetilde{X}$ path connected, then the number of sheets of $p: (\widetilde{X}, \widetilde{x}_0) \to (X, x_0)$ of the covering equals the index of $p_*(\pi_1(\widetilde{X}, \widetilde{x}_0))$ in $\pi_1(X, x_0)$.
\end{proposition}

\begin{proof}
    Let $g$ be a loop based at $x_0$ in $X$. Let $\tilde{g}$ be its lift in $\widetilde{X}$, starting at $\widetilde{x}_0$. 
    
    Now let $[h] \in H:= p_*(\pi_1(\widetilde{X}, \widetilde{x}_0))$, and let $\widetilde{h} \circ \widetilde{g}$ be a lift based at $\widetilde{x}_0$, starting at $\widetilde{x}_0$. We look at the lift of ${h} \circ g$, $\widetilde{h} \circ \widetilde{g}$ which ends at the same point as $\widetilde{g}$ since $\widetilde{h}$ is a loop at $\widetilde{x_0}$. 

    Thus we can define a function $\phi$ from the cosets $H[g]$ to $p^{-1}(x_0)$. $\phi$ is surjective since $\widetilde{X}$ is path-connected and hence $\widetilde{x_0}$ can be joined to any point in $p^{-1}(x_0)$ by a path $\widetilde{g}$ that projects to a loop $g$ at $x_0$. 

    To see that $\phi$ is injective. Let $\phi(H[g_1]) = \phi(H[g_2])$. Then, the lifts correspoding to $g_1$ and $g_2$ end at the same point, and thus $[g_1 \circ \overline{g_2}] = [g_1][{g_2}]^{-1} \in H \implies H[g_1] = H[g_2]$.  

    Thus there exists a bijection between the set of cosets of $\pi_1(X, x_0)$ and the fibre $p^{-1}(x_0)$ and $|p^{-1}(x_0)| = $ number of cosets $= [\pi_1(X, x_0) : H]$. 
\end{proof}

\begin{theorem}[Seifert-Van Kampen Theorem]

    If $X$ is a union of path-connected open sets $A_\alpha$ such that $x_0 \in X$, and $x_0 \in A_\alpha$ for all $\alpha$, and $A_\alpha \cap A_\beta$ is path-connected, and $A_\alpha \cap A_\beta \cap A_\gamma$ is also path-connected for all $\alpha, \beta, \gamma$. Then $\Phi$ induces an isomorphism,
    \[\pi_1(X) \iso \star_\alpha \pi_1(A_\alpha) / N\]
\end{theorem}

\begin{proof}
    We have shown before that $\Phi$ is surjective. Now we will show that $\ker(\Phi) = N$. Define a factorization of an element $[f] \in \pi_1(X)$ as the product $[f_1] \ldots [f_k]$ such that each $[f_i] \in \pi_1(A_\alpha)$ for some $A_\alpha$ and $f \sim_p f_1 \star f_2 \star \cdots \star f_k$. 
    A factorization of $f$ is thus a word in $\star_\alpha \pi_1(A_\alpha)$. We will say that any two such factorizations are equivalent if they are related by the following two moves or their inverses: 
    \begin{enumerate}
        \item Combine adjacent terms $[f_i][f_{i+1}]$ into a single term if $[f_i][f_{i+1}]$ are in the same group. 
        \item If $f_i \in A_\alpha \cap A_\beta$ we may regard $[f_i] \in \pi_1(A_\alpha)$ as $[f_i] \in \pi_1(A_\beta)$. 
    \end{enumerate}
    The first move does not change the factorization. The second move does not the element in $\star_\alpha \pi_1(A_\alpha)/N$. 


    Let us assume that $[f_1]\ldots[f_k]$ and $[f_1']\ldots[f_l']$ are two factorizations of $[f] \in \pi_1(X)$. We will show that they are equivalent. 
    We have, 
    \[f_1 \star \cdots \star f_k \iso f_1' \star \cdots \star f_l'\]
    That is, there exists a path-homotopy $F: I \times I \to X$. Now choose a partition of $I \times I$ by 
    \[0 = s_0 < \cdots < s_m = 1 \text{ and } 0 = t_0 < \cdots < t_n = 1\]
    such that each rectangle $R_{ij} = [s_{i-1}, s_i] \times [t_{j-1}, t_j]$ is mapped to a single $A_\alpha$ by $F$. We label this $A_\alpha$ by $A_{ij}$. 
    
    Now we will modify the partition in two ways: 
    \begin{enumerate}
        \item Add subdivisions to the top ($[t_{n-1}, t_n]$) and bottom row ($[t_0 - t_1]$) such that for each end point in the partition there exists a corresponding $s_i$. 
        \item Have at least 3 rows. 
        \item Every point in $I \times I$ must lie in at most 3 (smaller) rectangles 
        \item Rename the smaller rectangles (left to right, bottom to top) using $R_i$.
    \end{enumerate}
    Define $\gamma_r$ as the path from the left to the right edge of $I \times I$, and it separates rectangles $R_1, \ldots R_r$ from the rest of the rectangles.
    
    Note that $F_{\gamma_r}$ is a loop based at $x_0$. And all $F_{\gamma_r}$ are path homotopic. Then $F|_{\gamma_r}$ has a factorization. If $F(v) \neq x_0$, then we have a path from $x_0 \to F(v) = F |_{\gamma_1}$ 
\end{proof}

\begin{definition}
    We start with a discrete space whose points are regarded as $0$-cells. Now, we inductively define the \textbf{$n$-skeleton} $X^n$ from $X^{n-1}$ by attaching $n$-cells $e_\alpha^n$ via maps $\phi_\alpha: S^{n-1} \to X^{n-1}$ where $e^n_\alpha$s are $D^n_\alpha$. Hence, $X^n = X^{n-1} \sqcup D^n_\alpha / (x \sim \phi_\alpha(x) \text{ if } \alpha, \forall x \in S^{n-1})$. We say a topological space $X$ has a CW/CX structure if $X = \cup_n X^n$.
\end{definition}

\begin{definition}
    
    Each cell $e_\alpha^n$ of a CW CX has a map $\Phi_\alpha: D^n_\alpha \to X$ which extends $\phi_\alpha: \d D^n_\alpha \to X$. It is defined by 
    \[D_\alpha^n \hookrightarrow X^{n-1} \sqcup D_\alpha^n \to \underset{\phi_\alpha: \d D^n_\alpha \to X^{n-1}}{ X^{n-1} \sqcup D_\alpha^n} = X^n \hookrightarrow x\]
    This map is called the \textbf{characteristic map} asssociated to $\phi_\alpha$. 
\end{definition}

\begin{definition}
    A \textbf{subcomplex} $A$ of a CW CX $X$ is a subspace $A \subset X$ such that $A$ is a union of some number of cells in $X$ with a attaching map for each of those cells has a image contained in $A$. We say that $(X, A)$ is a CW pair. 
\end{definition}

\begin{definition}[Quotinent CX]
    If $(X, A)$ is a CW pair, then $X/A$ inherits a canonical cell structure from $X$. The cells of $X/A$ are the cells $X - A$, plus one $0$-cell given by the image of $A$ in $X/A$. For a cell $e_\alpha^n$ of $X - A$ with $\phi_\alpha: S^{n-1} \to X^{n-1}$ , the corresponding attaching map in $X/A$ is 
    \[S^{n-1} \to X^{n-1} \to X^{n-1} / A^{n-1}\]
\end{definition}

\begin{theorem}
    If $(X, A)$ is a CW pair and $A$ is contractible then $q: X \to X/A$ is a homotopy equivalence. 
\end{theorem}

\begin{proof}
    Hatcher Chapter 0, Homotopy Extension Lemma
\end{proof}

\section{Homology}

\begin{remark}
    $\pi_n(S^n) = \Z$, and $\pi_m(S^n) = 0$ if $m < n$. We don't know how to compute $\pi_m(S^n)$ for $m > n$.
\end{remark}
Homology distinguishes higher dimension space and it is easy to compute. We define simplical and singular homology separately, but eventually they have the same properties. 

\begin{remark}
    For spheres the homology groups $H_i(S^n)$ are isomorphic to the homotopy groups $\pi_i(S^n)$ for $1 \leq i \leq n$; and for $i > n$, $H_i(S^n) = 0$. 
\end{remark}

\subsection{Simplical Homology}

\begin{definition}
    A \textbf{$n$-simplex} is the smallest convex set in $\R^m$ spanned by $(n+1)$ points $v_0, \ldots, v_n$ such that all of them do not lie in any hyperplane of dimension less than $n$. That is, $v_1 - v_0, \, v_2 - v_0, \ldots, \, v_n - v_0$ are linearly independent. We say $(v_0, \ldots, v_n)$ comes with the order $v_0, \ldots, v_n$.

    The standard $n$-simplex is 
    \[\left\{(t_0, \ldots, t_n) \in \R^{n+1} \mid \sum_{i} t_i = 1, t_i \geq 0\right\}\]
    There is a canonical homeomorphism from $\Delta^n$ to any $n$-simplex. 
    \[(t_0, \ldots, t_n) \mapsto \sum t_iv_i\]
\end{definition}

\begin{definition}
    A \textbf{face} of an $n$-simplex $\Delta^n = [v_0, \ldots, v_n]$ is the $n-1$-simplex obtained by deleting one of its $n+1$ vertices. The vertices of a face, or of any subsimplex spanned by a subset of the vertices, will always be ordered according to their order in the larger simplex.
\end{definition}

\begin{definition}
    The \textbf{boundary} $\d{\Delta}^n$ is the union of all the faces of $\Delta^n$.
\end{definition}

\begin{definition}
    The \textbf{open simplex} $\mathring{\Delta}^n$ is the interior of $\Delta^n$, i.e. $\Delta^n - \d \Delta^n$. 
\end{definition}

\begin{definition}
    The \textbf{$\Delta$-complex structure} of a space $X$ is a collection of maps $\sigma_\alpha: \Delta^n \to X$. 
    \begin{enumerate}
        \item $\sigma_\alpha |_{\mathring{\Delta}^{n}}$ is injective and each point of $X$ is in the image of exactly one such $\sigma_\alpha |_{\mathring{\Delta}^{n}}$. 
        \item Restriction of $\sigma_\alpha$ to a face of ${\Delta}^{n}$  is one of the maps $\sigma_\beta: \Delta^{n-1} \to X$. We identify the face of $\Delta^n$ with $\Delta^{n-1}$ by the canonical linear homeomorphism between them that preserves the ordering of the vertices.
        \[\lambda_i: \Delta^{n-1} \to \Delta^n, \qquad (t_0, \ldots, t_{n-1}) \mapsto (t_0, \ldots,\, t_{i-1}, \, 0 , \, t_i, \ldots,\, t_{n-1} )\]
        \item A set $U \subset X$ is open iff $\sigma_\alpha^{-1} (U)$ is open in $\Delta^n$, for all $\alpha$.
    \end{enumerate}
\end{definition}


\begin{definition}
    Let $X$ be a $\Delta$-complex and let $\Delta_n(X)$ denote the free abelian group with basis $n$-simplicies of $X$. The elements of $\Delta_n(X)$ are of the form $\sum_i n_i \sigma_i$ where this is a finite sum, $n_i \in \Z$ and $\sigma_i: \Delta^n \to X$. 
\end{definition}

We denote the deletion of $v_i$ from the simplex as $[v_0, \ldots, \widehat{v_i}, \ldots, v_n] \cong [v_0, \ldots, v_{n-1}]$. We will always orient the $1$-simplex $[v_i, v_j]$ by an arrow going from $v_i$ to $v_j$ for $i < j$. 

\begin{definition}
    We define a \textbf{boundary homomorphism} $\d_n: \Delta_n(X) \to \Delta_{n-1}(X)$ where, 
    \[\d_n(\sigma_\alpha) = \sum_i (-1)^i \sigma_\alpha \mid _{[v_0, \ldots, \widehat{v_i}, \ldots, v_n]}\]
\end{definition}
\begin{proposition}
    The composition 
    \[\Delta_n(X) \xrightarrow{\d_n} \Delta_{n-1}(X) \xrightarrow{\d_{n-1}} \Delta_{n-2}(X)\] 
    is zero $(\d_{n-1} \circ \d_n = 0)$.
\end{proposition}

\begin{proof}
    \begin{align*}
        \d_n(\sigma) &= \sum_i(-1)^i \sigma |_{[v_0, \ldots, \widehat{v_i}, \ldots, v_n]} \\ 
        \d_{n-1} \circ \d_n (\sigma) &= \sum_{j <i}(-1)^i(-1)^j \sigma |_{[v_0, \ldots, \widehat{v_j}, \ldots \widehat{v_i}, \ldots, v_n]} +\sum_{i < j}(-1)^i(-1)^{j-1} \sigma |_{[v_0, \ldots, \widehat{v_i}, \ldots \widehat{v_j}, \ldots, v_n]}  \\
    \end{align*}
\end{proof}

\begin{definition}
    In general, whenever we have $(C_n, \d_n)$
    \[\cdots \xrightarrow{} C_{n+1} \xrightarrow{\d_{n+1}} C_n \xrightarrow{\d_n} C_{n-1} \xrightarrow{\d_{n-1} \cdots }\] 
    with $\d_n \circ \d_{n+1} = 0$, then $\img(\d_{n+1}) \subset \ker(\d_n)$. Hence we define the \textbf{$n$-th homology group}, 
    \[H_n := \ker(\d_n) / \im(\d_{n+1})\]
    where elements of $\ker(\d_n)$ are \textbf{cycles} and elements of $\im(\d_{n+1})$ are \textbf{boundaries}. The \textbf{$n$-th simplicial homology} of $X$ is $H_n^\Delta(X) = \ker(\d_n) / \im(\d_{n+1})$.
\end{definition}

\begin{example}
    Let $X = S^1$. There is one 0-simplex $v$ and one 1-simplex $e$. Then, 

    \[\Delta_0(S^1) = \Z \langle v \rangle, \, \Delta_1(S^1) = \Z \langle e \rangle, \, \Delta_2(S^1) = 0\]
    And we have, 
    \[0 \xrightarrow{\d_2} \Delta_1(S^1) \xrightarrow{\d_1} \Delta_0 (S^1) \xrightarrow{\d_0} 0\]
    with 
    \[H_0^\Delta(S^1) = \frac{\ker(\d_0)}{\im(\d_1)} = \frac{\Z \langle v \rangle}{ \{\sigma_1 \mid_{[v_1] }- \sigma_1 \mid_{[v_0]}\}} = \frac{\Z\langle v \rangle}{\{0\}} \iso \Z\langle v \rangle \iso \Z,\]
    \[H_1^\Delta(S^1) = \frac{\ker(\d_1)}{\im(\d_2)} = \frac{\Z \langle v \rangle}{ \{0\}} \iso \Z\langle v \rangle \iso \Z,\]
    and
    \[H_2^\Delta(S^1) = \frac{\ker(\d_2)}{\im(\d_3)} = { 0}\]

    We conclude that $H_n^\Delta(S^1) \iso \Z$ if $n = 0, 1$ and  $H_n^\Delta(S^1) = 0$ otherwise. 
\end{example}

\begin{definition}
    A \textbf{singular $n$-simplex} in a space $X$ is a map $\sigma: \Delta^n \to X$, where there are no conditions of $\im(\sigma)$ restricted to a face (i.e. singularities are allowed). The only requirement is that $\sigma$ is continuous. 
\end{definition}

\begin{definition}
    A \textbf{chain complex} $C_n(X)$ is the free abelian group with the set of singular $n$-simplices in $X$ as basis. Its elements are called \textbf{$n$-chains} which are (finite) formal sum of singular $n$-simplices. 
    \[C_n(X) := \left \{\sum_i n_i \sigma_i \mid \sigma_i : \Delta^n \to X\right \}\]
    with the differential $\d_n : C_n(X) \to C_{n-1}(X)$ where, 
    \[\d_n(\sigma):= \sum_i (-1)^i \sigma \mid_{[v_0, \ldots, \widehat{v_i}, \ldots, v_n]}\]
     and $\d_n \circ \d_{n+1} = 0$. The \textbf{$n$-th singular homology} is \[H_n(X) := \frac{\ker(\d_n)}{\im(\d_{n+1})}\]
\end{definition}

\begin{remark}
    $H_n(X)$ is an invariant under the homeomorphism class of $X$. 
\end{remark}

\begin{proposition}
    Let $X = \sqcup X_\alpha$ then $H_n(X) \iso \oplus_\alpha H_n(X_\alpha)$, where each $X_\alpha$ is a distinct path connected component of $X$. 
\end{proposition}

\begin{proof}
    We have $C_n(X) \iso \oplus_\alpha C_n(X_\alpha)$ and $\d_n = \oplus_\alpha \d_n^\alpha$ which implies 
    \[H_n(X) \iso \oplus_\alpha H_n(X_\alpha)\]
\end{proof}

\begin{proposition}
    If $X$ is non-empty and path-connected then $H_0(X) \iso \Z$. 
\end{proposition}

\begin{proof}
    By definition, 
    \[H_0(X) \iso \frac{C_0(X)}{\im(\d_1)}\]
    We define a homomorphism 
    \[\varepsilon: C_0(X) \to \Z, \quad \sum_i n_i\sigma_i \mapsto \sum_i n_i\]
    This map is surjective since for any $n \in \Z$, $\varepsilon(n\sigma_0) = n$ where $\sigma_0: \Delta^0 \to X$. 

    \textbf{Claim:} $\ker(\varepsilon) = \im(\d_1)$. 

    \begin{proof}
        Note that $\im(\d_1) \subseteq \ker(\varepsilon)$ since for a singular $1$-simplex $\sigma: \Delta^1 \to X$ we have $\varepsilon(\d_1(\sigma)) = 0$. 
        
        Now for $\ker(\varepsilon) \subseteq \im(\d_1)$, let $\varepsilon\left(\sum_i n_i\sigma_i\right) = 0 \implies \sum_i n_i = 0$. 
        
        Choose paths $\tau: I \to X$ for $i >0$, from a base point $x_0 \in X$ to $\sigma_i(v_0)$ and define $\sigma_0(v_0) = x_0$. We can view $\tau_i: [v_0, v_i] \to X$ as a singular $1$-simplex. Thus, $\delta(\tau_i) = \sigma_i(v_0) - x_0$. Hence, 
        \[\d\left(\sum_i n_i \tau_i\right) = \sum_i n_i \sigma_i(v_0) - \sum_i n_i \cdot x_0 = \sum_i n_i \sigma_i(v_0) \implies \sum_i n_i \sigma_i \in \im(\d_i)\]

        Thus $\ker(\varepsilon) = \im(\d_1)$. 
    \end{proof} 
    That is, 
    \[H_0(X) \iso \frac{C_0(X)}{\im(\d_1)} = \frac{C_0(X)}{\ker(\varepsilon)} \iso \Z\]
\end{proof}

\begin{proposition}
    If $X$ is a point, then $H_n(X)\iso 0$ for $n > 0$ and $H_0(X) \iso \Z$.  
\end{proposition}

\begin{proof}
     Let $x_0$ be a point. Then the constant map $\sigma_n: \Delta^n \to X$ gives us a $n$-simplex for all $n$ since constant maps are continuous. Since the restriction of the constant map is the constant map we get, 
    \[\d_n(\sigma_n) = \sum_{i=0}^n (-1)^i \sigma_{n-1} = \sigma_{n-1} \sum_{i=0}^n (-1)^i = \begin{cases*}
        0 ,& n \text{ odd} \\ 
        \sigma_{n-1}, & n \text{ even} \\ 
    \end{cases*}\]

    Thus we get, 
    \[\cdots \xrightarrow{\d_3} C_2(x_0) \xrightarrow{\d_2} C_1(x_0) \xrightarrow{\d_1} C_0(x_0) \xrightarrow{\d_0} 0 \]
    \[\cdots \xrightarrow{0} \Z \xrightarrow{\iso} \Z \xrightarrow{0} \Z \xrightarrow{} 0 \]
    Then for odd $n$ we have, 
    $\ker(\d_n) = \Z$ and $\im(\d_{n+1}) = \im(\id_{C_{n-1}}) = \Z$, so
    \[H_n(x_0) = \frac{\ker(\d_n)}{\im(\d_{n+1})} = \frac{\Z}{\Z} = \{0\}\]
    For even $n > 0$, we have
    $\ker(\d_n) = 0$,  so
    \[H_n(x_0) = \frac{\ker(\d_n)}{\im(\d_{n+1})} = \{0\}\]
    For $n = 0$, we have,
    \[H_0(x_0) = \frac{\ker(\d_0)}{\im(\d_{1})} = \frac{\Z}{0} = \Z\]

    Hence, the singular homology groups of a point are $0$ for $n > 0$ and $\Z$ for $n = 0$. 
\end{proof}

\begin{definition}
    We will change the definition of homology so that a point has trivial $H_0(-)$. The \textbf{reduced homology groups} $\widetilde{H_n}(X)$ is defined to be the homology of the following chain complex. 
    \[\to C_n(X) \xrightarrow{\d_n} C_{n-1}(X) \xrightarrow{\d_{n-1}}  \cdots \xrightarrow{\d_1} C_0(X) \xrightarrow{\varepsilon} \Z \xrightarrow{0} 0\]
    So, $\widetilde{H_n} (X) \iso H_n(X)$ if $n \geq 1$. Moreover we have $\im(\d_1) \subseteq \ker(\varepsilon)$. So, 
    \begin{align*}
        H_0(X) = \frac{C_0(X)}{\im(\d_1)} 
        \iso \frac{\ker(\varepsilon) \oplus \Z}{ \im(\d_1)} 
        = \widetilde{H_0}(X) \oplus \Z
    \end{align*}
    Hence, $\widetilde{H_0}(\text{pt}) =0$. 
\end{definition}

\begin{definition}
    If $\alpha \in C_n(X)$ we say that $\alpha$ is a \textbf{chain}. $\alpha$ is a \textbf{cycle} if $\d_n(\alpha) = 0$, and $\alpha$ is a \textbf{boundry} if $\alpha = \d(\beta)$ for some $\beta \in C_{n+1}(X)$. 
\end{definition}

\begin{definition}
    A map $\phi: C_n(X) \to C_n(Y)$ is called a \textbf{chain map} if $\d(\phi) = \phi(\d)$. Such a $\phi$ always induces a map in homology. 
\end{definition}

\begin{proposition} [Homotopy Invariance of Singular Homology]
    Let $f: X \to Y$ then $\exists$ an induced homomorphism defined by 
    \[f_{\#}: C_n(X) \to C_n(Y), \quad \sigma \mapsto f \circ \sigma\]
    which extends $\Z$-linearly.  This map satisfies $f_{\#}\d = \d f_{\#}$. 
\end{proposition}

\begin{proof}
    For some $\sigma \in C_n(X)$. 
    \begin{align*}
        f_{\#}\d(\sigma) &= f_{\#} \left(\sum_i (-1)^i \sigma | [v_0, \ldots, \widehat{v_i}, \ldots, v_n]\right) \\ 
        &= \sum_i (-1)^i (f \circ \sigma) \mid_{[v_0, \ldots, \widehat{v_i}, \ldots, v_n]} \\ 
        &= \d f_{\#} (\sigma)
    \end{align*}

    Hence $f_{\#}$ induces a map $f_\star: H_n(X) \to H_n(Y)$. 
    \begin{proof}
        Note $f_{\#} (\ker \d_{n_X}) \subset \ker \d_{n_Y}$. Since $\alpha \in C_n(X)$ such that $\d(\alpha) = 0$. Then, 
        \[f_{\#}(\d_X(\alpha)) = 0 = \d(f_{\#} (\alpha)) \implies f_{\#}(\alpha) \in \ker(\d_{n_Y})\]
        Again note that $f_{\#} (\im \d_{{n+1}_X}) \subset \im \d_{{n+1}_Y}$. Let $\alpha \in C_n(X)$ and $\beta \in C_{n+1}(\beta)$ such that $\alpha = \d (\beta)$. Then, 
        \[f_{\#}(\alpha) = 0 = f_{\#} \d (\beta) = \d(f_{\#}\beta) \implies f_{\#}(\alpha) \in \im (\d_{n+1})\]

        Since $f_{\#}$ sends $\ker(\d_n)$ to $\ker(\d_n)$ and $\im(\d_{n+1})$ to $\im(\d_{n+1})$ it induces a map $f_\star: H_n(X) \to H_n(Y)$. 
    \end{proof}
\end{proof}

\begin{theorem}
    If $f, g: X \to Y$ are homotopic, then they induce the same map on homology, 
    \[f_\star = g_\star : H_n(X) \to H_n(Y)\]
\end{theorem}

\begin{proof}
    The idea is to subdivide $\Delta^n \times I $ into $(n+1)$-simplicies. Let's say $\Delta^n \times \{0\} = [v_0, \ldots, v_n]$, $\Delta^n \times \{1\} = [w_0, \ldots, w_n]$. We start with $[v_0, \ldots, v_n]$ and consider the $(n+1)$-simplex $[v_0, v_n, w_n]$. Here we keep track of a distinguished edge from $v_n \to w_n$. Note that by contracting along the distinguished edge we go from the $[v_0, \ldots v_n]$-face to $[v_0, \ldots, v_{n-1}, w_n]$-face of the $(n+1)$-simplex. We say that $[v_0, \ldots v_n]$ is the lower face and $[v_0, \ldots, v_{n-1}, w_n]$ the upper face. 

    In a typical step, we will construct a $(n+1)$-simplex $[v_0, \ldots, v_i, w_i, \ldots, w_n]$ with lower face $[v_0, \ldots, v_i, w_{i+1}, \ldots, w_n]$ and upper face  $[v_0, \ldots, v_{i-1}, w_i, \ldots, w_n]$ which we get by contracting along the edge from $v_i$ to $w_i$. 
\end{proof}

\begin{proposition}
     Now given $F: X \times I \to Y$  a homotopy we can define for any singular $n$-simplex $\sigma: \Delta^n \to X$ the map, 
    \[F': F \circ (\sigma \times \Id) : \Delta^n \times I \to Y\] 
    Now define the \textbf{prism operator} $P: C_n(X) \to C_{n+1}(Y)$ by 

    \[P(\sigma) = \sum_i(-1)^i F' \mid_{[v_0, \ldots, v_i, w_i, \ldots, w_n]}\]

    It will turn out that $P$ is not a chain map but 
    \[\d P = g_{\#} - f_{\#} - P \d\]
\end{proposition}

\begin{proof}
    Let $\sigma: \Delta^n \to X$. Then, 
    \begin{align*}
        \d P(\alpha) &= \sum_{j \leq i} (-1)^i(-1)^j F' \mid_{[v_0, \ldots, \widehat{v_j}, \ldots, v_i, w_i, \ldots, w_n]} + \sum_{j \geq i} (-1)^i(-1)^{j+1} F' \mid_{[v_0, \ldots, v_i, w_i, \ldots, \widehat{v_j},\ldots, w_n]}
    \end{align*}

\begin{theorem}

    \[\ldots \to \widetilde{H_n}(A)\xrightarrow{i_\star} \widetilde{H_n}(X) \xrightarrow{j_\star} \widetilde{H_n}(X/A) \xrightarrow{\d} \widetilde{H_{n-1}}(A) \xrightarrow{i_\star} \widetilde{H_{n-1}}(X) \xrightarrow{j_\star} \widetilde{H_{n-1}}(X/A) \xrightarrow{\d}\ldots\]

\end{theorem}

\end{corollary}

\begin{proof}

\end{proof}

\begin{theorem}
    If nonempty open sets $U \subseteq \mathbb{R}^n$, $V \subseteq \mathbb{R}^m$ are homeomorphic, then $n = m$.
\end{theorem}

\begin{proof}
    For $x \in U$, consider $\mathbb{R}^n = U \cup \mathbb{R}^n \setminus \{x\}$. Then by excision,
    \[
        H_k(U, U \setminus \{x\}) \cong H_k(\mathbb{R}^n, \mathbb{R}^n \setminus \{x\}) \quad \forall k.
    \]
    Compute $H_k(\mathbb{R}^n, \mathbb{R}^n \setminus \{x\})$ via the LES:
    \[
        \to H_k(\mathbb{R}^n \setminus \{x\}) \to H_k(\mathbb{R}^n) \to H_k(\mathbb{R}^n, \mathbb{R}^n \setminus \{x\}) \to H_{k-1}(\mathbb{R}^n \setminus \{x\}) \to H_k(\mathbb{R}^n)
    \]
    Since $\mathbb{R}^n \setminus \{x\} \simeq S^{n-1}$ and $\mathbb{R}^n$ is contractible:
    \[
        H_k(S^{n-1}) \to 0 \to H_k(\mathbb{R}^n, \mathbb{R}^n \setminus \{x\}) \xrightarrow{\iso} H_k(S^{n-1}) \to 0
    \]
    So $H_k(\mathbb{R}^n, \mathbb{R}^n \setminus \{x\}) \cong H_k(S^{n-1}) \cong \mathbb{Z}$ if $k = n$, and $0$ otherwise.

    If $f: U \to V$ is a homeomorphism, then $f$ induces $f_*: (U, U\setminus\{x\}) \xrightarrow{\mathrm{homeo}} (V, V \setminus \{f(x)\})$, so
    \[
        f_*: H_k(U, U\setminus\{x\}) \to H_k(V, V\setminus\{f(x)\})
    \]
    is an isomorphism, giving $H_k(\mathbb{R}^n, \mathbb{R}^n\setminus\{x\}) \cong H_k(\mathbb{R}^m, \mathbb{R}^m\setminus\{f(x)\})$.

    The LHS is $\mathbb{Z}$ when $k=n$ and $0$ otherwise; the RHS is $\mathbb{Z}$ when $k=m$ and $0$ otherwise. Therefore $n=m$.
\end{proof}

\section{Equivalence of Simplicial and Singular Homology}

Put a $\Delta$-complex structure on $X$, then show $H_n^\Delta(X) \cong H_n(X)$.

To show this, we need the following definition.

\begin{definition}
    If $A \subseteq X$ is a subspace, the \textbf{relative simplicial chain complex} is
    \[
        \Delta_n(X,A) := \Delta_n(X)/\Delta_n(A),
    \]
    with $\partial: \Delta_n(X) \to \Delta_{n-1}(X)$ inducing $\partial: \Delta_n(X,A) \to \Delta_{n-1}(X,A)$. Define
    \[
        H_n^\Delta(X,A) := \ker\partial / \mathrm{im}\,\partial.
    \]
    We get a SES $0 \to \Delta_n(A) \to \Delta_n(X) \to \Delta_n(X,A) \to 0$ and hence a LES in homology $H_n^\Delta$ \quad $(*)$.
\end{definition}

Recall $\Delta_n(X) = \{\text{maps } \Delta^n \to X\}$ with some nice properties, and $C_n(X) = \{\text{maps } \Delta^n \to X\}$.

There is a canonical map $\phi: \Delta_n(X) \to C_n(X)$, $(\Delta^n \to X) \mapsto (\Delta^n \to X)$, forgetting the nice properties.

\textbf{Claim:} $\phi$ induces an isomorphism $H_n^\Delta(X) \xrightarrow{\sim} H_n(X)$.

Today we prove this claim for the case where $X$ is finite-dimensional, i.e.\ $X^k = \varnothing$ for all $k \geq k_0$ ($X^k$ is the $k$-skeleton, using simplices of dimension $\leq k$).

\begin{proof}
We have a commutative diagram with LES rows:
\begin{align*}
    &\to H_{n+1}^\Delta(X^k, X^{k-1}) \to H_n^\Delta(X^{k-1}) \to H_n^\Delta(X^k) \to H_n^\Delta(X^k, X^{k-1}) \to H_{n-1}^\Delta(X^{k-1}) \to \\
    &\quad\quad\quad \downarrow\phi_* \quad\quad\quad\quad \downarrow\phi_* \quad\quad\quad\quad \downarrow\phi_* \quad\quad\quad\quad \downarrow\phi_* \quad\quad\quad\quad \downarrow\phi_* \\
    &\to H_{n+1}(X^k, X^{k-1}) \to H_n(X^{k-1}) \to H_n(X^k) \to H_n(X^k, X^{k-1}) \to H_{n-1}(X^{k-1}) \to
\end{align*}
We show maps \textcircled{1} and \textcircled{4} are isomorphisms.

Note $\Delta_n(X^k, X^{k-1}) \cong \Delta_n(X^k)/\Delta_n(X^{k-1}) = \begin{cases} \Delta_n(X^k) & \text{if } n=k \\ 0 & \text{otherwise} \end{cases}$.

Now if $d \in \Delta_n(X^n)$, then $\partial d \in \Delta_{n-1}(X^{n-1})$ by face properties, so $\partial_\mathrm{rel} = \Delta_n(X^n) \to 0$ implies $\mathrm{Im}\, \partial_\mathrm{rel} = 0$. Since $\partial(\cdot) = 0$, $H_n^\Delta(X^k, X^{k-1})$ is free abelian for $n=k$ and $0$ otherwise.

Now consider $\bigsqcup_\alpha \Delta^k_\alpha / \bigsqcup_\alpha \partial \Delta^k_\alpha \xrightarrow{\sim} (X^k, X^{k-1})$.

But $\bigsqcup_\alpha \Delta^k_\alpha / \bigsqcup_\alpha \partial \Delta^k_\alpha \cong X^k/X^{k-1}$ by quotient map / homeomorphism ($\cong$ means homeomorphism here).

\textbf{$n$-simplices} (quotient by boundary): $\bigvee_\alpha S^k := \{\bigsqcup_\alpha S^k / x_\alpha \in S_\alpha^k \times \{\beta\}, \forall \alpha, \beta\}$ (\emph{wedge product}), so:
\[
    H_n\!\left(\bigvee_\alpha S^k\right) \cong \bigoplus_\alpha H_n(S^k) \to \text{free abelian for } n=k, \; 0 \text{ otherwise.}
\]
So the domains/codomains of \textcircled{1} and \textcircled{4} in the diagram are isomorphic. Why, induced by $\phi$, though? Because $\phi$ is almost the ``identity'' map; the isomorphism is the ``same'' map, so we use it.

By the Five Lemma (applied to maps \textcircled{2} and \textcircled{3} isomorphisms), we get \textcircled{2} and \textcircled{5} isomorphisms. The five lemma says: if you have an exact sequence
\[
    A \to B \to C \to D \to E, \qquad A\to A',\; B\to B',\; D\to D',\; E\to E' \text{isomorphisms},
\]
then $C \to C'$ is also an isomorphism.

Repeat until $X = X^{k_0}$, and remaining things are $0$. Done.
\end{proof}

\subsection*{Mayer--Vietoris Sequence}

Suppose $X = A \cup B$, with $C_n(A+B) \hookrightarrow C_n(X)$.

Claim: there is a SES
\[
    0 \to C_n(A \cap B) \xrightarrow{\phi} C_n(A) \oplus C_n(B) \xrightarrow{\psi} C_n(A+B) \to 0
\]
where $\phi(x) = (x, -x)$ and $\psi(x,y) = x+y$.

If we prove it's exact, we'll get a LES in homology.
\begin{itemize}
    \item $\phi$ is injective since $\phi(x) = 0 \Rightarrow x = 0$.
    \item $\ker\psi = \{(x,y) \mid x+y=0\} = \{(x,-x) \mid x \in C_n(A), -x \in C_n(B)\} = \{(x,-x) \mid x \in C_n(A \cap B)\} = \mathrm{im}\,\phi$.
    \item $\psi$ is surjective (from definition of $C_n(A+B)$).
\end{itemize}

Thus the sequence is exact, so we get a LES in homology, called the \textbf{Mayer--Vietoris sequence}:
\[
    \cdots \to H_n(A \cap B) \to H_n(A) \oplus H_n(B) \xrightarrow{\psi_*} H_n(A+B) \xrightarrow{\partial} H_{n-1}(A \cap B) \to \cdots
\]
(also has a reduced version.)

\textbf{Note:} If $A \cap B$ is path-connected, then
\[
    \widetilde{H}_1(A \cap B) \xrightarrow{\phi_*} \widetilde{H}_1(A) \oplus \widetilde{H}_1(B) \xrightarrow{\psi_*} \widetilde{H}_1(X) \to \widetilde{H}_0(A \cap B)^0
\]
then $H_1(X) \cong H_1(A) \oplus H_1(B) / \ker\psi_* \cong H_1(A) \oplus H_1(B) / \mathrm{im}\,\phi_*$. (Similar to Van Kampen's theorem.)
